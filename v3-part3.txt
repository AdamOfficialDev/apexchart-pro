function hexWithOpacity(hex: string, opacityPct: number): string {
  const alpha = Math.max(0, Math.min(1, opacityPct / 100));
  const clean = hex.replace('#', '');
  if (clean.length === 3) {
    const r = parseInt(clean[0] + clean[0], 16);
    const g = parseInt(clean[1] + clean[1], 16);
    const b = parseInt(clean[2] + clean[2], 16);
    return `rgba(${r}, ${g}, ${b}, ${alpha.toFixed(2)})`;
  }
  if (clean.length === 6) {
    const r = parseInt(clean.slice(0, 2), 16);
    const g = parseInt(clean.slice(2, 4), 16);
    const b = parseInt(clean.slice(4, 6), 16);
    return `rgba(${r}, ${g}, ${b}, ${alpha.toFixed(2)})`;
  }
  return hex;
}

function distToSegment(p: ScreenPoint, a: ScreenPoint, b: ScreenPoint): number {
  const dx = b.x - a.x;
  const dy = b.y - a.y;
  const lenSq = dx * dx + dy * dy;
  if (lenSq === 0) return Math.hypot(p.x - a.x, p.y - a.y);
  const t = Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / lenSq));
  return Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy));
}

function drawArrowHead(
  ctx: CanvasRenderingContext2D,
  from: ScreenPoint,
  to: ScreenPoint,
  size = 10
) {
  const angle = Math.atan2(to.y - from.y, to.x - from.x);
  ctx.beginPath();
  ctx.moveTo(to.x, to.y);
  ctx.lineTo(
    to.x - size * Math.cos(angle - Math.PI / 7),
    to.y - size * Math.sin(angle - Math.PI / 7)
  );
  ctx.lineTo(
    to.x - size * Math.cos(angle + Math.PI / 7),
    to.y - size * Math.sin(angle + Math.PI / 7)
  );
  ctx.closePath();
  ctx.fill();
}

        }

        for (const pv of structureData.pivots) {
          const sp = toScreen({ time: pv.time, price: pv.price });
          if (!sp) continue;
          const isHigh = pv.kind === 'high';
          ctx.font = '700 9px "JetBrains Mono", monospace';
          ctx.fillStyle = pv.label === 'HH' || pv.label === 'HL' ? '#00D4AA' : '#F23645';
          ctx.fillText(pv.label, sp.x - 8, isHigh ? sp.y - 6 : sp.y + 12);
        }
        ctx.restore();
      }
    }

    const drawSingle = (
      type: Exclude<DrawingToolType, 'cursor' | 'eraser'>,
      ptsData: DrawingPoint[],
      color: string,
      lineWidth: number,
      lineStyle: 'solid' | 'dashed' | 'dotted',
      isSelected: boolean,
      obj?: Partial<DrawingObject>
    ) => {
      const screenPts = ptsData.map(toScreen);
      if (screenPts.some((s) => s === null)) return;
      const pts = screenPts as ScreenPoint[];

      const strokeOp = obj?.opacity ?? 100;
      const strokeCol = hexWithOpacity(color, strokeOp);
      const fillBase = obj?.fillColor ?? color;
      const fillOp = obj?.showFill === false ? 0 : obj?.fillOpacity ?? 15;
      const fillCol = hexWithOpacity(fillBase, fillOp);
      const fontSpec = `${obj?.fontItalic ? 'italic ' : ''}${
        obj?.fontBold ? '700 ' : '600 '
      }${obj?.fontSize ?? 11}px Inter, sans-serif`;
      const txtCol = obj?.textColor ?? color;

      ctx.save();
      ctx.strokeStyle = strokeCol;
      ctx.fillStyle = strokeCol;
      ctx.lineWidth = lineWidth;
      ctx.lineCap = 'round';
      ctx.lineJoin = 'round';

      if (lineStyle === 'dashed') ctx.setLineDash([7, 5]);
      else if (lineStyle === 'dotted') ctx.setLineDash([2, 4]);
      else ctx.setLineDash([]);

      if ((type === 'trendline' || type === 'info_line') && pts.length >= 2) {
        const dx = pts[1].x - pts[0].x;
        const dy = pts[1].y - pts[0].y;
        const len = Math.hypot(dx, dy) || 1;
        const scale = Math.max(width, height) * 2.5;
        const pStart = obj?.extendLeft
          ? { x: pts[0].x - (dx / len) * scale, y: pts[0].y - (dy / len) * scale }
          : pts[0];
        const pEnd = obj?.extendRight
          ? { x: pts[1].x + (dx / len) * scale, y: pts[1].y + (dy / len) * scale }
          : pts[1];

        ctx.beginPath();
        ctx.moveTo(pStart.x, pStart.y);
        ctx.lineTo(pEnd.x, pEnd.y);
        ctx.stroke();

        if (obj?.text) {
          ctx.setLineDash([]);
          ctx.font = fontSpec;
          ctx.fillStyle = txtCol;
          const mx =
            obj.textAlign === 'left'
              ? pts[0].x + 6
              : obj.textAlign === 'right'
              ? pts[1].x - ctx.measureText(obj.text).width - 6
              : (pts[0].x + pts[1].x) / 2 - ctx.measureText(obj.text).width / 2;
          ctx.fillText(obj.text, mx, (pts[0].y + pts[1].y) / 2 - 6);
        }

        if (isSelected || type === 'info_line' || obj?.showStats) {
          ctx.setLineDash([]);
          const diff = ptsData[1].price - ptsData[0].price;
          const pct = (diff / Math.max(ptsData[0].price, 1e-9)) * 100;
          const bars = Math.round(Math.abs(ptsData[1].time - ptsData[0].time) / tfSec);
          const angle = Math.round(
            (-Math.atan2(pts[1].y - pts[0].y, pts[1].x - pts[0].x) * 180) / Math.PI
          );
          const label =
            type === 'info_line' || obj?.showStats
              ? `${diff >= 0 ? '+' : ''}${formatPrice(diff, precision)} (${
                  pct >= 0 ? '+' : ''
                }${pct.toFixed(2)}%) • ${bars}b • ${angle}°`
              : `${pct >= 0 ? '+' : ''}${pct.toFixed(2)}%`;

          ctx.font = '600 10px "JetBrains Mono", monospace';
          const tw = ctx.measureText(label).width + 12;
          const lx = (pts[0].x + pts[1].x) / 2 - tw / 2;
          const ly = (pts[0].y + pts[1].y) / 2 - 24;
          ctx.fillStyle = theme === 'dark' ? 'rgba(19,23,34,0.92)' : 'rgba(255,255,255,0.94)';
          ctx.strokeStyle = strokeCol;
          ctx.lineWidth = 1;
          ctx.beginPath();
          ctx.roundRect(lx, ly, tw, 18, 4);
          ctx.fill();
          ctx.stroke();
          ctx.fillStyle = color;
          ctx.fillText(label, lx + 6, ly + 12.5);
        }
      } else if (type === 'ray' && pts.length >= 2) {
        const dx = pts[1].x - pts[0].x;
        const dy = pts[1].y - pts[0].y;
        const len = Math.hypot(dx, dy) || 1;
        const scale = Math.max(width, height) * 2.5;
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        ctx.lineTo(pts[0].x + (dx / len) * scale, pts[0].y + (dy / len) * scale);
        ctx.stroke();
      } else if (type === 'extended_line' && pts.length >= 2) {
        const dx = pts[1].x - pts[0].x;
        const dy = pts[1].y - pts[0].y;
        const len = Math.hypot(dx, dy) || 1;
        const scale = Math.max(width, height) * 2.5;
        ctx.beginPath();
        ctx.moveTo(pts[0].x - (dx / len) * scale, pts[0].y - (dy / len) * scale);
        ctx.lineTo(pts[0].x + (dx / len) * scale, pts[0].y + (dy / len) * scale);
        ctx.stroke();
      } else if ((type === 'hline' || type === 'hray') && pts.length >= 1) {
        const startX = type === 'hray' ? pts[0].x : 0;
        ctx.beginPath();
        ctx.moveTo(startX, pts[0].y);
        ctx.lineTo(width, pts[0].y);
        ctx.stroke();
        ctx.setLineDash([]);

        if (obj?.text) {
          ctx.font = fontSpec;
          ctx.fillStyle = txtCol;
          ctx.fillText(obj.text, Math.max(plotLeft + 12, startX + 12), pts[0].y - 6);
        }

        const priceLabel = formatPrice(ptsData[0].price, precision);
        ctx.font = '600 10px "JetBrains Mono", monospace';
        const tw = ctx.measureText(priceLabel).width + 12;
        ctx.fillStyle = color;
        const tagX = isLeftScale ? 4 : width - tw - 4;
        ctx.fillRect(tagX, pts[0].y - 9, tw, 18);
        ctx.fillStyle = '#FFFFFF';
        ctx.fillText(priceLabel, tagX + 6, pts[0].y + 3.5);
      } else if (type === 'vline' && pts.length >= 1) {
        ctx.beginPath();
        ctx.moveTo(pts[0].x, 0);
        ctx.lineTo(pts[0].x, height);
        ctx.stroke();
        if (obj?.text) {
          ctx.setLineDash([]);
          ctx.font = fontSpec;
          ctx.fillStyle = txtCol;
          ctx.fillText(obj.text, pts[0].x + 6, 36);
        }
      } else if (type === 'cross_line' && pts.length >= 1) {
        ctx.beginPath();
        ctx.moveTo(0, pts[0].y);
        ctx.lineTo(width, pts[0].y);
        ctx.moveTo(pts[0].x, 0);
        ctx.lineTo(pts[0].x, height);
        ctx.stroke();
      } else if (type === 'channel' && pts.length >= 2) {
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        ctx.lineTo(pts[1].x, pts[1].y);
        ctx.stroke();
        if (pts.length >= 3) {
          const ox = pts[2].x - pts[0].x;
          const oy = pts[2].y - pts[0].y;
          const p3 = { x: pts[1].x + ox, y: pts[1].y + oy };
          ctx.beginPath();
          ctx.moveTo(pts[2].x, pts[2].y);
          ctx.lineTo(p3.x, p3.y);
          ctx.stroke();

          ctx.fillStyle = fillCol;
          ctx.beginPath();
          ctx.moveTo(pts[0].x, pts[0].y);
          ctx.lineTo(pts[1].x, pts[1].y);
          ctx.lineTo(p3.x, p3.y);
          ctx.lineTo(pts[2].x, pts[2].y);
          ctx.closePath();
          ctx.fill();

          ctx.setLineDash([4, 4]);
          ctx.beginPath();
          ctx.moveTo((pts[0].x + pts[2].x) / 2, (pts[0].y + pts[2].y) / 2);
          ctx.lineTo((pts[1].x + p3.x) / 2, (pts[1].y + p3.y) / 2);
          ctx.stroke();
        }
      } else if (type === 'regression_channel' && pts.length >= 2) {
        const reg = computeLinearRegression(candles, ptsData[0].time, ptsData[1].time);
        if (reg) {
          const m0 = toScreen({ time: reg.startTime, price: reg.midStart });
          const m1 = toScreen({ time: reg.endTime, price: reg.midEnd });
          const u0 = toScreen({ time: reg.startTime, price: reg.upperStart });
          const u1 = toScreen({ time: reg.endTime, price: reg.upperEnd });
          const l0 = toScreen({ time: reg.startTime, price: reg.lowerStart });
          const l1 = toScreen({ time: reg.endTime, price: reg.lowerEnd });
          if (m0 && m1 && u0 && u1 && l0 && l1) {
            ctx.fillStyle = fillCol;
            ctx.beginPath();
            ctx.moveTo(u0.x, u0.y);
            ctx.lineTo(u1.x, u1.y);
            ctx.lineTo(l1.x, l1.y);
            ctx.lineTo(l0.x, l0.y);
            ctx.closePath();
            ctx.fill();

            ctx.beginPath();
            ctx.moveTo(u0.x, u0.y);
            ctx.lineTo(u1.x, u1.y);
            ctx.moveTo(l0.x, l0.y);
            ctx.lineTo(l1.x, l1.y);
            ctx.stroke();

            ctx.setLineDash([5, 4]);
            ctx.beginPath();
            ctx.moveTo(m0.x, m0.y);
            ctx.lineTo(m1.x, m1.y);
            ctx.stroke();
          }
        }
      } else if (type === 'fib' && pts.length >= 2) {
        const minX = Math.min(pts[0].x, pts[1].x);
        const maxX = Math.max(pts[0].x, pts[1].x);
        const spanW = Math.max(maxX - minX, 95);
        const pStart = obj?.fibReverse ? ptsData[1].price : ptsData[0].price;
        const pEnd = obj?.fibReverse ? ptsData[0].price : ptsData[1].price;
        const priceSpan = pEnd - pStart;

        const levels = (obj?.fibLevels ?? DEFAULT_FIB_RETRACE_LEVELS).filter((l) => l.enabled);
        ctx.setLineDash([3, 3]);
        ctx.strokeStyle = 'rgba(120,123,134,0.6)';
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        ctx.lineTo(pts[1].x, pts[1].y);
        ctx.stroke();
        ctx.setLineDash([]);

        for (let i = 0; i < levels.length; i++) {
          const lvl = levels[i];
          const lvlPrice = pEnd - priceSpan * lvl.level;
          const yCoord = priceToY(lvlPrice);
          if (yCoord === null) continue;

          if (i < levels.length - 1) {
            const nextPrice = pEnd - priceSpan * levels[i + 1].level;
            const nextY = priceToY(nextPrice);
            if (nextY !== null) {
              ctx.fillStyle = hexWithOpacity(lvl.color, 7);
              ctx.fillRect(minX, Math.min(yCoord, nextY), spanW, Math.abs(nextY - yCoord));
            }
          }

          ctx.strokeStyle = lvl.color;
          ctx.lineWidth = Math.max(1, lineWidth * 0.75);
          ctx.beginPath();
          ctx.moveTo(minX, yCoord);
          ctx.lineTo(minX + spanW, yCoord);
          ctx.stroke();

          ctx.fillStyle = lvl.color;
          ctx.font = '600 10px "JetBrains Mono", monospace';
          const levelLabel = obj?.fibShowPercents
            ? `${(lvl.level * 100).toFixed(1)}%`
            : String(lvl.level);
          const textLabel =
            obj?.fibShowPrices === false
              ? levelLabel
              : `${levelLabel} (${formatPrice(lvlPrice, precision)})`;
          ctx.fillText(textLabel, minX + 6, yCoord - 3);
        }
      } else if (type === 'fib_ext' && pts.length >= 2) {
        ctx.setLineDash([3, 3]);
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        ctx.lineTo(pts[1].x, pts[1].y);
        if (pts.length >= 3) ctx.lineTo(pts[2].x, pts[2].y);
        ctx.stroke();
        ctx.setLineDash([]);

        if (pts.length >= 3) {
          const impulse = ptsData[1].price - ptsData[0].price;
          const startX = pts[2].x;
          const spanW = Math.max(120, Math.abs(pts[2].x - pts[0].x));
          const levels = (obj?.fibLevels ?? DEFAULT_FIB_EXT_LEVELS).filter((l) => l.enabled);
          for (const lvl of levels) {
            const extPrice = ptsData[2].price + impulse * lvl.level;
            const yCoord = priceToY(extPrice);
            if (yCoord === null) continue;
            ctx.strokeStyle = lvl.color;
            ctx.lineWidth = 1.25;
            ctx.beginPath();
            ctx.moveTo(startX, yCoord);
            ctx.lineTo(startX + spanW, yCoord);
            ctx.stroke();
            ctx.fillStyle = lvl.color;
            ctx.font = '600 10px "JetBrains Mono", monospace';
            const label =
              obj?.fibShowPrices === false
                ? String(lvl.level)
                : `${lvl.level} (${formatPrice(extPrice, precision)})`;
            ctx.fillText(label, startX + 6, yCoord - 3);
          }
        }
      } else if (type === 'fib_channel' && pts.length >= 2) {
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        ctx.lineTo(pts[1].x, pts[1].y);
        ctx.stroke();
        if (pts.length >= 3) {
          const ox = pts[2].x - pts[0].x;
          const oy = pts[2].y - pts[0].y;
          const levels = [0, 0.236, 0.382, 0.5, 0.618, 0.786, 1.0, 1.618];
          for (const lv of levels) {
            ctx.strokeStyle = lv === 0.618 || lv === 0.5 ? '#00D4AA' : strokeCol;
            ctx.lineWidth = lv === 0 || lv === 1 ? lineWidth : 1;
            ctx.beginPath();
            ctx.moveTo(pts[0].x + ox * lv, pts[0].y + oy * lv);
            ctx.lineTo(pts[1].x + ox * lv, pts[1].y + oy * lv);
            ctx.stroke();
            ctx.font = '600 9px "JetBrains Mono", monospace';
            ctx.fillStyle = ctx.strokeStyle;
            ctx.fillText(String(lv), pts[1].x + ox * lv + 4, pts[1].y + oy * lv);
          }
        }
      } else if (type === 'fib_timezone' && pts.length >= 2) {
        const dt = ptsData[1].time - ptsData[0].time;
        if (Math.abs(dt) > 0) {
          for (const n of FIB_TIME_NUMS) {
            const x = timeToX(ptsData[0].time + dt * n);
            if (x === null) continue;
            ctx.strokeStyle = n === 0 || n === 1 ? strokeCol : '#00D4AA';
            ctx.lineWidth = 1.2;
            ctx.setLineDash(n <= 1 ? [] : [4, 4]);
            ctx.beginPath();
            ctx.moveTo(x, 0);
            ctx.lineTo(x, height);
            ctx.stroke();
            ctx.font = '700 10px "JetBrains Mono", monospace';
            ctx.fillStyle = ctx.strokeStyle;
            ctx.fillText(String(n), x + 4, 24);
          }
        }
      } else if (type === 'fib_fan' && pts.length >= 2) {
        ctx.setLineDash([3, 3]);
        ctx.strokeRect(
          Math.min(pts[0].x, pts[1].x),
          Math.min(pts[0].y, pts[1].y),
          Math.abs(pts[1].x - pts[0].x),
          Math.abs(pts[1].y - pts[0].y)
        );
        ctx.setLineDash([]);
        const ratios = [0.236, 0.382, 0.5, 0.618, 0.786, 1.0];
        const scale = Math.max(width, height) * 2;
        for (const r of ratios) {
          const targetY = pts[0].y + (pts[1].y - pts[0].y) * r;
          const dx = pts[1].x - pts[0].x;
          const dy = targetY - pts[0].y;
          const len = Math.hypot(dx, dy) || 1;
          ctx.strokeStyle = r === 0.618 ? '#00D4AA' : strokeCol;
          ctx.lineWidth = 1.2;
          ctx.beginPath();
          ctx.moveTo(pts[0].x, pts[0].y);
          ctx.lineTo(pts[0].x + (dx / len) * scale, pts[0].y + (dy / len) * scale);
          ctx.stroke();
        }
      } else if (type === 'fib_arc' && pts.length >= 2) {
        ctx.setLineDash([3, 3]);
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        ctx.lineTo(pts[1].x, pts[1].y);
        ctx.stroke();
        ctx.setLineDash([]);
        const baseR = Math.hypot(pts[1].x - pts[0].x, pts[1].y - pts[0].y);
        const ratios = [0.382, 0.5, 0.618, 0.786, 1.0, 1.618];
        for (const r of ratios) {
          ctx.strokeStyle = r === 0.618 ? '#00D4AA' : strokeCol;
          ctx.lineWidth = 1.2;
          ctx.beginPath();
          ctx.arc(pts[0].x, pts[0].y, baseR * r, 0, Math.PI * 2);
          ctx.stroke();
        }
      } else if (type === 'pitchfork' && pts.length >= 2) {
        ctx.setLineDash([3, 3]);
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        ctx.lineTo(pts[1].x, pts[1].y);
        if (pts.length >= 3) {
          ctx.moveTo(pts[1].x, pts[1].y);
          ctx.lineTo(pts[2].x, pts[2].y);
        }
        ctx.stroke();
        ctx.setLineDash([]);

        if (pts.length >= 3) {
          const mid = { x: (pts[1].x + pts[2].x) / 2, y: (pts[1].y + pts[2].y) / 2 };
          const dx = mid.x - pts[0].x;
          const dy = mid.y - pts[0].y;
          const len = Math.hypot(dx, dy) || 1;
          const scale = Math.max(width, height) * 2;
          const ux = (dx / len) * scale;
          const uy = (dy / len) * scale;

          ctx.strokeStyle = '#F23645';
          ctx.lineWidth = lineWidth;
          ctx.beginPath();
          ctx.moveTo(pts[0].x, pts[0].y);
          ctx.lineTo(pts[0].x + ux, pts[0].y + uy);
          ctx.stroke();

          ctx.strokeStyle = strokeCol;
          ctx.beginPath();
          ctx.moveTo(pts[1].x, pts[1].y);
          ctx.lineTo(pts[1].x + ux, pts[1].y + uy);
          ctx.moveTo(pts[2].x, pts[2].y);
          ctx.lineTo(pts[2].x + ux, pts[2].y + uy);
          ctx.stroke();

          ctx.fillStyle = fillCol;
          ctx.beginPath();
          ctx.moveTo(pts[1].x, pts[1].y);
          ctx.lineTo(pts[1].x + ux, pts[1].y + uy);
          ctx.lineTo(pts[2].x + ux, pts[2].y + uy);
          ctx.lineTo(pts[2].x, pts[2].y);
          ctx.closePath();
          ctx.fill();
        }
      } else if (type === 'gann_box' && pts.length >= 2) {
        const x = Math.min(pts[0].x, pts[1].x);
        const y = Math.min(pts[0].y, pts[1].y);
        const w = Math.abs(pts[1].x - pts[0].x);
        const h = Math.abs(pts[1].y - pts[0].y);
        ctx.fillStyle = fillCol;
        ctx.fillRect(x, y, w, h);
        ctx.strokeRect(x, y, w, h);

        const divs = [0.25, 0.5, 0.75];
        ctx.lineWidth = 1;
        for (const d of divs) {
          ctx.setLineDash(d === 0.5 ? [] : [4, 4]);
          ctx.beginPath();
          ctx.moveTo(x + w * d, y);
          ctx.lineTo(x + w * d, y + h);
          ctx.moveTo(x, y + h * d);
          ctx.lineTo(x + w, y + h * d);
          ctx.stroke();
        }
        ctx.setLineDash([]);
        ctx.strokeStyle = '#F59E0B';
        ctx.beginPath();
        ctx.moveTo(x, y + h);
        ctx.lineTo(x + w, y);
        ctx.moveTo(x, y);
        ctx.lineTo(x + w, y + h);
        ctx.moveTo(x, y + h);
        ctx.lineTo(x + w, y + h * 0.5);
        ctx.moveTo(x, y + h);
        ctx.lineTo(x + w * 0.5, y);
        ctx.stroke();
      } else if ((type === 'xabcd' || type === 'abcd') && pts.length >= 2) {
        const labels = type === 'xabcd' ? ['X', 'A', 'B', 'C', 'D'] : ['A', 'B', 'C', 'D'];
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i].x, pts[i].y);
        ctx.stroke();

        if (pts.length >= 3) {
          ctx.fillStyle = fillCol;
          ctx.beginPath();
          ctx.moveTo(pts[0].x, pts[0].y);
          ctx.lineTo(pts[1].x, pts[1].y);
          ctx.lineTo(pts[2].x, pts[2].y);
          ctx.closePath();
          ctx.fill();
        }
        if (pts.length >= 5) {
          ctx.fillStyle = fillCol;
          ctx.beginPath();
          ctx.moveTo(pts[2].x, pts[2].y);
          ctx.lineTo(pts[3].x, pts[3].y);
          ctx.lineTo(pts[4].x, pts[4].y);
          ctx.closePath();
          ctx.fill();
        }

        ctx.setLineDash([4, 4]);
        ctx.font = '700 10px "JetBrains Mono", monospace';
        if (pts.length >= 3) {
          const xa = Math.abs(ptsData[1].price - ptsData[0].price);
          const ab = Math.abs(ptsData[2].price - ptsData[1].price);
          const ratio = xa > 0 ? ab / xa : 0;
          ctx.beginPath();
          ctx.moveTo(pts[0].x, pts[0].y);
          ctx.lineTo(pts[2].x, pts[2].y);
          ctx.stroke();
          ctx.fillStyle = '#00D4AA';
          ctx.fillText(ratio.toFixed(3), (pts[0].x + pts[2].x) / 2, (pts[0].y + pts[2].y) / 2 - 4);
        }
        if (pts.length >= 4) {
          const ab = Math.abs(ptsData[2].price - ptsData[1].price);
          const bc = Math.abs(ptsData[3].price - ptsData[2].price);
          const ratio = ab > 0 ? bc / ab : 0;
          ctx.beginPath();
          ctx.moveTo(pts[1].x, pts[1].y);
          ctx.lineTo(pts[3].x, pts[3].y);
          ctx.stroke();
          ctx.fillStyle = '#F59E0B';
          ctx.fillText(ratio.toFixed(3), (pts[1].x + pts[3].x) / 2, (pts[1].y + pts[3].y) / 2 - 4);
        }
        if (pts.length >= 5) {
          const bc = Math.abs(ptsData[3].price - ptsData[2].price);
          const cd = Math.abs(ptsData[4].price - ptsData[3].price);
          const ratio = bc > 0 ? cd / bc : 0;
          ctx.beginPath();
          ctx.moveTo(pts[2].x, pts[2].y);
          ctx.lineTo(pts[4].x, pts[4].y);
          ctx.stroke();
          ctx.fillStyle = '#EC4899';
          ctx.fillText(ratio.toFixed(3), (pts[2].x + pts[4].x) / 2, (pts[2].y + pts[4].y) / 2 - 4);
        }
        ctx.setLineDash([]);

        ctx.font = '700 11px Inter, sans-serif';
        pts.forEach((pt, idx) => {
          ctx.fillStyle = color;
          ctx.fillText(labels[idx] ?? String(idx), pt.x - 4, pt.y - 8);
        });
      } else if ((type === 'elliott_impulse' || type === 'elliott_abc') && pts.length >= 2) {
        const labels =
          type === 'elliott_impulse' ? ['(0)', '(1)', '(2)', '(3)', '(4)', '(5)'] : ['(0)', '(A)', '(B)', '(C)'];
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i].x, pts[i].y);
        ctx.stroke();
        ctx.font = '700 11px "JetBrains Mono", monospace';
        pts.forEach((pt, idx) => {
          ctx.fillStyle = color;
          ctx.fillText(labels[idx] ?? String(idx), pt.x - 9, pt.y - 8);
        });
      } else if (type === 'head_shoulders' && pts.length >= 2) {
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i].x, pts[i].y);
        ctx.stroke();
        if (pts.length >= 5) {
          ctx.setLineDash([5, 4]);
          ctx.strokeStyle = '#F59E0B';
          ctx.beginPath();
          ctx.moveTo(pts[0].x, pts[2].y);
          ctx.lineTo(pts[pts.length - 1].x, pts[4].y);
          ctx.stroke();
          ctx.setLineDash([]);
        }
        const hsLabels = ['L', 'LS', 'NL', 'HEAD', 'NL', 'RS', 'R'];
        ctx.font = '700 10px Inter, sans-serif';
        pts.forEach((pt, idx) => {
          ctx.fillStyle = idx === 3 ? '#F23645' : color;
          ctx.fillText(hsLabels[idx] ?? '', pt.x - 10, pt.y - 8);
        });
      } else if (type === 'bar_pattern' && pts.length >= 2) {
        const minT = Math.min(ptsData[0].time, ptsData[1].time);
        const maxT = Math.max(ptsData[0].time, ptsData[1].time);
        const srcBars = candles.filter((c) => c.time >= minT && c.time <= maxT);
        const x1 = Math.min(pts[0].x, pts[1].x);
        const x2 = Math.max(pts[0].x, pts[1].x);
        const y1 = Math.min(pts[0].y, pts[1].y);
        const y2 = Math.max(pts[0].y, pts[1].y);

        ctx.fillStyle = 'rgba(41, 98, 255, 0.1)';
        ctx.strokeStyle = 'rgba(41, 98, 255, 0.5)';
        ctx.setLineDash([4, 3]);
        ctx.fillRect(x1, y1, Math.max(12, x2 - x1), Math.max(12, y2 - y1));
        ctx.strokeRect(x1, y1, Math.max(12, x2 - x1), Math.max(12, y2 - y1));
        ctx.setLineDash([]);

        if (pts.length >= 3 && srcBars.length > 0) {
          const baseSrcOpen = srcBars[0].open;
          const anchorPrice = ptsData[2].price;
          const anchorTime = ptsData[2].time;
          const barW = Math.max(3, ((x2 - x1) / Math.max(srcBars.length, 1)) * 0.65);

          for (let i = 0; i < srcBars.length; i++) {
            const b = srcBars[i];
            const tOffset = b.time - srcBars[0].time;
            const gx = timeToX(anchorTime + tOffset);
            if (gx === null) continue;
            const go = priceToY(anchorPrice * (b.open / baseSrcOpen));
            const gc = priceToY(anchorPrice * (b.close / baseSrcOpen));
            const gh = priceToY(anchorPrice * (b.high / baseSrcOpen));
            const gl = priceToY(anchorPrice * (b.low / baseSrcOpen));
            if (go === null || gc === null || gh === null || gl === null) continue;

            const isUp = b.close >= b.open;
            ctx.strokeStyle = isUp ? 'rgba(0, 212, 170, 0.7)' : 'rgba(242, 54, 69, 0.7)';
            ctx.fillStyle = isUp ? 'rgba(0, 212, 170, 0.4)' : 'rgba(242, 54, 69, 0.4)';
            ctx.beginPath();
            ctx.moveTo(gx, gh);
            ctx.lineTo(gx, gl);
            ctx.stroke();
            ctx.fillRect(gx - barW / 2, Math.min(go, gc), barW, Math.max(1.5, Math.abs(gc - go)));
          }
        }
      } else if (type === 'rect' && pts.length >= 2) {
        const x = Math.min(pts[0].x, pts[1].x);
        const y = Math.min(pts[0].y, pts[1].y);
        const w = Math.abs(pts[1].x - pts[0].x);
        const h = Math.abs(pts[1].y - pts[0].y);
        ctx.fillStyle = fillCol;
        ctx.fillRect(x, y, w, h);
        ctx.strokeRect(x, y, w, h);
        if (obj?.showMidline !== false) {
          ctx.setLineDash([4, 4]);
          ctx.beginPath();
          ctx.moveTo(x, y + h / 2);
          ctx.lineTo(x + w, y + h / 2);
          ctx.stroke();
        }
        if (obj?.text) {
          ctx.setLineDash([]);
          ctx.font = fontSpec;
          ctx.fillStyle = txtCol;
          const tx =
            obj.textAlign === 'left'
              ? x + 8
              : obj.textAlign === 'right'
              ? x + w - ctx.measureText(obj.text).width - 8
              : x + w / 2 - ctx.measureText(obj.text).width / 2;
          ctx.fillText(obj.text, tx, y + h / 2 + 4);
        }
      } else if (type === 'rotated_rect' && pts.length >= 2) {
        if (pts.length === 2) {
          ctx.beginPath();
          ctx.moveTo(pts[0].x, pts[0].y);
          ctx.lineTo(pts[1].x, pts[1].y);
          ctx.stroke();
        } else {
          const ox = pts[2].x - pts[0].x;
          const oy = pts[2].y - pts[0].y;
          const p3 = { x: pts[1].x + ox, y: pts[1].y + oy };
          ctx.fillStyle = fillCol;
          ctx.beginPath();
          ctx.moveTo(pts[0].x, pts[0].y);
          ctx.lineTo(pts[1].x, pts[1].y);
          ctx.lineTo(p3.x, p3.y);
          ctx.lineTo(pts[2].x, pts[2].y);
          ctx.closePath();
          ctx.fill();
          ctx.stroke();
        }
      } else if (type === 'ellipse' && pts.length >= 2) {
        const cx = (pts[0].x + pts[1].x) / 2;
        const cy = (pts[0].y + pts[1].y) / 2;
        const rx = Math.max(4, Math.abs(pts[1].x - pts[0].x) / 2);
        const ry = Math.max(4, Math.abs(pts[1].y - pts[0].y) / 2);
        ctx.fillStyle = fillCol;
        ctx.beginPath();
        ctx.ellipse(cx, cy, rx, ry, 0, 0, Math.PI * 2);
        ctx.fill();
        ctx.stroke();
      } else if (type === 'triangle' && pts.length >= 2) {
        ctx.fillStyle = fillCol;
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        ctx.lineTo(pts[1].x, pts[1].y);
        if (pts.length >= 3) {
          ctx.lineTo(pts[2].x, pts[2].y);
          ctx.closePath();
          ctx.fill();
        }
        ctx.stroke();
      } else if (type === 'arc' && pts.length >= 2) {
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        if (pts.length >= 3) {
          ctx.quadraticCurveTo(pts[1].x, pts[1].y, pts[2].x, pts[2].y);
        } else {
          ctx.lineTo(pts[1].x, pts[1].y);
        }
        ctx.stroke();
      } else if (type === 'polyline' && pts.length >= 2) {
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i].x, pts[i].y);
        ctx.stroke();
      } else if ((type === 'long_pos' || type === 'short_pos') && pts.length >= 3) {
        const entryPt = pts[0];
        const tpPt = pts[1];
        const slPt = pts[2];
        const boxLeft = Math.min(entryPt.x, tpPt.x);
        const boxW = Math.max(Math.abs(tpPt.x - entryPt.x), 95);

        const tpTop = Math.min(entryPt.y, tpPt.y);
        const tpH = Math.abs(tpPt.y - entryPt.y);
        ctx.fillStyle = 'rgba(8, 153, 129, 0.22)';
        ctx.fillRect(boxLeft, tpTop, boxW, tpH);
        ctx.strokeStyle = '#089981';
        ctx.lineWidth = 1.2;
        ctx.strokeRect(boxLeft, tpTop, boxW, tpH);

        const slTop = Math.min(entryPt.y, slPt.y);
        const slH = Math.abs(slPt.y - entryPt.y);
        ctx.fillStyle = 'rgba(242, 54, 69, 0.22)';
        ctx.fillRect(boxLeft, slTop, boxW, slH);
        ctx.strokeStyle = '#F23645';
        ctx.strokeRect(boxLeft, slTop, boxW, slH);

        ctx.strokeStyle = '#94A3B8';
        ctx.beginPath();
        ctx.moveTo(boxLeft, entryPt.y);
        ctx.lineTo(boxLeft + boxW, entryPt.y);
        ctx.stroke();

        const reward = Math.abs(ptsData[1].price - ptsData[0].price);
        const risk = Math.max(Math.abs(ptsData[2].price - ptsData[0].price), 1e-9);
        const rr = reward / risk;
        const pctReward = (reward / Math.max(ptsData[0].price, 1e-9)) * 100;
        const pctRisk = (risk / Math.max(ptsData[0].price, 1e-9)) * 100;
        const acct = obj?.accountSize ?? 100000;
        const rPct = obj?.riskPercent ?? 1;
        const riskUsd = acct * (rPct / 100);
        const qtyCalc = riskUsd / risk;

        ctx.font = '600 10px "JetBrains Mono", monospace';
        const rrText = `${type === 'long_pos' ? 'LONG' : 'SHORT'} R:R 1:${rr.toFixed(
          2
        )} • Qty ${qtyCalc >= 10 ? qtyCalc.toFixed(1) : qtyCalc.toFixed(3)}`;
        ctx.fillStyle = theme === 'dark' ? 'rgba(19,23,34,0.92)' : 'rgba(255,255,255,0.94)';
        const bw = ctx.measureText(rrText).width + 12;
        ctx.fillRect(boxLeft + 6, entryPt.y - 10, bw, 18);
        ctx.fillStyle = rr >= 1.5 ? '#089981' : '#F59E0B';
        ctx.fillText(rrText, boxLeft + 12, entryPt.y + 3);

        ctx.fillStyle = '#089981';
        ctx.fillText(
          `TP +${pctReward.toFixed(2)}% (+$${(riskUsd * rr).toFixed(0)})`,
          boxLeft + 6,
          tpPt.y + (type === 'long_pos' ? 12 : -5)
        );
        ctx.fillStyle = '#F23645';
        ctx.fillText(
          `SL -${pctRisk.toFixed(2)}% (-$${riskUsd.toFixed(0)})`,
          boxLeft + 6,
          slPt.y + (type === 'long_pos' ? -5 : 12)
        );
      } else if ((type === 'brush' || type === 'highlighter') && pts.length >= 2) {
        if (type === 'highlighter') {
          ctx.lineWidth = Math.max(12, lineWidth * 5);
          ctx.globalAlpha = 0.34;
        }
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i].x, pts[i].y);
        ctx.stroke();
      } else if (type === 'arrow_marker' && pts.length >= 2) {
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        ctx.lineTo(pts[1].x, pts[1].y);
        ctx.stroke();
        ctx.setLineDash([]);
        drawArrowHead(ctx, pts[0], pts[1], 10 + lineWidth);
      } else if (type === 'text' && pts[0]) {
        ctx.setLineDash([]);
        const label = obj?.text || 'Note';
        ctx.font = fontSpec;
        const tw = ctx.measureText(label).width + 14;
        ctx.fillStyle = theme === 'dark' ? 'rgba(19, 23, 34, 0.9)' : 'rgba(255, 255, 255, 0.94)';
        ctx.strokeStyle = strokeCol;
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.roundRect(pts[0].x, pts[0].y - 16, tw, 24, 5);
        ctx.fill();
        ctx.stroke();
        ctx.fillStyle = txtCol;
        ctx.fillText(label, pts[0].x + 7, pts[0].y);
      } else if (type === 'callout' && pts.length >= 2) {
        ctx.setLineDash([]);
        const label = obj?.text || 'Callout';
        ctx.font = fontSpec;
        const tw = Math.max(70, ctx.measureText(label).width + 16);
        const bx = pts[1].x - tw / 2;
        const by = pts[1].y - 14;

        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        ctx.lineTo(pts[1].x, pts[1].y);
        ctx.stroke();

        ctx.fillStyle = theme === 'dark' ? '#1E222D' : '#FFFFFF';
        ctx.strokeStyle = strokeCol;
        ctx.lineWidth = 1.5;
        ctx.beginPath();
        ctx.roundRect(bx, by, tw, 26, 6);
        ctx.fill();
        ctx.stroke();
        ctx.fillStyle = txtCol;
        ctx.fillText(label, bx + 8, by + 16.5);
      } else if (type === 'price_label' && pts[0]) {
        ctx.setLineDash([]);
        const txt = formatPrice(ptsData[0].price, precision);
        ctx.font = '700 11px "JetBrains Mono", monospace';
        const tw = ctx.measureText(txt).width + 18;
        ctx.fillStyle = color;
        ctx.beginPath();
        ctx.moveTo(pts[0].x, pts[0].y);
        ctx.lineTo(pts[0].x + 8, pts[0].y - 10);
        ctx.lineTo(pts[0].x + tw, pts[0].y - 10);
        ctx.lineTo(pts[0].x + tw, pts[0].y + 10);
        ctx.lineTo(pts[0].x + 8, pts[0].y + 10);
        ctx.closePath();
        ctx.fill();
        ctx.fillStyle = '#FFFFFF';
        ctx.fillText(txt, pts[0].x + 11, pts[0].y + 4);
      } else if ((type === 'ruler' || type === 'date_price_range') && pts.length >= 2) {
        const x = Math.min(pts[0].x, pts[1].x);
        const y = Math.min(pts[0].y, pts[1].y);
        const w = Math.abs(pts[1].x - pts[0].x);
        const h = Math.abs(pts[1].y - pts[0].y);
        const isUp = ptsData[1].price >= ptsData[0].price;
        const rc = type === 'date_price_range' ? color : isUp ? '#089981' : '#F23645';

        ctx.fillStyle = hexWithOpacity(rc, obj?.fillOpacity ?? 15);
        ctx.fillRect(x, y, w, h);
        ctx.strokeStyle = rc;
        ctx.setLineDash([4, 3]);
        ctx.strokeRect(x, y, w, h);
        ctx.setLineDash([]);

        const diff = ptsData[1].price - ptsData[0].price;
        const pct = (diff / Math.max(ptsData[0].price, 1e-9)) * 100;
        const bars = Math.round(Math.abs(ptsData[1].time - ptsData[0].time) / tfSec);
        const minT = Math.min(ptsData[0].time, ptsData[1].time);
        const maxT = Math.max(ptsData[0].time, ptsData[1].time);
        const volSum = candles
          .filter((c) => c.time >= minT && c.time <= maxT)
          .reduce((a, b) => a + b.volume, 0);

        const infoText = `${diff >= 0 ? '+' : ''}${formatPrice(diff, precision)} (${
          pct >= 0 ? '+' : ''
        }${pct.toFixed(2)}%) • ${bars}b • Vol ${formatCompactNumber(volSum)}`;

        ctx.font = '600 10px "JetBrains Mono", monospace';
        const tw = ctx.measureText(infoText).width + 14;
        const bx = Math.min(Math.max(8, (pts[0].x + pts[1].x) / 2 - tw / 2), width - tw - 8);
        const by = isUp ? Math.max(8, y - 26) : Math.min(height - 30, y + h + 6);
        ctx.fillStyle = rc;
        ctx.beginPath();
        ctx.roundRect(bx, by, tw, 20, 5);
        ctx.fill();
        ctx.fillStyle = '#FFFFFF';
        ctx.fillText(infoText, bx + 7, by + 13.5);
      }

      if (isSelected && type !== 'brush' && type !== 'highlighter') {
        ctx.setLineDash([]);
        ctx.globalAlpha = 1;
        for (const hp of pts) {
          ctx.fillStyle = '#FFFFFF';
          ctx.strokeStyle = '#2962FF';
          ctx.lineWidth = 2;
          ctx.beginPath();
          ctx.arc(hp.x, hp.y, 4.5, 0, Math.PI * 2);
          ctx.fill();
          ctx.stroke();
        }
      }

      ctx.restore();
    };

    if (drawingsVisible) {
      for (const d of symbolDrawings) {
        if (d.visible === false) continue;
        drawSingle(
          d.type,
          d.points,
          d.color,
          d.lineWidth,
          d.lineStyle ?? 'solid',
          d.id === selectedDrawingId,
          d
        );
      }
    }

    if (
      draftPoints.length > 0 &&
      cursorPoint &&
      activeTool !== 'cursor' &&
      activeTool !== 'eraser' &&
      activeTool !== 'brush' &&
      activeTool !== 'highlighter' &&
      activeTool !== 'text'
    ) {
      drawSingle(
        activeTool,
        [...draftPoints, cursorPoint],
        drawingColor,
        drawingLineWidth,
        drawingLineStyle,
        false
      );
    }

    if (isBrushing && brushPointsRef.current.length > 1) {
      drawSingle(
        activeTool === 'highlighter' ? 'highlighter' : 'brush',
        brushPointsRef.current,
        drawingColor,
        drawingLineWidth,
        drawingLineStyle,
        false
      );
    }

    if (rulerStart && rulerEnd) {
      drawSingle('ruler', [rulerStart, rulerEnd], '#2962FF', 1.5, 'solid', false);
    }

    if (magnetMode && cursorPoint && activeTool !== 'cursor' && activeTool !== 'eraser') {
      const sp = toScreen(cursorPoint);
      if (sp) {
        ctx.fillStyle = '#00D4AA';
        ctx.beginPath();
        ctx.arc(sp.x, sp.y, 4, 0, Math.PI * 2);
        ctx.fill();
      }
    }

    ctx.restore();
  }, [
    width,
    height,
    drawingsVisible,
    symbolDrawings,
    selectedDrawingId,
    draftPoints,
    cursorPoint,
    activeTool,
    drawingColor,
    drawingLineWidth,
    drawingLineStyle,
    isBrushing,
    rulerStart,
    rulerEnd,
    magnetMode,
    toScreen,
    priceToY,
    timeToX,
    precision,
    tfSec,
    theme,
    candles,
    vpvrData,
    structureData,
    smartOverlays,
    chartSettings.priceScalePosition,
  ]);

  useEffect(() => {
    render();
  }, [render]);

  useEffect(() => {
    if (!chart) return;
    const onRangeChange = () => render();
    chart.timeScale().subscribeVisibleLogicalRangeChange(onRangeChange);
    return () => {
      chart.timeScale().unsubscribeVisibleLogicalRangeChange(onRangeChange);
    };
  }, [chart, render]);

  const commitDrawing = useCallback(
    (type: Exclude<DrawingToolType, 'cursor' | 'eraser' | 'ruler'>, points: DrawingPoint[]) => {
      addDrawing({
        id: generateId('drw'),
        symbol,
        type,
        points,
        color: drawingColor,
        lineWidth: drawingLineWidth,
        lineStyle: drawingLineStyle,
        createdAt: Date.now(),
      });
      setDraftPoints([]);
      finishToolUsage();
    },
    [addDrawing, symbol, drawingColor, drawingLineWidth, drawingLineStyle, finishToolUsage]
  );

  const handlePointerDown = (e: React.PointerEvent<HTMLCanvasElement>) => {
    if (e.button === 2) return;

    const rect = e.currentTarget.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;
    const dataPt = screenToData(x, y, true);
    if (!dataPt) return;

    if (activeTool === 'eraser') {
      const hit = hitTestDrawings(x, y, true);
      if (hit) removeDrawing(symbol, hit.id);
      return;
    }

    if (activeTool === 'cursor') {
      const hit = hitTestDrawings(x, y);
      if (hit) {
        const target = symbolDrawings.find((d) => d.id === hit.id);
        if (target) {
          setSelectedDrawingId(target.id);
          dragStateRef.current = {
            drawingId: target.id,
            mode: hit.mode,
            pointIndex: hit.pointIndex,
            startPointer: dataPt,
            originalPoints: target.points.map((pt) => ({ ...pt })),
          };
          e.currentTarget.setPointerCapture(e.pointerId);
        }
      } else {
        setSelectedDrawingId(null);
      }
      return;
    }

    if (activeTool === 'ruler') {
      if (!rulerStart || (rulerStart && rulerEnd && draftPoints.length === 0)) {
        setRulerStart(dataPt);
        setRulerEnd(dataPt);
        setDraftPoints([dataPt]);
      } else {
        setRulerEnd(dataPt);
        setDraftPoints([]);
      }
      return;
    }

    if (activeTool === 'brush' || activeTool === 'highlighter') {
      setIsBrushing(true);
      brushPointsRef.current = [dataPt];
      e.currentTarget.setPointerCapture(e.pointerId);
      return;
    }

    if (activeTool === 'text') {
      setTextDraft({ x, y, points: [dataPt], kind: 'text' });
      setTextValue('');
      return;
    }

    if (activeTool === 'callout') {
      if (draftPoints.length === 0) {
        setDraftPoints([dataPt]);
      } else {
        setTextDraft({ x, y, points: [draftPoints[0], dataPt], kind: 'callout' });
        setTextValue('');
        setDraftPoints([]);
      }
      return;
    }

    if (activeTool === 'polyline') {
      setDraftPoints((prev) => [...prev, dataPt]);
      return;
    }

    if (activeTool === 'long_pos' || activeTool === 'short_pos') {
      const topPrice = series?.coordinateToPrice(Math.max(10, y - height * 0.14));
      const botPrice = series?.coordinateToPrice(Math.min(height - 10, y + height * 0.08));
      const tpDelta = topPrice ? Math.abs(Number(topPrice) - dataPt.price) : dataPt.price * 0.02;
      const slDelta = botPrice ? Math.abs(dataPt.price - Number(botPrice)) : dataPt.price * 0.01;
      const isLong = activeTool === 'long_pos';

      const tpPrice = isLong ? dataPt.price + tpDelta : Math.max(0.00001, dataPt.price - tpDelta);
      const slPrice = isLong ? Math.max(0.00001, dataPt.price - slDelta) : dataPt.price + slDelta;

      addDrawing({
        id: generateId('drw'),
        symbol,
        type: activeTool,
        points: [
          dataPt,
          { time: dataPt.time + tfSec * 14, price: tpPrice },
          { time: dataPt.time + tfSec * 14, price: slPrice },
        ],
        color: isLong ? '#089981' : '#F23645',
        lineWidth: 1.5,
        lineStyle: 'solid',
        riskPercent: chartSettings.defaultRiskPercent,
        accountSize: 100000,
        createdAt: Date.now(),
      });
      finishToolUsage();
      return;
    }

    const meta = DRAWING_TOOLS_CATALOG.find((t) => t.id === activeTool);
    const reqPts = meta?.pointsRequired ?? 2;
    const nextPts = [...draftPoints, dataPt];

    if (nextPts.length >= reqPts) {
      commitDrawing(activeTool, nextPts);
    } else {
      setDraftPoints(nextPts);
    }
  };

  const handlePointerMove = (e: React.PointerEvent<HTMLCanvasElement>) => {
    const rect = e.currentTarget.getBoundingClientRect();
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;

    if (dragStateRef.current) {
      const pt = screenToData(x, y, true);
      if (!pt) return;
      const drag = dragStateRef.current;
      const updatedPts = drag.originalPoints.map((op) => ({ ...op }));

      if (drag.mode === 'point') {
        updatedPts[drag.pointIndex] = pt;
      } else {
        const dt = pt.time - drag.startPointer.time;
        const dp = pt.price - drag.startPointer.price;
        for (let i = 0; i < updatedPts.length; i++) {
          updatedPts[i] = {
            time: updatedPts[i].time + dt,

function formatTickInOffset(
  unixSec: number,
  offsetMinutes: number | 'local',
  tickType: TickMarkType
): string {
  const pad = (n: number) => String(n).padStart(2, '0');
  const d =
    offsetMinutes === 'local'
      ? new Date(unixSec * 1000)
      : new Date((unixSec + offsetMinutes * 60) * 1000);

  const yr = offsetMinutes === 'local' ? d.getFullYear() : d.getUTCFullYear();
  const mo = offsetMinutes === 'local' ? d.getMonth() : d.getUTCMonth();
  const day = offsetMinutes === 'local' ? d.getDate() : d.getUTCDate();
  const hr = offsetMinutes === 'local' ? d.getHours() : d.getUTCHours();
  const min = offsetMinutes === 'local' ? d.getMinutes() : d.getUTCMinutes();
  const sec = offsetMinutes === 'local' ? d.getSeconds() : d.getUTCSeconds();

  if (tickType === TickMarkType.Year) return String(yr);
  if (tickType === TickMarkType.Month) return `${MONTHS[mo]} '${String(yr).slice(-2)}`;
  if (tickType === TickMarkType.DayOfMonth) return `${day} ${MONTHS[mo]}`;
  if (tickType === TickMarkType.TimeWithSeconds) return `${pad(hr)}:${pad(min)}:${pad(sec)}`;
  return `${pad(hr)}:${pad(min)}`;
}

function formatCountdown(secondsLeft: number): string {
  const s = Math.max(0, Math.floor(secondsLeft));
  const hrs = Math.floor(s / 3600);
  const mins = Math.floor((s % 3600) / 60);
  const secs = s % 60;
  const pad = (n: number) => String(n).padStart(2, '0');
  if (hrs > 0) return `${pad(hrs)}:${pad(mins)}:${pad(secs)}`;
  return `${pad(mins)}:${pad(secs)}`;
}

  const setEditingIndicatorId = useTradingStore((s) => s.setEditingIndicatorId);

  const setActivePaneId = useTradingStore((s) => s.setActivePaneId);
  const updatePaneTimeframe = useTradingStore((s) => s.updatePaneTimeframe);
  const updatePaneScaleMode = useTradingStore((s) => s.updatePaneScaleMode);
  const toggleIndicatorVisibility = useTradingStore((s) => s.toggleIndicatorVisibility);
  const removeIndicator = useTradingStore((s) => s.removeIndicator);
  const updateQuote = useTradingStore((s) => s.updateQuote);
  const processPaper = useTradingStore((s) => s.processMarketPriceForPaperTrading);

  const placeMarketOrder = useTradingStore((s) => s.placeMarketOrder);
  const placeLimitOrder = useTradingStore((s) => s.placeLimitOrder);
  const addAlert = useTradingStore((s) => s.addAlert);
  const addDrawing = useTradingStore((s) => s.addDrawing);
  const drawingColor = useTradingStore((s) => s.drawingColor);

  const syncCrosshair = useTradingStore((s) => s.syncCrosshair);
  const syncedCrosshairTime = useTradingStore((s) => s.syncedCrosshairTime);
  const setSyncedCrosshairTime = useTradingStore((s) => s.setSyncedCrosshairTime);

  const replay = useTradingStore((s) => s.replay);
  const startReplayAt = useTradingStore((s) => s.startReplayAt);
  const setReplayTotalBars = useTradingStore((s) => s.setReplayTotalBars);
  const setReplayIndex = useTradingStore((s) => s.setReplayIndex);

  const alerts = useTradingStore((s) => s.alerts);
  const positions = useTradingStore((s) => s.positions);
  const snapshotCounter = useTradingStore((s) => s.snapshotRequestCounter);
  const addToast = useTradingStore((s) => s.addToast);

  const symbolInfo = useMemo(() => getSymbolInfo(pane.symbol), [pane.symbol]);
  const effectivePrecision =
    chartSettings.precisionOverride === 'auto'
      ? symbolInfo.precision
      : chartSettings.precisionOverride;
  const effectiveMinMove = Math.pow(10, -effectivePrecision);

  const tfMeta = TIMEFRAME_MAP[pane.timeframe];
  const tzPreset = useMemo(() => getTimezonePreset(chartSettings.timezone), [chartSettings.timezone]);

  const mainContainerRef = useRef<HTMLDivElement | null>(null);
  const chartRef = useRef<IChartApi | null>(null);
  const mainSeriesRef = useRef<MainSeriesApi | null>(null);
  const overlaySeriesRef = useRef<Map<string, ISeriesApi<'Line' | 'Histogram'>[]>>(new Map());
  const priceLinesRef = useRef<IPriceLine[]>([]);
  const drawingCanvasRef = useRef<HTMLCanvasElement | null>(null);
  const hitTestDrawingRef = useRef<((x: number, y: number) => string | null) | null>(null);
  const isExternalCrosshairRef = useRef(false);

  const [chartSize, setChartSize] = useState({ width: 0, height: 0 });
  const [fullCandles, setFullCandles] = useState<Candle[]>([]);
  const [dataSource, setDataSource] = useState<FeedSourceType>('simulated');
  const [loading, setLoading] = useState(true);
  const [hoveredBar, setHoveredBar] = useState<Candle | null>(null);

  const [legendExpanded, setLegendExpanded] = useState(true);
  const [contextMenu, setContextMenu] = useState<ChartContextMenuData | null>(null);
  const [hoverChart, setHoverChart] = useState(false);
  const [priceAxisHover, setPriceAxisHover] = useState<{ y: number; price: number } | null>(null);
  const [plusMenuOpen, setPlusMenuOpen] = useState<{ y: number; price: number } | null>(null);
  const [scrolledAway, setScrolledAway] = useState(false);
  const [quickQty, setQuickQty] = useState(() =>
    symbolInfo.category === 'Crypto' ? '0.10' : symbolInfo.category === 'IDX' ? '1000' : '10'
  );
  const [countdownText, setCountdownText] = useState('00:00');
  const [clockText, setClockText] = useState('');
  const [lastPriceY, setLastPriceY] = useState<number | null>(null);

  const isThisPaneInReplay = replay.enabled && replay.paneId === pane.id;

  useEffect(() => {
    setQuickQty(
      symbolInfo.category === 'Crypto' ? '0.10' : symbolInfo.category === 'IDX' ? '1000' : '10'
    );
  }, [symbolInfo.category, pane.symbol]);

  const displayCandles = useMemo(() => {
    if (isThisPaneInReplay && !replay.isSelectingStart && fullCandles.length > 0) {
      const end = Math.min(fullCandles.length, Math.max(5, replay.currentIndex + 1));
      return fullCandles.slice(0, end);
    }
    return fullCandles;
  }, [fullCandles, isThisPaneInReplay, replay.isSelectingStart, replay.currentIndex]);

  useEffect(() => {
    const tickClock = () => {
      const nowSec = Math.floor(Date.now() / 1000);
      const nextClose = (Math.floor(nowSec / tfMeta.seconds) + 1) * tfMeta.seconds;
      setCountdownText(formatCountdown(nextClose - nowSec));
      const timeStr = formatTimeInOffset(nowSec, tzPreset.offsetMinutes, false, true);
      setClockText(`${timeStr} (${tzPreset.short})`);
    };
    tickClock();
    const id = window.setInterval(tickClock, 1000);
    return () => window.clearInterval(id);
  }, [tfMeta.seconds, tzPreset]);

  const refreshLastPriceY = useCallback(() => {
    if (!mainSeriesRef.current || displayCandles.length === 0) {
      setLastPriceY(null);
      return;
    }
    const last = displayCandles[displayCandles.length - 1];
    const coord = mainSeriesRef.current.priceToCoordinate(last.close);
    setLastPriceY(coord !== null ? Number(coord) : null);
  }, [displayCandles]);

  // 1. Initialize main chart
  useEffect(() => {
    const el = mainContainerRef.current;
    if (!el) return;

    const isDark = theme === 'dark';
    const bgDark = chartSettings.bgColorDark || '#131722';
    const bgDarkBot = chartSettings.bgColorDarkBottom || '#0B0E14';
    const bgLight = chartSettings.bgColorLight || '#FFFFFF';
    const bgLightBot = chartSettings.bgColorLightBottom || '#F0F3FA';
    const isLeft = chartSettings.priceScalePosition === 'left';

    const chart = createChart(el, {
      width: el.clientWidth,
      height: el.clientHeight,
      layout: {
        background:
          chartSettings.bgType === 'gradient'
            ? {
                type: ColorType.VerticalGradient,
                topColor: isDark ? bgDark : bgLight,
                bottomColor: isDark ? bgDarkBot : bgLightBot,
              }
            : { type: ColorType.Solid, color: isDark ? bgDark : bgLight },
        textColor: isDark ? '#787B86' : '#475569',
        fontFamily: '"JetBrains Mono", monospace',
        fontSize: chartSettings.scaleFontSize,
      },
      crosshair: { mode: CrosshairMode.Normal },
      rightPriceScale: {
        visible: !isLeft,
        borderColor: isDark ? '#2A2E39' : '#E0E3EB',
        scaleMargins: {
          top: chartSettings.scaleMarginTop / 100,
          bottom: chartSettings.scaleMarginBottom / 100,
        },
        invertScale: chartSettings.invertScale,
      },
      leftPriceScale: {
        visible: isLeft,
        borderColor: isDark ? '#2A2E39' : '#E0E3EB',
        scaleMargins: {
          top: chartSettings.scaleMarginTop / 100,
          bottom: chartSettings.scaleMarginBottom / 100,
        },
        invertScale: chartSettings.invertScale,
      },
      timeScale: {
        borderColor: isDark ? '#2A2E39' : '#E0E3EB',
        timeVisible: true,
        secondsVisible: tfMeta.seconds < 60,
        rightOffset: chartSettings.rightOffset,
        barSpacing: chartSettings.barSpacing,
      },
      handleScroll: { vertTouchDrag: true },
      handleScale: { axisPressedMouseMove: true },
    });

    chartRef.current = chart;
    setChartSize({ width: el.clientWidth, height: el.clientHeight });

    const ro = new ResizeObserver((entries) => {
      for (const entry of entries) {
        const { width, height } = entry.contentRect;
        if (width > 0 && height > 0) {
          chart.applyOptions({ width, height });
          setChartSize({ width, height });
        }
      }
    });
    ro.observe(el);

    return () => {
      ro.disconnect();
      chart.remove();
      chartRef.current = null;
      mainSeriesRef.current = null;
      overlaySeriesRef.current.clear();
      priceLinesRef.current = [];
    };
  }, [theme]);

  // Live apply ALL ChartSettings without destroying chart
  useEffect(() => {
    const chart = chartRef.current;
    if (!chart) return;
    const isDark = theme === 'dark';
    const bgDark = chartSettings.bgColorDark || '#131722';
    const bgDarkBot = chartSettings.bgColorDarkBottom || '#0B0E14';
    const bgLight = chartSettings.bgColorLight || '#FFFFFF';
    const bgLightBot = chartSettings.bgColorLightBottom || '#F0F3FA';
    const gridAlpha = Math.max(0.05, Math.min(1, (chartSettings.gridOpacity ?? 55) / 100));
    const isLeft = chartSettings.priceScalePosition === 'left';

    chart.applyOptions({
      layout: {
        background:
          chartSettings.bgType === 'gradient'
            ? {
                type: ColorType.VerticalGradient,
                topColor: isDark ? bgDark : bgLight,
                bottomColor: isDark ? bgDarkBot : bgLightBot,
              }
            : { type: ColorType.Solid, color: isDark ? bgDark : bgLight },
        fontSize: chartSettings.scaleFontSize,
      },
      localization: {
        timeFormatter: (time: Time) => {
          if (typeof time === 'number') {
            return formatTimeInOffset(
              time,
              tzPreset.offsetMinutes,
              true,
              tfMeta.seconds < 60
            );
          }
          return String(time);
        },
      },
      grid: {
        vertLines: {
          visible:
            chartSettings.showGrid &&
            (chartSettings.gridStyle === 'both' || chartSettings.gridStyle === 'vert'),
          style: toChartLineStyle(chartSettings.gridLineStyle),
          color: isDark
            ? `rgba(42, 46, 57, ${gridAlpha.toFixed(2)})`
            : `rgba(15, 23, 42, ${(gridAlpha * 0.14).toFixed(2)})`,
        },
        horzLines: {
          visible:
            chartSettings.showGrid &&
            (chartSettings.gridStyle === 'both' || chartSettings.gridStyle === 'horz'),
          style: toChartLineStyle(chartSettings.gridLineStyle),
          color: isDark
            ? `rgba(42, 46, 57, ${gridAlpha.toFixed(2)})`
            : `rgba(15, 23, 42, ${(gridAlpha * 0.14).toFixed(2)})`,
        },
      },
      crosshair: {
        vertLine: {
          style: toChartLineStyle(chartSettings.crosshairStyle),
          color: chartSettings.crosshairColor,
          width: chartSettings.crosshairWidth,
          labelBackgroundColor: '#2962FF',
        },
        horzLine: {
          style: toChartLineStyle(chartSettings.crosshairStyle),
          color: chartSettings.crosshairColor,
          width: chartSettings.crosshairWidth,
          labelBackgroundColor: '#2962FF',
        },
      },
      timeScale: {
        timeVisible: true,
        secondsVisible: tfMeta.seconds < 60,
        rightOffset: chartSettings.rightOffset,
        tickMarkFormatter: (time: Time, tickType: TickMarkType) => {
          if (typeof time === 'number') {
            return formatTickInOffset(time, tzPreset.offsetMinutes, tickType);
          }
          return String(time);
        },
      },
      rightPriceScale: {
        visible: !isLeft,
        invertScale: chartSettings.invertScale,
        scaleMargins: {
          top: chartSettings.scaleMarginTop / 100,
          bottom: chartSettings.scaleMarginBottom / 100,
        },
        mode:
          pane.scaleMode === 'logarithmic'
            ? PriceScaleMode.Logarithmic
            : pane.scaleMode === 'percentage'
            ? PriceScaleMode.Percentage
            : PriceScaleMode.Normal,
      },
      leftPriceScale: {
        visible: isLeft,
        invertScale: chartSettings.invertScale,
        scaleMargins: {
          top: chartSettings.scaleMarginTop / 100,
          bottom: chartSettings.scaleMarginBottom / 100,
        },
        mode:
          pane.scaleMode === 'logarithmic'
            ? PriceScaleMode.Logarithmic
            : pane.scaleMode === 'percentage'
            ? PriceScaleMode.Percentage
            : PriceScaleMode.Normal,
      },
    });
  }, [theme, chartSettings, tzPreset, tfMeta.seconds, pane.scaleMode]);

  // 2. Fetch Historical Candles (re-runs on symbol, timeframe, or DataFeed config change!)
  useEffect(() => {
    let cancelled = false;
    setLoading(true);
    setHoveredBar(null);

    fetchHistoricalCandles(pane.symbol, pane.timeframe, dataFeed).then(({ candles, source }) => {
      if (cancelled) return;
      setFullCandles(candles);
      setDataSource(source);
      setLoading(false);

      if (candles.length > 0) {
        const last = candles[candles.length - 1];
        updateQuote(pane.symbol, last.close, source, last.volume);
      }

      if (
        useTradingStore.getState().replay.enabled &&
        useTradingStore.getState().replay.paneId === pane.id
      ) {
        setReplayTotalBars(candles.length);
      }
    });

    return () => {
      cancelled = true;
    };
  }, [
    pane.symbol,
    pane.timeframe,
    pane.id,
    dataFeed.twelveDataApiKey,
    dataFeed.preferTwelveData,
    dataFeed.useGoldProxyWhenNoKey,
    updateQuote,
    setReplayTotalBars,
  ]);

  useEffect(() => {
    if (isThisPaneInReplay && fullCandles.length > 0) {
      setReplayTotalBars(fullCandles.length);
    }
  }, [isThisPaneInReplay, fullCandles.length, setReplayTotalBars]);

  // 3. Subscribe to Live Ticks
  useEffect(() => {
    if (loading || fullCandles.length === 0 || isThisPaneInReplay) return;
    const lastCandle = fullCandles[fullCandles.length - 1];

    const unsubscribe = subscribeLiveTicks(
      pane.symbol,
      pane.timeframe,
      lastCandle,
      (tickCandle, source) => {
        setDataSource(source);
        setFullCandles((prev) => {
          if (prev.length === 0) return [tickCandle];
          const tail = prev[prev.length - 1];
          if (tickCandle.time === tail.time) {
            const copy = prev.slice();
            copy[copy.length - 1] = tickCandle;
            return copy;
          }
          if (tickCandle.time > tail.time) {
            return [...prev.slice(-799), tickCandle];
          }
          return prev;
        });

        updateQuote(pane.symbol, tickCandle.close, source, tickCandle.volume);
        processPaper(pane.symbol, tickCandle.high, tickCandle.low, tickCandle.close);
      },
      dataFeed
    );

    return unsubscribe;
  }, [
    pane.symbol,
    pane.timeframe,
    loading,
    isThisPaneInReplay,
    fullCandles.length > 0,
    dataFeed.twelveDataApiKey,
    dataFeed.preferTwelveData,
    dataFeed.useGoldProxyWhenNoKey,
  ]);

  useEffect(() => {
    if (!isThisPaneInReplay || replay.isSelectingStart || displayCandles.length === 0) return;
    const cur = displayCandles[displayCandles.length - 1];
    updateQuote(pane.symbol, cur.close, 'simulated', cur.volume);
    processPaper(pane.symbol, cur.high, cur.low, cur.close);
    if (replay.currentTimestamp !== cur.time) {
      setReplayIndex(replay.currentIndex, cur.time);
    }
  }, [isThisPaneInReplay, replay.isSelectingStart, displayCandles, pane.symbol]);

  // 4. Create Main Price Series
  useEffect(() => {
    const chart = chartRef.current;
    if (!chart) return;

    if (mainSeriesRef.current) {
      try {
        chart.removeSeries(mainSeriesRef.current);
      } catch {
        // ignore
      }
      mainSeriesRef.current = null;
      priceLinesRef.current = [];
    }

    const priceFormat = {
      type: 'price' as const,
      precision: effectivePrecision,
      minMove: effectiveMinMove,
    };

    const commonLineOpts = {
      priceLineVisible: chartSettings.showLastPriceLine,
      priceLineStyle: toChartLineStyle(chartSettings.lastPriceLineStyle),
      priceScaleId: chartSettings.priceScalePosition === 'left' ? 'left' : 'right',
      title: chartSettings.showSymbolAxisLabel ? symbolInfo.displaySymbol : '',
    };

    let s: MainSeriesApi;
    if (pane.chartStyle === 'hollow') {
      s = chart.addCandlestickSeries({
        upColor: 'transparent',
        downColor: chartSettings.downColor,
        borderUpColor: chartSettings.upColor,
        borderDownColor: chartSettings.downColor,
        wickVisible: chartSettings.showWicks,
        wickUpColor: chartSettings.wickUpColor,
        wickDownColor: chartSettings.wickDownColor,
        priceFormat,
        ...commonLineOpts,
      });
    } else if (pane.chartStyle === 'bars') {
      s = chart.addBarSeries({
        upColor: chartSettings.upColor,
        downColor: chartSettings.downColor,
        priceFormat,
        ...commonLineOpts,
      });
    } else if (pane.chartStyle === 'line') {
      s = chart.addLineSeries({
        color: '#2962FF',
        lineWidth: 2,
        priceFormat,
        ...commonLineOpts,
      });
    } else if (pane.chartStyle === 'area') {
      s = chart.addAreaSeries({
        lineColor: '#2962FF',
        topColor: 'rgba(41, 98, 255, 0.34)',
        bottomColor: 'rgba(41, 98, 255, 0.02)',
        lineWidth: 2,
        priceFormat,
        ...commonLineOpts,
      });
    } else {
      s = chart.addCandlestickSeries({
        upColor: chartSettings.upColor,
        downColor: chartSettings.downColor,
        borderVisible: chartSettings.showBorders,
        borderUpColor: chartSettings.borderUpColor,
        borderDownColor: chartSettings.borderDownColor,
        wickVisible: chartSettings.showWicks,
        wickUpColor: chartSettings.wickUpColor,
        wickDownColor: chartSettings.wickDownColor,
        priceFormat,
        ...commonLineOpts,
      });
    }

    mainSeriesRef.current = s;
  }, [
    pane.chartStyle,
    effectivePrecision,
    effectiveMinMove,
    theme,
    symbolInfo.displaySymbol,
    chartSettings.upColor,
    chartSettings.downColor,
    chartSettings.wickUpColor,
    chartSettings.wickDownColor,
    chartSettings.borderUpColor,
    chartSettings.borderDownColor,
    chartSettings.showWicks,
    chartSettings.showBorders,
    chartSettings.showLastPriceLine,
    chartSettings.lastPriceLineStyle,
    chartSettings.priceScalePosition,
    chartSettings.showSymbolAxisLabel,
  ]);

  // 5. Feed Data & Overlay Indicators (with Source, Offset, Style, Visibility!)
  useEffect(() => {
    const chart = chartRef.current;
    const series = mainSeriesRef.current;
    if (!chart || !series || displayCandles.length === 0) return;

    const effectiveCandles =
      pane.chartStyle === 'heikin_ashi' ? computeHeikinAshi(displayCandles) : displayCandles;

    if (pane.chartStyle === 'line' || pane.chartStyle === 'area') {
      (series as ISeriesApi<'Line' | 'Area'>).setData(
        effectiveCandles.map((c) => ({ time: c.time as UTCTimestamp, value: c.close }))
      );
    } else {
      (series as ISeriesApi<'Candlestick' | 'Bar'>).setData(
        effectiveCandles.map((c) => ({
          time: c.time as UTCTimestamp,
          open: c.open,
          high: c.high,
          low: c.low,
          close: c.close,
        }))
      );
    }

    refreshLastPriceY();

    for (const list of overlaySeriesRef.current.values()) {
      for (const s of list) {
        try {
          chart.removeSeries(s);
        } catch {
          // ignore
        }
      }
    }
    overlaySeriesRef.current.clear();

    const targetScaleId = chartSettings.priceScalePosition === 'left' ? 'left' : 'right';
    const overlays = pane.indicators.filter(
      (i) =>
        i.placement === 'overlay' &&
        i.visible &&
        isVisibleOnTimeframeGroup(i.visibility, tfMeta.group)
    );

    for (const ind of overlays) {
      const created: ISeriesApi<'Line' | 'Histogram'>[] = [];
      const src = ind.source ?? 'close';
      const offset = ind.offset ?? 0;
      const lw = (ind.lineWidth ?? 2) as 1 | 2 | 3 | 4;
      const ls = toChartLineStyle(ind.lineStyle);
      const showLast = Boolean(ind.showLastValue);

      if (ind.type === 'VOLUME') {
        const volSeries = chart.addHistogramSeries({
          priceFormat: { type: 'volume' },
          priceScaleId: '',
          lastValueVisible: false,
          priceLineVisible: false,
        });
        volSeries.priceScale().applyOptions({
          scaleMargins: { top: 0.84, bottom: 0 },
        });
        volSeries.setData(
          displayCandles.map((c) => ({
            time: c.time as UTCTimestamp,
            value: c.volume,
            color:
              c.close >= c.open ? 'rgba(8, 153, 129, 0.34)' : 'rgba(242, 54, 69, 0.34)',
          }))
        );
        created.push(volSeries);
      } else if (ind.type === 'SMA') {
        const pts = applyIndicatorOffset(
          computeSMA(displayCandles, ind.params.period ?? 50, src),
          offset,
          tfMeta.seconds
        );
        const l = chart.addLineSeries({
          color: ind.color,
          lineWidth: lw,
          lineStyle: ls,
          priceScaleId: targetScaleId,
          priceLineVisible: false,
          lastValueVisible: showLast,
        });
        l.setData(pts.map((p) => ({ time: p.time as UTCTimestamp, value: p.value })));
        created.push(l);
      } else if (ind.type === 'EMA') {
        const pts = applyIndicatorOffset(
          computeEMA(displayCandles, ind.params.period ?? 21, src),
          offset,
          tfMeta.seconds
        );
        const l = chart.addLineSeries({
          color: ind.color,
          lineWidth: lw,
          lineStyle: ls,
          priceScaleId: targetScaleId,
          priceLineVisible: false,
          lastValueVisible: showLast,
        });
        l.setData(pts.map((p) => ({ time: p.time as UTCTimestamp, value: p.value })));
        created.push(l);
      } else if (ind.type === 'VWAP') {
        const pts = applyIndicatorOffset(computeVWAP(displayCandles), offset, tfMeta.seconds);
        const l = chart.addLineSeries({
          color: ind.color,
          lineWidth: lw,
          lineStyle: ls,
          priceScaleId: targetScaleId,
          priceLineVisible: false,
          lastValueVisible: showLast,
        });
        l.setData(pts.map((p) => ({ time: p.time as UTCTimestamp, value: p.value })));
        created.push(l);
      } else if (ind.type === 'BB') {
        const bb = computeBollingerBands(
          displayCandles,
          ind.params.period ?? 20,
          ind.params.stdDev ?? 2,
          src
        );
        const up = chart.addLineSeries({
          color: ind.color,
          lineWidth: 1,
          lineStyle: ls,
          priceScaleId: targetScaleId,
          priceLineVisible: false,
          lastValueVisible: showLast,
        });
        const mid = chart.addLineSeries({
          color: ind.secondaryColor ?? '#64748B',
          lineWidth: 1,
          lineStyle: LineStyle.Dashed,
          priceScaleId: targetScaleId,
          priceLineVisible: false,
          lastValueVisible: false,
        });
        const low = chart.addLineSeries({
          color: ind.color,
          lineWidth: 1,
          lineStyle: ls,
          priceScaleId: targetScaleId,
          priceLineVisible: false,
          lastValueVisible: showLast,
        });
        up.setData(
          applyIndicatorOffset(bb.upper, offset, tfMeta.seconds).map((p) => ({
            time: p.time as UTCTimestamp,
            value: p.value,
          }))
        );
        mid.setData(
          applyIndicatorOffset(bb.middle, offset, tfMeta.seconds).map((p) => ({
            time: p.time as UTCTimestamp,
            value: p.value,
          }))
        );
        low.setData(
          applyIndicatorOffset(bb.lower, offset, tfMeta.seconds).map((p) => ({
            time: p.time as UTCTimestamp,
            value: p.value,
          }))
        );
        created.push(up, mid, low);
      } else if (ind.type === 'DONCHIAN') {
        const dc = computeDonchian(displayCandles, ind.params.period ?? 20);
        const up = chart.addLineSeries({
          color: ind.color,
          lineWidth: lw,
          lineStyle: ls,
          priceScaleId: targetScaleId,
          priceLineVisible: false,
          lastValueVisible: showLast,
        });
        const low = chart.addLineSeries({
          color: ind.color,
          lineWidth: lw,
          lineStyle: ls,
          priceScaleId: targetScaleId,
          priceLineVisible: false,
          lastValueVisible: showLast,
        });
        up.setData(
          applyIndicatorOffset(dc.upper, offset, tfMeta.seconds).map((p) => ({
            time: p.time as UTCTimestamp,
            value: p.value,
          }))
        );
        low.setData(
          applyIndicatorOffset(dc.lower, offset, tfMeta.seconds).map((p) => ({
            time: p.time as UTCTimestamp,
            value: p.value,
          }))
        );
        created.push(up, low);
      } else if (ind.type === 'SUPERTREND') {
        const st = applyIndicatorOffset(
          computeSupertrend(
            displayCandles,
            ind.params.period ?? 10,
            ind.params.multiplier ?? 3,
            ind.color,
            ind.secondaryColor ?? '#F23645'
          ),
          offset,
          tfMeta.seconds
        );
        const l = chart.addLineSeries({
          lineWidth: lw,
          priceScaleId: targetScaleId,
          priceLineVisible: false,
          lastValueVisible: showLast,
        });
        l.setData(st.map((p) => ({ time: p.time as UTCTimestamp, value: p.value, color: p.color })));
        created.push(l);
      }

      overlaySeriesRef.current.set(ind.id, created);
    }
  }, [
    displayCandles,
    pane.chartStyle,
    pane.indicators,
    theme,
    tfMeta.group,
    tfMeta.seconds,
    chartSettings.priceScalePosition,
    refreshLastPriceY,
  ]);

  // 6. Plot Alerts, Positions & High/Low Lines
  useEffect(() => {
    const series = mainSeriesRef.current;
    if (!series) return;

    for (const pl of priceLinesRef.current) {
      try {
        series.removePriceLine(pl);
      } catch {
        // ignore
      }
    }
    priceLinesRef.current = [];

    if (chartSettings.showHighLowLines && displayCandles.length > 0) {
      let hi = -Infinity;
      let lo = Infinity;
      for (const c of displayCandles) {
        if (c.high > hi) hi = c.high;
        if (c.low < lo) lo = c.low;
      }
      priceLinesRef.current.push(
        series.createPriceLine({
          price: hi,
          color: 'rgba(8, 153, 129, 0.55)',
          lineWidth: 1,
          lineStyle: LineStyle.Dotted,
          axisLabelVisible: true,
          title: 'High',
        }),
        series.createPriceLine({
          price: lo,
          color: 'rgba(242, 54, 69, 0.55)',
          lineWidth: 1,
          lineStyle: LineStyle.Dotted,
          axisLabelVisible: true,
          title: 'Low',
        })
      );
    }

    if (chartSettings.showAlertsOnChart) {
      const symAlerts = alerts.filter((a) => a.active && a.symbol === pane.symbol);
      for (const a of symAlerts) {
        priceLinesRef.current.push(
          series.createPriceLine({
            price: a.price,
            color: '#F59E0B',
            lineWidth: 1,
            lineStyle: LineStyle.Dashed,
            axisLabelVisible: true,
            title: 'Alert',
          })
        );
      }
    }

    if (chartSettings.showPositionsOnChart) {
      const symPositions = positions.filter((p) => p.symbol === pane.symbol);
      for (const pos of symPositions) {
        priceLinesRef.current.push(
          series.createPriceLine({
            price: pos.entryPrice,
            color: pos.side === 'buy' ? '#00D4AA' : '#F23645',
            lineWidth: 2,
            lineStyle: LineStyle.Solid,
            axisLabelVisible: true,
            title: `${pos.side === 'buy' ? 'LONG' : 'SHORT'} ${pos.quantity}`,
          })
        );
        if (pos.stopLoss) {
          priceLinesRef.current.push(
            series.createPriceLine({
              price: pos.stopLoss,
              color: '#F23645',
              lineWidth: 1,
              lineStyle: LineStyle.Dotted,
              axisLabelVisible: true,
              title: 'SL',
            })
          );
        }
        if (pos.takeProfit) {
          priceLinesRef.current.push(
            series.createPriceLine({
              price: pos.takeProfit,
              color: '#089981',
              lineWidth: 1,
              lineStyle: LineStyle.Dotted,
              axisLabelVisible: true,
              title: 'TP',
            })
          );
        }
      }
    }
  }, [
    alerts,
    positions,
    pane.symbol,
    pane.chartStyle,
    displayCandles.length,
    chartSettings.showAlertsOnChart,
    chartSettings.showPositionsOnChart,
    chartSettings.showHighLowLines,
  ]);

  // 7. Crosshair & Replay Click
  useEffect(() => {
    const chart = chartRef.current;
    if (!chart) return;

    const candleByTime = new Map<number, Candle>();
    for (const c of displayCandles) candleByTime.set(c.time, c);

    const onCrosshair = (param: { time?: Time }) => {
      if (param.time !== undefined) {
        const t = Number(param.time);
        const found = candleByTime.get(t);
        if (found) setHoveredBar(found);
        if (syncCrosshair && !isExternalCrosshairRef.current) {
          setSyncedCrosshairTime(t);
        }
      } else {
        setHoveredBar(null);
      }
    };

    const onRange = () => {
      refreshLastPriceY();
      const pos = chart.timeScale().scrollPosition();
      setScrolledAway(pos < -6);
    };

    const onClick = (param: { time?: Time }) => {
      setActivePaneId(pane.id);
      setPlusMenuOpen(null);
      if (isThisPaneInReplay && replay.isSelectingStart && param.time !== undefined) {
        const clickedTime = Number(param.time);
        const idx = fullCandles.findIndex((c) => c.time >= clickedTime);
        if (idx >= 0) {
          startReplayAt(pane.id, idx, fullCandles.length, fullCandles[idx].time);
        }
      }
    };

    chart.subscribeCrosshairMove(onCrosshair);
    chart.subscribeClick(onClick);
    chart.timeScale().subscribeVisibleLogicalRangeChange(onRange);

    return () => {
      chart.unsubscribeCrosshairMove(onCrosshair);
      chart.unsubscribeClick(onClick);
      chart.timeScale().unsubscribeVisibleLogicalRangeChange(onRange);
    };
  }, [
    displayCandles,
    fullCandles,
    syncCrosshair,
    isThisPaneInReplay,
    replay.isSelectingStart,
    pane.id,
    setActivePaneId,
    setSyncedCrosshairTime,
    startReplayAt,
    refreshLastPriceY,
  ]);

  useEffect(() => {
    const chart = chartRef.current;
    const series = mainSeriesRef.current;
    if (!chart || !series || !syncCrosshair || isActive || syncedCrosshairTime === null) return;

    const bar = displayCandles.find((c) => c.time === syncedCrosshairTime);
    if (!bar) return;

    isExternalCrosshairRef.current = true;
    try {
      chart.setCrosshairPosition(bar.close, bar.time as UTCTimestamp, series);
      setHoveredBar(bar);
    } catch {
      // ignore
    }
    isExternalCrosshairRef.current = false;
  }, [syncedCrosshairTime, syncCrosshair, isActive, displayCandles]);

  // 8. Export PNG Snapshot
  const prevSnapshotRef = useRef(snapshotCounter);
  useEffect(() => {
    if (snapshotCounter === prevSnapshotRef.current) return;
    prevSnapshotRef.current = snapshotCounter;
    if (!isActive || !chartRef.current) return;

    try {
      const chartCanvas = chartRef.current.takeScreenshot();
      const out = document.createElement('canvas');
      out.width = chartCanvas.width;
      out.height = chartCanvas.height;
      const ctx = out.getContext('2d');
      if (!ctx) return;

      ctx.drawImage(chartCanvas, 0, 0);
      if (drawingCanvasRef.current) {
        ctx.drawImage(drawingCanvasRef.current, 0, 0, out.width, out.height);
      }

      ctx.font = '600 13px Inter, sans-serif';
      ctx.fillStyle = theme === 'dark' ? 'rgba(255,255,255,0.45)' : 'rgba(15,23,42,0.45)';
      ctx.fillText(
        `ApexChart Pro  •  ${symbolInfo.displaySymbol}  •  ${pane.timeframe}`,
        18,
        out.height - 16
      );

      const url = out.toDataURL('image/png');
      const a = document.createElement('a');
      a.href = url;
      a.download = `${pane.symbol}_${pane.timeframe}_${Date.now()}.png`;
      a.click();

      addToast({
        title: 'Chart Snapshot Saved',
        description: `${symbolInfo.displaySymbol} (${pane.timeframe}) downloaded as PNG.`,
        variant: 'success',
      });
    } catch {
      addToast({ title: 'Snapshot Failed', variant: 'danger' });
    }
  }, [
    snapshotCounter,
    isActive,
    pane.symbol,
    pane.timeframe,
    symbolInfo.displaySymbol,
    theme,
    addToast,
  ]);

  const handleResetView = useCallback(() => {
    const chart = chartRef.current;
    if (!chart) return;
    chart.priceScale(chartSettings.priceScalePosition === 'left' ? 'left' : 'right').applyOptions({
      autoScale: true,
    });
    chart.timeScale().applyOptions({
      barSpacing: chartSettings.barSpacing,
      rightOffset: chartSettings.rightOffset,
    });
    chart.timeScale().scrollToRealTime();
    setScrolledAway(false);
  }, [chartSettings.priceScalePosition, chartSettings.barSpacing, chartSettings.rightOffset]);

  const handleFitContent = useCallback(() => {
    const chart = chartRef.current;
    if (!chart) return;
    chart.priceScale(chartSettings.priceScalePosition === 'left' ? 'left' : 'right').applyOptions({
      autoScale: true,
    });
    chart.timeScale().fitContent();
  }, [chartSettings.priceScalePosition]);

  const handleZoom = useCallback((factor: number) => {
    const chart = chartRef.current;
    if (!chart) return;
    const range = chart.timeScale().getVisibleLogicalRange();
    if (!range) return;
    const span = range.to - range.from;
    const nextSpan = Math.max(12, Math.min(600, span * factor));
    const delta = nextSpan - span;
    chart.timeScale().setVisibleLogicalRange({
      from: range.from - delta * 0.7,
      to: range.to + delta * 0.3,
    });
  }, []);

  const handlePan = useCallback((barsShift: number) => {
    const chart = chartRef.current;
    if (!chart) return;
    const range = chart.timeScale().getVisibleLogicalRange();
    if (!range) return;
    chart.timeScale().setVisibleLogicalRange({
      from: range.from + barsShift,
      to: range.to + barsShift,
    });
  }, []);

  const openContextMenuAt = useCallback(
    (
      clientX: number,
      clientY: number,
      localX: number,
      localY: number,
      drawingId: string | null
    ) => {
      const chart = chartRef.current;
      const series = mainSeriesRef.current;
      if (!chart || !series) return;

      const price = series.coordinateToPrice(localY);
      const logical = chart.timeScale().coordinateToLogical(localX);
      const lastPrice = displayCandles[displayCandles.length - 1]?.close ?? symbolInfo.basePrice;
      const resolvedPrice = price !== null ? Number(price) : lastPrice;

      let barIndex: number | null = null;
      let time: number | null = null;
      if (logical !== null && fullCandles.length > 0) {
        const idx = Math.max(0, Math.min(fullCandles.length - 1, Math.round(Number(logical))));
        barIndex = idx;
        time = fullCandles[idx].time;
      }

      const detectedDrawing = drawingId ?? hitTestDrawingRef.current?.(localX, localY) ?? null;

      setContextMenu({
        x: clientX,
        y: clientY,
        price: resolvedPrice,
        time,
        barIndex,
        drawingId: detectedDrawing,
      });
    },
    [displayCandles, fullCandles, symbolInfo.basePrice]
  );

  const handleContainerContextMenu = (e: React.MouseEvent<HTMLDivElement>) => {
    e.preventDefault();
    const rect = mainContainerRef.current?.getBoundingClientRect();
    if (!rect) return;
    openContextMenuAt(
      e.clientX,
      e.clientY,
      e.clientX - rect.left,
      e.clientY - rect.top,
      null
    );
  };

  const handleContainerDoubleClick = (e: React.MouseEvent<HTMLDivElement>) => {
    const rect = mainContainerRef.current?.getBoundingClientRect();
    if (!rect || !chartRef.current) return;
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;
    const isLeft = chartSettings.priceScalePosition === 'left';

    if ((isLeft && x <= 74) || (!isLeft && x >= chartSize.width - 74)) {
      chartRef.current.priceScale(isLeft ? 'left' : 'right').applyOptions({ autoScale: true });
      addToast({ title: 'Price Scale Auto-Fitted', variant: 'info' });
      return;
    }

    if (y >= chartSize.height - 28) {
      handleResetView();
      return;
    }

    openChartSettingsTab('symbol');
  };

  const handleContainerMouseMove = (e: React.MouseEvent<HTMLDivElement>) => {
    const rect = mainContainerRef.current?.getBoundingClientRect();
    if (!rect || !mainSeriesRef.current) return;
    const x = e.clientX - rect.left;
    const y = e.clientY - rect.top;
    const isLeft = chartSettings.priceScalePosition === 'left';
    const inPlotX = isLeft ? x >= 68 && x <= chartSize.width : x >= 0 && x <= chartSize.width - 68;

    if (inPlotX && y >= 8 && y <= chartSize.height - 32) {
      const p = mainSeriesRef.current.coordinateToPrice(y);
      if (p !== null) {
        setPriceAxisHover({ y, price: Number(p) });
        return;
      }
    }
    setPriceAxisHover(null);
  };

  const handleDrawHLineAtPrice = (price: number, type: 'hline' | 'hray' = 'hline') => {
    const t = displayCandles[displayCandles.length - 1]?.time ?? Math.floor(Date.now() / 1000);
    addDrawing({
      id: generateId('drw'),
      symbol: pane.symbol,
      type,
      points: [{ time: t, price }],
      color: drawingColor,
      lineWidth: 1.5,
      lineStyle: 'solid',
      createdAt: Date.now(),
    });
  };

  const handleStartReplayHere = (barIndex: number, time: number) => {
    startReplayAt(pane.id, barIndex, fullCandles.length, time);
    addToast({
      title: 'Bar Replay Started',
      description: `Cut at bar #${barIndex + 1}`,
      variant: 'warning',
    });
  };

  useEffect(() => {
    if (!isActive) return;
    const onKey = (e: KeyboardEvent) => {
      if (e.altKey && e.key.toLowerCase() === 'r' && e.shiftKey) {
        e.preventDefault();
        handleResetView();
      }
    };
    window.addEventListener('keydown', onKey);
    return () => window.removeEventListener('keydown', onKey);
  }, [isActive, handleResetView]);

  const activeBar = hoveredBar ?? displayCandles[displayCandles.length - 1] ?? null;
  const lastBar = displayCandles[displayCandles.length - 1] ?? null;
  const barChange = activeBar ? activeBar.close - activeBar.open : 0;
  const barChangePct = activeBar && activeBar.open > 0 ? (barChange / activeBar.open) * 100 : 0;
  const isBull = barChange >= 0;

  const spreadStep = effectiveMinMove * (effectivePrecision >= 4 ? 2 : 3);
  const bidPrice = lastBar
    ? Math.max(effectiveMinMove, lastBar.close - spreadStep)
    : symbolInfo.basePrice;
  const askPrice = lastBar ? lastBar.close + spreadStep : symbolInfo.basePrice;

  const subPaneIndicators = pane.indicators.filter((i) => i.placement === 'subpane');
  const overlayIndicators = pane.indicators.filter((i) => i.placement === 'overlay');
  const isLeftScale = chartSettings.priceScalePosition === 'left';

  return (
    <div
      onMouseDown={() => {
        setActivePaneId(pane.id);
        setPlusMenuOpen(null);
      }}
      className={cn(
        'relative flex h-full w-full flex-col overflow-hidden bg-white dark:bg-[#131722]',
        isMultiPane && isActive
          ? 'ring-1 ring-inset ring-[#2962FF]'
          : isMultiPane
          ? 'ring-1 ring-inset ring-slate-200 dark:ring-[#2A2E39]'
          : ''
      )}
    >
      <div
        className="relative flex-1 min-h-0 w-full"
        onMouseEnter={() => setHoverChart(true)}
        onMouseLeave={() => {
          setHoverChart(false);
          setPriceAxisHover(null);
        }}
        onMouseMove={handleContainerMouseMove}
        onContextMenu={handleContainerContextMenu}
        onDoubleClick={handleContainerDoubleClick}
      >
        {chartSettings.showWatermark && (
          <div
            style={{ opacity: (chartSettings.watermarkOpacity ?? 5) / 100 }}
            className="pointer-events-none absolute inset-0 z-[1] flex flex-col items-center justify-center select-none"
          >
            <div className="text-4xl sm:text-6xl font-black tracking-tight text-slate-900 dark:text-white">
              {symbolInfo.displaySymbol}
            </div>
            <div className="mt-1 text-lg sm:text-2xl font-bold tracking-widest uppercase text-slate-900 dark:text-white">
              {pane.timeframe} • {symbolInfo.exchange}
            </div>
          </div>
        )}

        {/* Top-Left Inline Legend */}
        <div
          className={`pointer-events-none absolute top-2 z-20 flex max-w-[82%] flex-col gap-1.5 ${
            isLeftScale ? 'left-20' : 'left-3'
          }`}
        >
          <div className="pointer-events-auto flex flex-wrap items-center gap-2 text-xs">
            {chartSettings.showStatusSymbol && (
              <>
                <span className="font-bold tracking-tight text-slate-900 dark:text-[#D1D4DC]">
                  {symbolInfo.displaySymbol}
                </span>
                <span className="text-[11px] font-medium text-[#787B86]">· {pane.timeframe} ·</span>
                <span className="text-[11px] font-medium text-[#787B86]">{symbolInfo.exchange}</span>
              </>
            )}

            {chartSettings.showStatusFeedBadge && (
              <button
                onClick={() => openChartSettingsTab('feeds')}
                title="Click to configure Twelve Data Live API / Data Feeds"
                className="inline-flex items-center gap-1 rounded px-1.5 py-0.5 text-[10px] font-bold transition hover:brightness-125"
              >
                {isThisPaneInReplay ? (
                  <span className="inline-flex items-center gap-1 text-amber-400">
                    <span className="h-1.5 w-1.5 animate-pulse rounded-full bg-amber-400" />
                    REPLAY
                  </span>
                ) : dataSource === 'twelvedata' ? (
                  <span className="inline-flex items-center gap-1 text-[#2962FF]">
                    <span className="h-2 w-2 animate-pulse rounded-full bg-[#2962FF]" />
                    TWELVE WS
                  </span>
                ) : dataSource === 'gold-proxy' ? (
                  <span className="inline-flex items-center gap-1 text-amber-400">
                    <span className="h-2 w-2 animate-pulse rounded-full bg-amber-400" />
                    LIVE GOLD
                  </span>
                ) : dataSource === 'binance' ? (
                  <span className="inline-flex items-center gap-1 text-[#089981]">
                    <span className="h-2 w-2 animate-pulse rounded-full bg-[#089981]" />
                    LIVE WS
                  </span>
                ) : (
                  <span className="inline-flex items-center gap-1 text-cyan-400">
                    <span className="h-2 w-2 rounded-full bg-cyan-400" />
                    SIM (Add API)
                  </span>
                )}
              </button>
            )}

            {activeBar && (
              <div className="hidden sm:flex items-center gap-2 font-mono text-[11px] tabular-nums">
                {chartSettings.showStatusOHLC && (
                  <>
                    <span className="text-[#787B86]">
                      O
                      <span className={cn('ml-0.5', isBull ? 'text-[#089981]' : 'text-[#F23645]')}>
                        {formatPrice(activeBar.open, effectivePrecision)}
                      </span>
                    </span>
                    <span className="text-[#787B86]">
                      H
                      <span className={cn('ml-0.5', isBull ? 'text-[#089981]' : 'text-[#F23645]')}>
                        {formatPrice(activeBar.high, effectivePrecision)}
                      </span>
                    </span>
                    <span className="text-[#787B86]">
                      L
                      <span className={cn('ml-0.5', isBull ? 'text-[#089981]' : 'text-[#F23645]')}>
                        {formatPrice(activeBar.low, effectivePrecision)}
                      </span>
                    </span>
                    <span className="text-[#787B86]">
                      C
                      <span className={cn('ml-0.5', isBull ? 'text-[#089981]' : 'text-[#F23645]')}>
                        {formatPrice(activeBar.close, effectivePrecision)}
                      </span>
                    </span>
                  </>
                )}
                {chartSettings.showStatusBarChange && (
                  <span className={cn('font-semibold', isBull ? 'text-[#089981]' : 'text-[#F23645]')}>
                    {barChange >= 0 ? '+' : ''}
                    {formatPrice(barChange, effectivePrecision)} ({barChange >= 0 ? '+' : ''}
                    {barChangePct.toFixed(2)}%)
                  </span>
                )}
                {chartSettings.showStatusVolume && (
                  <span className="text-[#787B86]">
                    Vol{' '}
                    <span className="text-slate-700 dark:text-[#D1D4DC]">
                      {formatCompactNumber(activeBar.volume)}
                    </span>
                  </span>
                )}
              </div>
            )}

            <button
              onClick={() => openChartSettingsTab('symbol')}
              title="Chart Settings"
              className="rounded p-0.5 text-[#787B86] hover:bg-white/10 hover:text-[#D1D4DC]"
            >
              <Settings2 className="h-3.5 w-3.5" />
            </button>
          </div>

          {chartSettings.showQuickTrade && lastBar && (
            <div className="pointer-events-auto inline-flex w-fit items-center overflow-hidden rounded border border-slate-200/90 bg-white/90 shadow-sm backdrop-blur dark:border-[#2A2E39] dark:bg-[#131722]/90">
              <button
                onClick={() => {
                  const q = Number(quickQty) || 0.1;
                  placeMarketOrder({
                    symbol: pane.symbol,
                    side: 'sell',
                    quantity: q,
                    price: bidPrice,
                  });
                }}
                className="flex flex-col items-start px-2.5 py-0.5 text-left transition hover:bg-[#F23645]/15"
              >
                <span className="font-mono text-[11px] font-bold leading-tight text-[#F23645] tabular-nums">
                  {formatPrice(bidPrice, effectivePrecision)}
                </span>
                <span className="text-[9px] font-semibold tracking-wider text-[#F23645]/80 uppercase">
                  SELL
                </span>
              </button>

              <input
                value={quickQty}
                onChange={(e) => setQuickQty(e.target.value)}
                title="Quick Order Quantity"
                className="h-full w-14 border-x border-slate-200 bg-transparent px-1 text-center font-mono text-[11px] font-semibold text-slate-900 outline-none dark:border-[#2A2E39] dark:text-[#D1D4DC]"
              />

              <button
                onClick={() => {
                  const q = Number(quickQty) || 0.1;
                  placeMarketOrder({
                    symbol: pane.symbol,
                    side: 'buy',
                    quantity: q,
                    price: askPrice,
                  });
                }}
                className="flex flex-col items-end px-2.5 py-0.5 text-right transition hover:bg-[#089981]/15"
              >
                <span className="font-mono text-[11px] font-bold leading-tight text-[#089981] tabular-nums">
                  {formatPrice(askPrice, effectivePrecision)}
                </span>
                <span className="text-[9px] font-semibold tracking-wider text-[#089981]/80 uppercase">
                  BUY
                </span>
              </button>
            </div>
          )}

          {chartSettings.showIndicatorTitles && overlayIndicators.length > 0 && (
            <div className="pointer-events-auto flex flex-col items-start gap-0.5">
              <button
                onClick={() => setLegendExpanded(!legendExpanded)}
                className="inline-flex items-center gap-1 rounded px-1 py-0.5 text-[10px] font-medium text-[#787B86] hover:bg-white/5 hover:text-[#D1D4DC]"
              >
                {legendExpanded ? (
                  <ChevronUp className="h-3 w-3" />
                ) : (
                  <ChevronDown className="h-3 w-3" />
                )}
                <span>Indicators ({overlayIndicators.length})</span>
              </button>

              {legendExpanded &&
                overlayIndicators.map((ind) => {
                  const paramStr = Object.values(ind.params).join(', ');
                  return (
                    <div
                      key={ind.id}
                      className={cn(
                        'group flex items-center gap-2 rounded px-1.5 py-0.5 text-[11px] transition hover:bg-black/5 dark:hover:bg-white/[0.06]',
                        !ind.visible && 'opacity-45'
                      )}
                    >
                      <span
                        className="h-2 w-2 rounded-full"
                        style={{ backgroundColor: ind.color }}
                      />
                      <span className="font-medium text-slate-700 dark:text-[#D1D4DC]">
                        {ind.name}
                        {chartSettings.showIndicatorArgs && paramStr ? ` (${paramStr})` : ''}
                      </span>
                      <div className="flex items-center gap-1 opacity-0 transition group-hover:opacity-100">
                        <button
                          onClick={() => setEditingIndicatorId(ind.id)}
                          title="Indicator Settings"
                          className="text-[#787B86] hover:text-white"
                        >
                          <Settings2 className="h-3 w-3" />
                        </button>
                        <button
                          onClick={() => toggleIndicatorVisibility(pane.id, ind.id)}
                          title={ind.visible ? 'Hide' : 'Show'}
                          className="text-[#787B86] hover:text-white"
                        >
                          {ind.visible ? (
                            <Eye className="h-3 w-3" />
                          ) : (
                            <EyeOff className="h-3 w-3" />
                          )}
                        </button>
                        <button
                          onClick={() => removeIndicator(pane.id, ind.id)}
                          title="Remove"
                          className="text-[#787B86] hover:text-rose-400"
                        >
                          <X className="h-3 w-3" />
                        </button>
                      </div>
                    </div>
                  );
                })}
            </div>
          )}
        </div>

        {isActive && <FloatingDrawingToolbar symbol={pane.symbol} />}

        {isThisPaneInReplay && replay.isSelectingStart && (
          <div className="pointer-events-none absolute inset-x-0 top-14 z-20 flex justify-center">
            <div className="flex items-center gap-2 rounded-full border border-amber-500/40 bg-amber-500/15 px-4 py-1.5 text-xs font-medium text-amber-300 shadow-lg backdrop-blur-md">
              <Scissors className="h-3.5 w-3.5" />
              <span>Click any candle on the chart to cut history & start Bar Replay here</span>
            </div>
          </div>
        )}

        {chartSettings.showCountdown && lastBar && lastPriceY !== null && !isThisPaneInReplay && (
          <div
            style={{
              top: Math.max(18, Math.min(chartSize.height - 46, lastPriceY + 11)),
            }}
            className={cn(
              'pointer-events-none absolute z-20 rounded px-1.5 py-[1px] font-mono text-[10px] font-bold text-white shadow-sm tabular-nums',
              isLeftScale ? 'left-[4px]' : 'right-[4px]',
              lastBar.close >= lastBar.open ? 'bg-[#089981]' : 'bg-[#F23645]'
            )}
          >
            {countdownText}
          </div>
        )}

        {priceAxisHover && !plusMenuOpen && (
          <button
            onClick={(e) => {
              e.stopPropagation();
              setPlusMenuOpen(priceAxisHover);
            }}
            style={{
              top: Math.max(8, Math.min(chartSize.height - 40, priceAxisHover.y - 10)),
            }}
            title={`Quick Action @ ${formatPrice(priceAxisHover.price, effectivePrecision)}`}
            className={cn(
              'absolute z-20 flex h-5 w-5 items-center justify-center rounded bg-[#2962FF] text-white shadow-md transition hover:scale-110',
              isLeftScale ? 'left-[72px]' : 'right-[72px]'
            )}
          >
            <Plus className="h-3.5 w-3.5" />
          </button>
        )}

        {plusMenuOpen && (
          <div
            style={{
              top: Math.max(8, Math.min(chartSize.height - 170, plusMenuOpen.y - 16)),
            }}
            onMouseDown={(e) => e.stopPropagation()}
            className={cn(
              'absolute z-40 w-52 rounded-lg border border-slate-200 bg-white py-1 text-xs shadow-2xl dark:border-[#2A2E39] dark:bg-[#1E222D] dark:text-[#D1D4DC]',
              isLeftScale ? 'left-[76px]' : 'right-[76px]'
            )}
          >
            <div className="border-b border-slate-200 px-3 py-1 font-mono text-[10px] font-bold text-[#2962FF] dark:border-[#2A2E39]">
              @ {formatPrice(plusMenuOpen.price, effectivePrecision)}
            </div>
            <button
              onClick={() => {
                const px = Number(plusMenuOpen.price.toFixed(effectivePrecision));
                const live = lastBar?.close ?? symbolInfo.basePrice;
                addAlert({
                  symbol: pane.symbol,
                  price: px,
                  condition: px >= live ? 'crossing_up' : 'crossing_down',
                  note: 'Quick price axis alert',
                });
                addToast({
                  title: `Alert Created @ ${formatPrice(px, effectivePrecision)}`,
                  variant: 'warning',
                });
                setPlusMenuOpen(null);
              }}
              className="flex w-full items-center gap-2 px-3 py-1.5 hover:bg-slate-100 dark:hover:bg-[#2A2E39]"
            >
              <Bell className="h-3.5 w-3.5 text-amber-400" />
              <span>Add Alert</span>
            </button>
            <button
              onClick={() => {
                handleDrawHLineAtPrice(
                  Number(plusMenuOpen.price.toFixed(effectivePrecision)),
                  'hline'
                );
                setPlusMenuOpen(null);
              }}
              className="flex w-full items-center gap-2 px-3 py-1.5 hover:bg-slate-100 dark:hover:bg-[#2A2E39]"
            >
              <Minus className="h-3.5 w-3.5 text-[#00D4AA]" />
              <span>Horizontal Line</span>
            </button>
            <button
              onClick={() => {
                handleDrawHLineAtPrice(
                  Number(plusMenuOpen.price.toFixed(effectivePrecision)),
                  'hray'
                );
                setPlusMenuOpen(null);
              }}
              className="flex w-full items-center gap-2 px-3 py-1.5 hover:bg-slate-100 dark:hover:bg-[#2A2E39]"
            >
              <MoveRight className="h-3.5 w-3.5 text-[#00D4AA]" />
              <span>Horizontal Ray</span>
            </button>
            <button
              onClick={() => {
                const px = Number(plusMenuOpen.price.toFixed(effectivePrecision));
                const qty =
                  symbolInfo.category === 'Crypto' ? 0.25 : symbolInfo.category === 'IDX' ? 1000 : 10;
                placeLimitOrder({
                  symbol: pane.symbol,
                  side: 'buy',
                  quantity: qty,
                  limitPrice: px,
                });
                setPlusMenuOpen(null);
              }}
              className="flex w-full items-center gap-2 px-3 py-1.5 text-[#089981] hover:bg-[#089981]/10"
            >
              <TrendingUp className="h-3.5 w-3.5" />
              <span>Place Buy Limit</span>
            </button>
            <button
              onClick={() => {
                const px = Number(plusMenuOpen.price.toFixed(effectivePrecision));
                const qty =
                  symbolInfo.category === 'Crypto' ? 0.25 : symbolInfo.category === 'IDX' ? 1000 : 10;
                placeLimitOrder({
                  symbol: pane.symbol,
                  side: 'sell',
                  quantity: qty,
                  limitPrice: px,
                });
                setPlusMenuOpen(null);
              }}
              className="flex w-full items-center gap-2 px-3 py-1.5 text-[#F23645] hover:bg-[#F23645]/10"
            >
              <TrendingDown className="h-3.5 w-3.5" />
              <span>Place Sell Limit</span>
            </button>
          </div>
        )}

        <div
          className={cn(
            'pointer-events-auto absolute bottom-9 left-1/2 z-20 flex -translate-x-1/2 items-center gap-1 rounded-lg border border-slate-200/80 bg-white/90 px-1.5 py-1 shadow-lg backdrop-blur transition-opacity duration-200 dark:border-[#2A2E39] dark:bg-[#1E222D]/92',
            hoverChart || scrolledAway ? 'opacity-100' : 'opacity-0 pointer-events-none'
          )}
        >
          <button
            onClick={() => handleZoom(1.35)}
            title="Zoom Out"
            className="rounded p-1 text-slate-600 hover:bg-slate-100 dark:text-[#D1D4DC] dark:hover:bg-[#2A2E39]"
          >
            <ZoomOut className="h-3.5 w-3.5" />
          </button>
          <button
            onClick={() => handleZoom(0.74)}
            title="Zoom In"
            className="rounded p-1 text-slate-600 hover:bg-slate-100 dark:text-[#D1D4DC] dark:hover:bg-[#2A2E39]"
          >
            <ZoomIn className="h-3.5 w-3.5" />
          </button>
          <div className="h-3.5 w-px bg-slate-200 dark:bg-[#2A2E39]" />
          <button
            onClick={() => handlePan(-20)}
            title="Scroll Left"
            className="rounded p-1 text-slate-600 hover:bg-slate-100 dark:text-[#D1D4DC] dark:hover:bg-[#2A2E39]"
          >
            <ChevronLeft className="h-3.5 w-3.5" />
          </button>
          <button
            onClick={() => handlePan(20)}
            title="Scroll Right"
            className="rounded p-1 text-slate-600 hover:bg-slate-100 dark:text-[#D1D4DC] dark:hover:bg-[#2A2E39]"
          >
            <ChevronRight className="h-3.5 w-3.5" />
          </button>
          <div className="h-3.5 w-px bg-slate-200 dark:bg-[#2A2E39]" />
          <button
            onClick={handleResetView}
            title="Reset Chart View (Alt+Shift+R)"
            className="flex items-center gap-1 rounded px-1.5 py-0.5 text-[11px] font-medium text-slate-600 hover:bg-slate-100 dark:text-[#D1D4DC] dark:hover:bg-[#2A2E39]"
          >
            <RotateCcw className="h-3 w-3" />
            <span>Reset</span>
          </button>
          {scrolledAway && (
            <button
              onClick={handleResetView}
              title="Jump to Latest Candle"
              className="flex items-center gap-0.5 rounded bg-[#2962FF] px-1.5 py-0.5 text-[10px] font-bold text-white hover:bg-blue-500"
            >
              <ChevronsRight className="h-3 w-3" />
              <span>Latest</span>
            </button>
          )}
        </div>

        {loading && (
          <div className="absolute inset-0 z-30 flex items-center justify-center bg-[#131722]/70 backdrop-blur-sm">
            <div className="flex items-center gap-2.5 rounded-lg border border-[#2A2E39] bg-[#1E222D] px-4 py-2.5 text-xs text-[#D1D4DC] shadow-xl">
              <div className="h-4 w-4 animate-spin rounded-full border-2 border-[#2962FF] border-t-transparent" />
              <span>
                Loading {symbolInfo.displaySymbol} ({pane.timeframe})...
              </span>
            </div>
          </div>
        )}

        <div ref={mainContainerRef} className="h-full w-full" />

        <DrawingOverlay
          chart={chartRef.current}
          series={mainSeriesRef.current}
          candles={displayCandles}
          symbol={pane.symbol}
          timeframe={pane.timeframe}
          precision={effectivePrecision}
          width={chartSize.width}
          height={chartSize.height}
          theme={theme}
          onCanvasReady={(c) => {
            drawingCanvasRef.current = c;
          }}
          onHitTestExpose={(fn) => {
            hitTestDrawingRef.current = fn;
          }}
          onRequestContextMenu={openContextMenuAt}
        />
      </div>

      {subPaneIndicators.map((ind) => (
        <SubPaneChart
          key={ind.id}
          indicator={ind}
          candles={displayCandles}
          timeframe={pane.timeframe}
          theme={theme}
          bgColorDark={chartSettings.bgColorDark || '#131722'}
          mainChart={chartRef.current}
          onToggleVisibility={() => toggleIndicatorVisibility(pane.id, ind.id)}
          onRemove={() => removeIndicator(pane.id, ind.id)}
        />
      ))}

      {/* Bottom Chart Range Presets + Live Timezone Clock + Scale Controls */}
      <div className="flex h-7 w-full shrink-0 items-center justify-between border-t border-slate-200 bg-white px-2.5 text-[11px] dark:border-[#2A2E39] dark:bg-[#131722]">
        <div className="flex items-center gap-0.5">
          {RANGE_PRESETS.map((preset) => (
            <button
              key={preset.label}
              onClick={() => {
                updatePaneTimeframe(pane.id, preset.tf);
                window.setTimeout(() => {
                  chartRef.current?.timeScale().fitContent();
                }, 250);
              }}
              className={cn(
                'rounded px-1.5 py-0.5 font-medium transition',
                pane.timeframe === preset.tf
                  ? 'text-[#2962FF] font-bold bg-[#2962FF]/10'
                  : 'text-slate-500 hover:bg-slate-100 hover:text-slate-900 dark:text-[#787B86] dark:hover:bg-[#1E222D] dark:hover:text-[#D1D4DC]'
              )}
            >
              {preset.label}
            </button>
          ))}
        </div>

        <div className="flex items-center gap-2 font-mono text-[11px]">
          <button
            onClick={() => openChartSettingsTab('trading')}
            title="Click to configure Timezone"
            className="hidden sm:inline text-slate-500 hover:text-slate-900 dark:text-[#787B86] dark:hover:text-[#D1D4DC] tabular-nums"
          >
            {clockText}
          </button>

          <div className="h-3 w-px bg-slate-200 dark:bg-[#2A2E39]" />

          <button
            onClick={() =>
              updatePaneScaleMode(
                pane.id,
                pane.scaleMode === 'percentage' ? 'normal' : 'percentage'
              )
            }
            className={cn(
              'rounded px-1 py-0.5 text-[10px] font-semibold transition',
              pane.scaleMode === 'percentage'
                ? 'text-[#2962FF]'
                : 'text-[#787B86] hover:text-[#D1D4DC]'
            )}
          >
            %
          </button>
          <button
            onClick={() =>
              updatePaneScaleMode(
                pane.id,
                pane.scaleMode === 'logarithmic' ? 'normal' : 'logarithmic'
              )
            }
            className={cn(
              'rounded px-1 py-0.5 text-[10px] font-semibold transition',
              pane.scaleMode === 'logarithmic'
                ? 'text-[#2962FF]'
                : 'text-[#787B86] hover:text-[#D1D4DC]'
            )}
          >
            log
          </button>
          <button
            onClick={handleResetView}
            className="rounded px-1 py-0.5 text-[10px] font-semibold text-[#2962FF] hover:bg-[#2962FF]/10"
          >
            auto
          </button>
        </div>
      </div>

      <ChartContextMenu
        menu={contextMenu}
        paneId={pane.id}
        symbol={pane.symbol}
        currentStyle={pane.chartStyle}
        currentScale={pane.scaleMode}
        onClose={() => setContextMenu(null)}
        onResetView={handleResetView}
        onFitContent={handleFitContent}
        onDrawHLineAtPrice={handleDrawHLineAtPrice}
        onStartReplayHere={handleStartReplayHere}
      />
    </div>
  );
};
