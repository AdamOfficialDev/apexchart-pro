function createIndicatorInstance(
  type: IndicatorType,
  overrideParams?: Record<string, number>,
  overrideColor?: string
): IndicatorConfig {
  const meta = INDICATOR_CATALOG.find((i) => i.type === type) ?? INDICATOR_CATALOG[0];
  return {
    id: generateId('ind'),
    type: meta.type,
    name: meta.shortName,
    placement: meta.placement,
    visible: true,
    color: overrideColor ?? meta.defaultColor,
    secondaryColor: meta.secondaryColor,
    tertiaryColor: meta.tertiaryColor,
    lineWidth: 2,
    lineStyle: 'solid',
    source: 'close',
    offset: 0,
    upperLevel: meta.defaultUpperLevel,
    lowerLevel: meta.defaultLowerLevel,
    showBands: true,
    showLastValue: true,
    visibility: { seconds: true, minutes: true, hours: true, days: true },
    params: { ...meta.defaultParams, ...(overrideParams ?? {}) },
  };
}

export const useTradingStore = create<TradingStore>()(
  persist(
    (set, get) => ({
      theme: 'dark',
      toggleTheme: () =>
        set((state) => {
          const next = state.theme === 'dark' ? 'light' : 'dark';
          if (typeof document !== 'undefined') {
            document.documentElement.classList.toggle('dark', next === 'dark');
          }
          return { theme: next };
        }),

      chartSettings: DEFAULT_CHART_SETTINGS,
      updateChartSettings: (patch) =>
        set((state) => ({
          chartSettings: { ...state.chartSettings, ...patch },
        })),
      resetChartSettings: () => set({ chartSettings: DEFAULT_CHART_SETTINGS }),

      smartOverlays: DEFAULT_SMART_OVERLAYS,
      updateSmartOverlays: (patch) =>
        set((state) => ({
          smartOverlays: { ...state.smartOverlays, ...patch },
        })),

      rsiDivConfig: DEFAULT_RSI_DIV_CONFIG,
      updateRsiDivConfig: (patch) =>
        set((state) => ({
          rsiDivConfig: { ...state.rsiDivConfig, ...patch },
        })),
      resetRsiDivConfig: () => set({ rsiDivConfig: DEFAULT_RSI_DIV_CONFIG }),

      dataFeed: DEFAULT_DATA_FEED,
      updateDataFeed: (patch) =>
        set((state) => ({
          dataFeed: { ...state.dataFeed, ...patch },
        })),

      rightSidebarTab: 'watchlist',
      setRightSidebarTab: (tab) => set({ rightSidebarTab: tab }),
      bottomPanelOpen: false,
      setBottomPanelOpen: (open) => set({ bottomPanelOpen: open }),
      bottomPanelTab: 'positions',
      setBottomPanelTab: (tab) => set({ bottomPanelTab: tab }),
      panelSizes: { rightWidth: 288, bottomHeight: 256, subPaneHeight: 112 },
      updatePanelSizes: (patch) =>
        set((state) => ({
          panelSizes: { ...state.panelSizes, ...patch },
        })),

      searchModalOpen: false,
      setSearchModalOpen: (open) => set({ searchModalOpen: open }),
      indicatorsModalOpen: false,
      setIndicatorsModalOpen: (open) => set({ indicatorsModalOpen: open }),
      shortcutsModalOpen: false,
      setShortcutsModalOpen: (open) => set({ shortcutsModalOpen: open }),
      compareModalOpen: false,
      setCompareModalOpen: (open) => set({ compareModalOpen: open }),
      templatesModalOpen: false,
      setTemplatesModalOpen: (open) => set({ templatesModalOpen: open }),
      chartSettingsModalOpen: false,
      setChartSettingsModalOpen: (open) => set({ chartSettingsModalOpen: open }),
      chartSettingsInitialTab: 'symbol',
      openChartSettingsTab: (tab) =>
        set({ chartSettingsInitialTab: tab, chartSettingsModalOpen: true }),
      editingDrawingId: null,
      setEditingDrawingId: (editingDrawingId) => set({ editingDrawingId }),
      editingIndicatorId: null,
      setEditingIndicatorId: (editingIndicatorId) => set({ editingIndicatorId }),
      snapshotRequestCounter: 0,
      triggerSnapshot: () =>
        set((s) => ({ snapshotRequestCounter: s.snapshotRequestCounter + 1 })),

      layout: '1x1',
      setLayout: (layout) => set({ layout, maximizedPaneId: null }),
      maximizedPaneId: null,
      setMaximizedPaneId: (maximizedPaneId) => set({ maximizedPaneId }),
      panes: DEFAULT_PANES,
      activePaneId: 'pane-1',
      setActivePaneId: (paneId) => set({ activePaneId: paneId }),

      favoriteTimeframes: ['1s', '5s', '1m', '5m', '15m', '1h', '4h', '1D'],
      toggleFavoriteTimeframe: (tf) =>
        set((state) => {
          const exists = state.favoriteTimeframes.includes(tf);
          if (exists && state.favoriteTimeframes.length <= 2) return state;
          const next = exists
            ? state.favoriteTimeframes.filter((t) => t !== tf)
            : [...state.favoriteTimeframes, tf];
          return { favoriteTimeframes: next };
        }),

      syncCrosshair: true,
      setSyncCrosshair: (syncCrosshair) => set({ syncCrosshair }),
      syncSymbol: false,
      setSyncSymbol: (syncSymbol) => set({ syncSymbol }),
      syncTimeframe: false,
      setSyncTimeframe: (syncTimeframe) => set({ syncTimeframe }),
      syncedCrosshairTime: null,
      setSyncedCrosshairTime: (syncedCrosshairTime) => set({ syncedCrosshairTime }),

      updatePaneSymbol: (paneId, symbol) =>
        set((state) => ({
          panes: state.panes.map((p) =>
            state.syncSymbol || p.id === paneId ? { ...p, symbol } : p
          ),
        })),

      updatePaneTimeframe: (paneId, timeframe) =>
        set((state) => ({
          panes: state.panes.map((p) =>
            state.syncTimeframe || p.id === paneId ? { ...p, timeframe } : p
          ),
        })),

      updatePaneChartStyle: (paneId, chartStyle) =>
        set((state) => ({
          panes: state.panes.map((p) => (p.id === paneId ? { ...p, chartStyle } : p)),
        })),

      updatePaneScaleMode: (paneId, scaleMode) =>
        set((state) => ({
          panes: state.panes.map((p) => (p.id === paneId ? { ...p, scaleMode } : p)),
        })),

      setPaneCompareSymbol: (paneId, compareSymbol) =>
        set((state) => ({
          panes: state.panes.map((p) =>
            p.id === paneId ? { ...p, compareSymbol } : p
          ),
        })),

      chartTemplates: DEFAULT_CHART_TEMPLATES,
      saveChartTemplate: (name, paneId) => {
        const pane = get().panes.find((p) => p.id === paneId);
        if (!pane || !name.trim()) return;
        const tpl: ChartTemplate = {
          id: generateId('tpl'),
          name: name.trim(),
          chartStyle: pane.chartStyle,
          indicators: pane.indicators.map(({ id: _id, ...rest }) => ({ ...rest })),
          smartOverlays: { ...get().smartOverlays },
        };
        set((state) => ({
          chartTemplates: [tpl, ...state.chartTemplates],
        }));
        get().addToast({ title: `Template Saved: ${tpl.name}`, variant: 'success' });
      },

      applyChartTemplate: (paneId, templateId) => {
        const tpl = get().chartTemplates.find((t) => t.id === templateId);
        if (!tpl) return;
        set((state) => ({
          panes: state.panes.map((p) =>
            p.id === paneId
              ? {
                  ...p,
                  chartStyle: tpl.chartStyle,
                  indicators: tpl.indicators.map((ind) => ({
                    ...ind,
                    id: generateId('ind'),
                  })),
                }
              : p
          ),
          smartOverlays: { ...tpl.smartOverlays },
        }));
        get().addToast({ title: `Template Applied: ${tpl.name}`, variant: 'info' });
      },

      deleteChartTemplate: (templateId) =>
        set((state) => ({
          chartTemplates: state.chartTemplates.filter((t) => t.id !== templateId),
        })),

      backtestTrades: [],
      showBacktestOnChart: false,
      setBacktestOverlay: (backtestTrades, showBacktestOnChart) =>
        set({ backtestTrades, showBacktestOnChart }),

      addIndicator: (paneId, type) =>
        set((state) => ({
          panes: state.panes.map((p) =>
            p.id === paneId
              ? { ...p, indicators: [...p.indicators, createIndicatorInstance(type)] }
              : p
          ),
        })),

      removeIndicator: (paneId, indicatorId) =>
        set((state) => ({
          panes: state.panes.map((p) =>
            p.id === paneId
              ? { ...p, indicators: p.indicators.filter((ind) => ind.id !== indicatorId) }
              : p
          ),
        })),

      toggleIndicatorVisibility: (paneId, indicatorId) =>
        set((state) => ({
          panes: state.panes.map((p) =>
            p.id === paneId
              ? {
                  ...p,
                  indicators: p.indicators.map((ind) =>
                    ind.id === indicatorId ? { ...ind, visible: !ind.visible } : ind
                  ),
                }
              : p
          ),
        })),

      updateIndicator: (paneId, indicatorId, patch) =>
        set((state) => ({
          panes: state.panes.map((p) =>
            p.id === paneId
              ? {
                  ...p,
                  indicators: p.indicators.map((ind) =>
                    ind.id === indicatorId ? { ...ind, ...patch } : ind
                  ),
                }
              : p
          ),
        })),

      activeDrawingTool: 'cursor',
      setActiveDrawingTool: (activeDrawingTool) => set({ activeDrawingTool }),
      groupDefaultTools: {
        cursors: 'cursor',
        lines: 'trendline',
        fib_gann: 'fib',
        patterns: 'xabcd',
        forecast: 'long_pos',
        shapes: 'rect',
        annotations: 'brush',
        measure: 'ruler',
      },
      setGroupTool: (group, tool) =>
        set((state) => ({
          groupDefaultTools: { ...state.groupDefaultTools, [group]: tool },
          activeDrawingTool: tool,
        })),

      drawingColor: '#2962FF',
      setDrawingColor: (drawingColor) => set({ drawingColor }),
      drawingLineWidth: 2,
      setDrawingLineWidth: (drawingLineWidth) => set({ drawingLineWidth }),
      drawingLineStyle: 'solid',
      setDrawingLineStyle: (drawingLineStyle) => set({ drawingLineStyle }),
      stayInDrawingMode: false,
      toggleStayInDrawingMode: () => set((s) => ({ stayInDrawingMode: !s.stayInDrawingMode })),
      magnetMode: false,
      toggleMagnetMode: () => set((s) => ({ magnetMode: !s.magnetMode })),
      drawingsLocked: false,
      toggleDrawingsLocked: () => set((s) => ({ drawingsLocked: !s.drawingsLocked })),
      drawingsVisible: true,
      toggleDrawingsVisible: () => set((s) => ({ drawingsVisible: !s.drawingsVisible })),
      selectedDrawingId: null,
      setSelectedDrawingId: (selectedDrawingId) => set({ selectedDrawingId }),
      drawingsBySymbol: {},
      redoBySymbol: {},

      addDrawing: (drawing) =>
        set((state) => {
          const list = state.drawingsBySymbol[drawing.symbol] ?? [];
          return {
            drawingsBySymbol: {
              ...state.drawingsBySymbol,
              [drawing.symbol]: [...list, drawing],
            },
            redoBySymbol: {
              ...state.redoBySymbol,
              [drawing.symbol]: [],
            },
            selectedDrawingId: drawing.id,
          };
        }),

      updateDrawing: (symbol, id, patch) =>
        set((state) => {
          const list = state.drawingsBySymbol[symbol] ?? [];
          return {
            drawingsBySymbol: {
              ...state.drawingsBySymbol,
              [symbol]: list.map((d) => (d.id === id ? { ...d, ...patch } : d)),
            },
          };
        }),

      duplicateDrawing: (symbol, id) =>
        set((state) => {
          const list = state.drawingsBySymbol[symbol] ?? [];
          const src = list.find((d) => d.id === id);
          if (!src) return state;
          const clone: DrawingObject = {
            ...src,
            id: generateId('drw'),
            locked: false,
            points: src.points.map((pt) => ({
              time: pt.time,
              price: pt.price * 1.003,
            })),
            createdAt: Date.now(),
          };
          return {
            drawingsBySymbol: {
              ...state.drawingsBySymbol,
              [symbol]: [...list, clone],
            },
            selectedDrawingId: clone.id,
          };
        }),

      removeDrawing: (symbol, id) =>
        set((state) => {
          const list = state.drawingsBySymbol[symbol] ?? [];
          const removed = list.find((d) => d.id === id);
          const redo = state.redoBySymbol[symbol] ?? [];
          return {
            drawingsBySymbol: {
              ...state.drawingsBySymbol,
              [symbol]: list.filter((d) => d.id !== id),
            },

import { useEffect, useRef } from 'react';
import { getSymbolInfo, SYMBOLS } from '@/data/symbols';
import { formatPrice, playAlertSound } from '@/lib/utils';
import { fetchBinanceWatchlistTickers } from '@/services/marketData';
import { useTradingStore } from '@/store/useTradingStore';

export function useAlertMonitor() {
  const quotes = useTradingStore((s) => s.quotes);
  const alerts = useTradingStore((s) => s.alerts);
  const watchlist = useTradingStore((s) => s.watchlist);
  const panes = useTradingStore((s) => s.panes);
  const layout = useTradingStore((s) => s.layout);
  const prevPricesRef = useRef<Record<string, number>>({});

  // Real-time Binance 24h batch ticker sync for Watchlist + fallback drift for offline/non-Binance
  useEffect(() => {
    const syncWatchlist = async () => {
      const store = useTradingStore.getState();
      const allSymbols = Array.from(new Set([...store.watchlist, ...SYMBOLS.map((s) => s.symbol)]));
      const binanceMap = new Map<string, string[]>();

      for (const sym of allSymbols) {
        const info = getSymbolInfo(sym);
        if (info.binanceSymbol) {
          const arr = binanceMap.get(info.binanceSymbol) ?? [];
          arr.push(sym);
          binanceMap.set(info.binanceSymbol, arr);
        }
      }

      const liveBatch = await fetchBinanceWatchlistTickers(Array.from(binanceMap.keys()));

      for (const [bSym, targetSyms] of binanceMap.entries()) {
        const item = liveBatch[bSym];
        if (item) {
          for (const sym of targetSyms) {
            store.updateQuote(
              sym,
              item.price,
              sym === 'XAUUSD' ? 'gold-proxy' : 'binance',
              0,
              item.changePercent
            );
          }
        }
      }

      const visiblePaneCount =
        layout === '1x1' ? 1 : layout === '1x2' || layout === '2x1' ? 2 : layout === '1+2' ? 3 : 4;
      const activeChartSymbols = new Set(
        store.panes.slice(0, visiblePaneCount).map((p) => p.symbol)
      );

      for (const sym of store.watchlist) {
        if (activeChartSymbols.has(sym)) continue;
        const info = getSymbolInfo(sym);
        if (info.binanceSymbol && liveBatch[info.binanceSymbol]) continue;

        const curr = store.quotes[sym]?.price ?? info.basePrice;
        const drift = (Math.random() - 0.495) * info.volatility * 0.08 * curr;
        const nextPrice = Math.max(info.minMove, Number((curr + drift).toFixed(info.precision)));
        store.updateQuote(sym, nextPrice, 'simulated', Math.round(Math.random() * 50));
      }
    };

    syncWatchlist();
    const timer = window.setInterval(syncWatchlist, 5000);
    return () => window.clearInterval(timer);
  }, [layout, watchlist, panes]);

  useEffect(() => {
    const store = useTradingStore.getState();
    const activeAlerts = alerts.filter((a) => a.active);
    if (activeAlerts.length === 0) {
      for (const [sym, q] of Object.entries(quotes)) {
        prevPricesRef.current[sym] = q.price;
      }
      return;
    }

    for (const alert of activeAlerts) {
      const q = quotes[alert.symbol];
      if (!q) continue;
      const prev = prevPricesRef.current[alert.symbol];
      const curr = q.price;
      if (prev === undefined) continue;

      let triggered = false;
      if (alert.condition === 'crossing_up') {
        triggered = prev < alert.price && curr >= alert.price;
      } else if (alert.condition === 'crossing_down') {
        triggered = prev > alert.price && curr <= alert.price;
      } else {
        triggered =
          (prev < alert.price && curr >= alert.price) ||
          (prev > alert.price && curr <= alert.price);
      }

      if (triggered) {
        const info = getSymbolInfo(alert.symbol);
        store.markAlertTriggered(alert.id);
        if (store.chartSettings.playAlertSound) {
          playAlertSound();
        }
        store.addToast({
          title: `Alert Triggered: ${info.displaySymbol}`,
          description: `${alert.note ? alert.note + ' — ' : ''}Price hit ${formatPrice(
            curr,
            info.precision
          )}`,
          variant: 'warning',
        });
      }
    }

    for (const [sym, q] of Object.entries(quotes)) {
      prevPricesRef.current[sym] = q.price;
    }
  }, [quotes, alerts]);
}
