export function isVisibleOnTimeframeGroup(
  vis: TimeframeVisibility | undefined,
  group: TimeframeGroup
): boolean {
  if (!vis) return true;
  if (group === 'Seconds') return vis.seconds;
  if (group === 'Minutes') return vis.minutes;
  if (group === 'Hours') return vis.hours;
  return vis.days;
}

export function getSymbolInfo(rawSymbol: string): SymbolInfo {
  const clean = rawSymbol.trim().toUpperCase().replace(/\s+/g, '');
  const found = SYMBOLS.find(
    (s) => s.symbol.toUpperCase() === clean || s.displaySymbol.toUpperCase() === clean
  );
  if (found) return found;

  if (clean.endsWith('USDT') || clean.endsWith('USDC') || clean.endsWith('FDUSD')) {
    const base = clean.replace(/(USDT|USDC|FDUSD)$/, '');
    const isMicro = base === 'PEPE' || base === 'SHIB' || base === 'BONK' || base === 'FLOKI';
    return {
      symbol: clean,
      displaySymbol: `${base}/USDT`,
      name: `${base} / TetherUS (Custom)`,
      category: 'Crypto',
      exchange: 'BINANCE',
      basePrice: isMicro ? 0.000012 : 12.5,
      precision: isMicro ? 8 : 4,
      minMove: isMicro ? 0.00000001 : 0.0001,
      volatility: 0.045,
      binanceSymbol: clean,
      twelveDataSymbol: `${base}/USD`,
      currency: 'USDT',
      description: `Live Binance spot pair ${base}/USDT.`,
    };
  }

  if (clean.endsWith('.JK')) {
    const code = clean.replace('.JK', '');
    return {
      symbol: clean,
      displaySymbol: code,
      name: `${code} Tbk (IDX)`,
      category: 'IDX',
      exchange: 'IDX',
      basePrice: 2500,
      precision: 0,
      minMove: 10,
      volatility: 0.02,
      twelveDataSymbol: `${code}:IDX`,
      currency: 'IDR',
      description: `Indonesian Equity ${code} on IDX.`,
    };
  }

  if (/^[A-Z]{6}$/.test(clean)) {
    const b = clean.slice(0, 3);
    const q = clean.slice(3, 6);
    const isJpy = q === 'JPY';
    return {
      symbol: clean,
      displaySymbol: `${b}/${q}`,
      name: `${b} / ${q} Spot FX`,
      category: 'Forex',
      exchange: 'FX',
      basePrice: isJpy ? 150 : 1.1,
      precision: isJpy ? 3 : 5,
      minMove: isJpy ? 0.001 : 0.00001,
      volatility: 0.006,
      twelveDataSymbol: `${b}/${q}`,
      fxPair: [b, q],
      currency: q,
      description: `Foreign exchange spot pair ${b}/${q}.`,
    };
  }

  return {
    symbol: clean,
    displaySymbol: clean,
    name: `${clean} (Custom Instrument)`,
    category: 'Stocks',
    exchange: 'CUSTOM',
    basePrice: 150,
    precision: 2,
    minMove: 0.01,
    volatility: 0.022,
    twelveDataSymbol: clean,
    currency: 'USD',
    description: `Custom instrument ${clean}.`,
  };
}

function createIndicatorInstance(
  type: IndicatorType,
  overrideParams?: Record<string, number>,
  overrideColor?: string
): IndicatorConfig {
  const meta = INDICATOR_CATALOG.find((i) => i.type === type) ?? INDICATOR_CATALOG[0];
  return {
    id: generateId('ind'),
    type: meta.type,
    name: meta.shortName,
    placement: meta.placement,
    visible: true,
    color: overrideColor ?? meta.defaultColor,
    secondaryColor: meta.secondaryColor,
    tertiaryColor: meta.tertiaryColor,
    lineWidth: 2,
    lineStyle: 'solid',
    source: 'close',
    offset: 0,
    upperLevel: meta.defaultUpperLevel,
    lowerLevel: meta.defaultLowerLevel,
    showBands: true,
    showLastValue: true,
    visibility: { seconds: true, minutes: true, hours: true, days: true },
    params: { ...meta.defaultParams, ...(overrideParams ?? {}) },
  };
}
