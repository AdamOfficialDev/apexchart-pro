export function isVisibleOnTimeframeGroup(
  vis: TimeframeVisibility | undefined,
  group: TimeframeGroup
): boolean {
  if (!vis) return true;
  if (group === 'Seconds') return vis.seconds;
  if (group === 'Minutes') return vis.minutes;
  if (group === 'Hours') return vis.hours;
  return vis.days;
}

import { Candle, PriceSource } from '@/types/trading';

export interface TimeValuePoint {
  time: number;
  value: number;
}

export interface ColoredPoint {
  time: number;
  value: number;
  color?: string;
}

export function getCandleSourceValue(c: Candle, source: PriceSource = 'close'): number {
  switch (source) {
    case 'open':
      return c.open;
    case 'high':
      return c.high;
    case 'low':
      return c.low;
    case 'hl2':
      return (c.high + c.low) / 2;
    case 'hlc3':
      return (c.high + c.low + c.close) / 3;
    case 'ohlc4':
      return (c.open + c.high + c.low + c.close) / 4;
    case 'close':
    default:
      return c.close;
  }
}

export function applyIndicatorOffset<T extends { time: number }>(
  points: T[],
  offset: number | undefined,
  tfSec: number
): T[] {
  if (!offset || offset === 0) return points;
  return points.map((p) => ({
    ...p,
    time: p.time + offset * tfSec,
  }));
}

export function computeHeikinAshi(candles: Candle[]): Candle[] {
  if (candles.length === 0) return [];
  const result: Candle[] = [];
  for (let i = 0; i < candles.length; i++) {
    const c = candles[i];
    const haClose = (c.open + c.high + c.low + c.close) / 4;
    const haOpen =
      i === 0 ? (c.open + c.close) / 2 : (result[i - 1].open + result[i - 1].close) / 2;
    const haHigh = Math.max(c.high, haOpen, haClose);
    const haLow = Math.min(c.low, haOpen, haClose);
    result.push({
      time: c.time,
      open: haOpen,
      high: haHigh,
      low: haLow,
      close: haClose,
      volume: c.volume,
    });
  }
  return result;
}

export function computeSMA(
  candles: Candle[],
  period = 20,
  source: PriceSource = 'close'
): TimeValuePoint[] {
  if (candles.length < period || period < 1) return [];
  const out: TimeValuePoint[] = [];
  let sum = 0;
  for (let i = 0; i < candles.length; i++) {
    sum += getCandleSourceValue(candles[i], source);
    if (i >= period) sum -= getCandleSourceValue(candles[i - period], source);
    if (i >= period - 1) {
      out.push({ time: candles[i].time, value: sum / period });
    }
  }
  return out;
}

export function computeEMA(
  candles: Candle[],
  period = 21,
  source: PriceSource = 'close'
): TimeValuePoint[] {
  if (candles.length < period || period < 1) return [];
  const k = 2 / (period + 1);
  const out: TimeValuePoint[] = [];
  let sum = 0;
  for (let i = 0; i < period; i++) sum += getCandleSourceValue(candles[i], source);
  let ema = sum / period;
  out.push({ time: candles[period - 1].time, value: ema });
  for (let i = period; i < candles.length; i++) {
    const v = getCandleSourceValue(candles[i], source);
    ema = v * k + ema * (1 - k);
    out.push({ time: candles[i].time, value: ema });
  }
  return out;
}

export function computeBollingerBands(
  candles: Candle[],
  period = 20,
  stdDevMult = 2,
  source: PriceSource = 'close'
) {
  const upper: TimeValuePoint[] = [];
  const middle: TimeValuePoint[] = [];
  const lower: TimeValuePoint[] = [];
  if (candles.length < period) return { upper, middle, lower };

  for (let i = period - 1; i < candles.length; i++) {
    let sum = 0;
    for (let j = i - period + 1; j <= i; j++) sum += getCandleSourceValue(candles[j], source);
    const mean = sum / period;

    let sqDiff = 0;
    for (let j = i - period + 1; j <= i; j++) {
      const diff = getCandleSourceValue(candles[j], source) - mean;
      sqDiff += diff * diff;
    }
    const sd = Math.sqrt(sqDiff / period);
    const t = candles[i].time;
    middle.push({ time: t, value: mean });
    upper.push({ time: t, value: mean + stdDevMult * sd });
    lower.push({ time: t, value: mean - stdDevMult * sd });
  }

  return { upper, middle, lower };
}

export function computeVWAP(candles: Candle[]): TimeValuePoint[] {
  if (candles.length === 0) return [];
  const out: TimeValuePoint[] = [];
  let cumPV = 0;
  let cumVol = 0;
  let currentDay = -1;

  for (const c of candles) {
    const utcDay = Math.floor(c.time / 86400);
    if (utcDay !== currentDay) {
      cumPV = 0;
      cumVol = 0;
      currentDay = utcDay;
    }
    const typicalPrice = (c.high + c.low + c.close) / 3;
    const vol = Math.max(c.volume, 1);
    cumPV += typicalPrice * vol;
    cumVol += vol;
    out.push({ time: c.time, value: cumPV / cumVol });
  }
  return out;
}

export function computeATR(candles: Candle[], period = 14): TimeValuePoint[] {
  if (candles.length < period + 1) return [];
  const trs: number[] = [];
  for (let i = 1; i < candles.length; i++) {
    const c = candles[i];
    const prev = candles[i - 1];
    const tr = Math.max(
      c.high - c.low,
      Math.abs(c.high - prev.close),
      Math.abs(c.low - prev.close)
    );
    trs.push(tr);
  }

  const out: TimeValuePoint[] = [];
  let atr = trs.slice(0, period).reduce((a, b) => a + b, 0) / period;
  out.push({ time: candles[period].time, value: atr });

  for (let i = period; i < trs.length; i++) {
    atr = (atr * (period - 1) + trs[i]) / period;
    out.push({ time: candles[i + 1].time, value: atr });
  }
  return out;
}

export function computeSupertrend(
  candles: Candle[],
  period = 10,
  multiplier = 3,
  bullColor = '#089981',
  bearColor = '#F23645'
): ColoredPoint[] {
  const atrPoints = computeATR(candles, period);
  if (atrPoints.length === 0) return [];

  const atrMap = new Map<number, number>();
  for (const p of atrPoints) atrMap.set(p.time, p.value);

  const out: ColoredPoint[] = [];
  let prevFinalUpper = 0;
  let prevFinalLower = 0;
  let prevTrend: 1 | -1 = 1;
  let initialized = false;

  for (let i = 1; i < candles.length; i++) {
    const c = candles[i];
    const atr = atrMap.get(c.time);
    if (atr === undefined) continue;

    const hl2 = (c.high + c.low) / 2;
    const basicUpper = hl2 + multiplier * atr;
    const basicLower = hl2 - multiplier * atr;

    if (!initialized) {
      prevFinalUpper = basicUpper;
      prevFinalLower = basicLower;
      prevTrend = c.close >= hl2 ? 1 : -1;
      initialized = true;
      out.push({
        time: c.time,
        value: prevTrend === 1 ? prevFinalLower : prevFinalUpper,
        color: prevTrend === 1 ? bullColor : bearColor,
      });
      continue;
    }

    const prevClose = candles[i - 1].close;
    const finalUpper =
      basicUpper < prevFinalUpper || prevClose > prevFinalUpper ? basicUpper : prevFinalUpper;
    const finalLower =
      basicLower > prevFinalLower || prevClose < prevFinalLower ? basicLower : prevFinalLower;

    let trend: 1 | -1 = prevTrend;
    if (prevTrend === -1 && c.close > prevFinalUpper) trend = 1;
    else if (prevTrend === 1 && c.close < prevFinalLower) trend = -1;

    prevFinalUpper = finalUpper;
    prevFinalLower = finalLower;
    prevTrend = trend;

    out.push({
      time: c.time,
      value: trend === 1 ? finalLower : finalUpper,
      color: trend === 1 ? bullColor : bearColor,
    });
  }
  return out;
}

export function computeDonchian(candles: Candle[], period = 20) {
  const upper: TimeValuePoint[] = [];
  const middle: TimeValuePoint[] = [];
  const lower: TimeValuePoint[] = [];
  if (candles.length < period) return { upper, middle, lower };

  for (let i = period - 1; i < candles.length; i++) {
    let maxH = -Infinity;
    let minL = Infinity;
    for (let j = i - period + 1; j <= i; j++) {
      if (candles[j].high > maxH) maxH = candles[j].high;
      if (candles[j].low < minL) minL = candles[j].low;
    }
    const t = candles[i].time;
    upper.push({ time: t, value: maxH });
    lower.push({ time: t, value: minL });
    middle.push({ time: t, value: (maxH + minL) / 2 });
  }
  return { upper, middle, lower };
}

export function computeRSI(
  candles: Candle[],
  period = 14,
  source: PriceSource = 'close'
): TimeValuePoint[] {
  if (candles.length <= period) return [];
  const out: TimeValuePoint[] = [];

  let gainSum = 0;
  let lossSum = 0;
  for (let i = 1; i <= period; i++) {
    const diff =
      getCandleSourceValue(candles[i], source) - getCandleSourceValue(candles[i - 1], source);
    if (diff >= 0) gainSum += diff;
    else lossSum += Math.abs(diff);
  }

  let avgGain = gainSum / period;
  let avgLoss = lossSum / period;

  const calcRsi = (g: number, l: number) => (l === 0 ? 100 : 100 - 100 / (1 + g / l));
  out.push({ time: candles[period].time, value: calcRsi(avgGain, avgLoss) });

  for (let i = period + 1; i < candles.length; i++) {
    const diff =
      getCandleSourceValue(candles[i], source) - getCandleSourceValue(candles[i - 1], source);
    const gain = diff > 0 ? diff : 0;
    const loss = diff < 0 ? Math.abs(diff) : 0;
    avgGain = (avgGain * (period - 1) + gain) / period;
    avgLoss = (avgLoss * (period - 1) + loss) / period;
    out.push({ time: candles[i].time, value: calcRsi(avgGain, avgLoss) });
  }

  return out;
}

export function computeMACD(
  candles: Candle[],
  fast = 12,
  slow = 26,
  signalPeriod = 9,
  source: PriceSource = 'close'
) {
  const fastEma = computeEMA(candles, fast, source);
  const slowEma = computeEMA(candles, slow, source);
  if (fastEma.length === 0 || slowEma.length === 0) {
    return {
      macd: [] as TimeValuePoint[],
      signal: [] as TimeValuePoint[],
      histogram: [] as ColoredPoint[],
    };
  }

  const fastMap = new Map(fastEma.map((p) => [p.time, p.value]));
  const macdLine: TimeValuePoint[] = [];

  for (const s of slowEma) {
    const f = fastMap.get(s.time);
    if (f !== undefined) {
      macdLine.push({ time: s.time, value: f - s.value });
    }
  }

  if (macdLine.length < signalPeriod) {
    return { macd: macdLine, signal: [] as TimeValuePoint[], histogram: [] as ColoredPoint[] };
  }

  const signalLine: TimeValuePoint[] = [];
  const histogram: ColoredPoint[] = [];
  const k = 2 / (signalPeriod + 1);

  let sum = 0;
  for (let i = 0; i < signalPeriod; i++) sum += macdLine[i].value;
  let sig = sum / signalPeriod;

  for (let i = signalPeriod - 1; i < macdLine.length; i++) {
    if (i > signalPeriod - 1) {
      sig = macdLine[i].value * k + sig * (1 - k);
    }
    const histVal = macdLine[i].value - sig;
    signalLine.push({ time: macdLine[i].time, value: sig });
    histogram.push({
      time: macdLine[i].time,
      value: histVal,
      color: histVal >= 0 ? 'rgba(8, 153, 129, 0.72)' : 'rgba(242, 54, 69, 0.72)',
    });
  }

  return { macd: macdLine, signal: signalLine, histogram };
}

export function computeStochastic(
  candles: Candle[],
  kPeriod = 14,
  dPeriod = 3,
  smooth = 3
): { k: TimeValuePoint[]; d: TimeValuePoint[] } {
  if (candles.length < kPeriod + smooth + dPeriod) return { k: [], d: [] };

  const rawK: TimeValuePoint[] = [];
  for (let i = kPeriod - 1; i < candles.length; i++) {
    let highest = -Infinity;
    let lowest = Infinity;
    for (let j = i - kPeriod + 1; j <= i; j++) {
      if (candles[j].high > highest) highest = candles[j].high;
      if (candles[j].low < lowest) lowest = candles[j].low;
    }
    const range = highest - lowest;
    const val = range === 0 ? 50 : ((candles[i].close - lowest) / range) * 100;
    rawK.push({ time: candles[i].time, value: val });
  }

  const smoothArr = (arr: TimeValuePoint[], p: number): TimeValuePoint[] => {
    const out: TimeValuePoint[] = [];
    let sum = 0;
    for (let i = 0; i < arr.length; i++) {
      sum += arr[i].value;

    }
    return out;
  };

  const d = smoothArr(k, Math.max(1, dPeriod));

function createSeededRng(seedStr: string) {
  let h = 2166136261 >>> 0;
  for (let i = 0; i < seedStr.length; i++) {
    h ^= seedStr.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return () => {
    h += 0x6d2b79f5;
    let t = Math.imul(h ^ (h >>> 15), 1 | h);
    t ^= t + Math.imul(t ^ (t >>> 7), 61 | t);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function roundToPrecision(val: number, precision: number): number {
  const factor = Math.pow(10, precision);
  return Math.round(val * factor) / factor;
}

export function generateSyntheticCandles(
  symbolInfo: SymbolInfo,
  timeframe: TimeframeId,
  count: number,
  endAnchorTimeSec?: number,
  endAnchorPrice?: number
): Candle[] {
  const tfSec = TIMEFRAME_MAP[timeframe].seconds;
  const nowBucket = Math.floor((endAnchorTimeSec ?? Math.floor(Date.now() / 1000)) / tfSec) * tfSec;
  const rng = createSeededRng(`${symbolInfo.symbol}-${timeframe}`);

  const stepScale = Math.sqrt(tfSec / 86400);
  const stepVol = Math.max(symbolInfo.volatility * stepScale, symbolInfo.volatility * 0.045);

  const rawCloses: number[] = [];
  let price = endAnchorPrice ?? symbolInfo.basePrice;

  let trendRegime = 0;
  for (let i = 0; i < count; i++) {
    if (i % 24 === 0) {
      trendRegime = (rng() - 0.49) * stepVol * 0.45;
    }
    const u1 = Math.max(rng(), 1e-7);
    const u2 = rng();
    const gaussian = Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
    const shock = rng() < 0.035 ? (rng() - 0.5) * stepVol * 3.2 : 0;
    const ret = trendRegime + gaussian * stepVol + shock;
    price = Math.max(symbolInfo.minMove * 10, price * (1 + ret));
    rawCloses.push(price);
  }

  const finalGenerated = rawCloses[rawCloses.length - 1];
  const anchorFactor = endAnchorPrice ? endAnchorPrice / finalGenerated : 1;

  const candles: Candle[] = [];
  let prevClose = rawCloses[0] * anchorFactor;

  for (let i = 0; i < count; i++) {
    const t = nowBucket - (count - 1 - i) * tfSec;
    const close = roundToPrecision(rawCloses[i] * anchorFactor, symbolInfo.precision);
    const open = i === 0 ? close : prevClose;

    const bodySize = Math.abs(close - open);
    const upperWick = (rng() * 0.85 + 0.1) * Math.max(bodySize, open * stepVol * 0.65);
    const lowerWick = (rng() * 0.85 + 0.1) * Math.max(bodySize, open * stepVol * 0.65);

    const high = roundToPrecision(Math.max(open, close) + upperWick, symbolInfo.precision);
    const low = roundToPrecision(
      Math.max(symbolInfo.minMove, Math.min(open, close) - lowerWick),
      symbolInfo.precision
    );

    const baseVol =
      symbolInfo.category === 'Crypto' ? 850 : symbolInfo.category === 'IDX' ? 420000 : 18500;
    const volume = Math.round(
      baseVol * (0.4 + rng() * 1.9) * (1 + (bodySize / Math.max(open * stepVol, 1e-9)) * 0.5)
    );

    candles.push({ time: t, open, high, low, close, volume });
    prevClose = close;
  }

  return candles;
}

function aggregateCandles(source: Candle[], targetTfSec: number): Candle[] {
  if (source.length === 0) return [];
  const buckets = new Map<number, Candle>();

  for (const c of source) {
    const bucketTime = Math.floor(c.time / targetTfSec) * targetTfSec;
    const existing = buckets.get(bucketTime);
    if (!existing) {
      buckets.set(bucketTime, { ...c, time: bucketTime });
    } else {
      existing.high = Math.max(existing.high, c.high);
      existing.low = Math.min(existing.low, c.low);
      existing.close = c.close;
      existing.volume += c.volume;
    }
  }

  return Array.from(buckets.values()).sort((a, b) => a.time - b.time);
}

export function generateOrderBookSnapshot(
  symbol: string,
  midPrice: number,
  levels = 12
): { asks: OrderBookLevel[]; bids: OrderBookLevel[]; spread: number; spreadPct: number } {
  const info = getSymbolInfo(symbol);
  const step = Math.max(info.minMove, midPrice * 0.00025);

  const asksRaw: Omit<OrderBookLevel, 'pct'>[] = [];
  const bidsRaw: Omit<OrderBookLevel, 'pct'>[] = [];

  let askCum = 0;
  let bidCum = 0;

  for (let i = 1; i <= levels; i++) {
    const askPrice = roundToPrecision(
      midPrice + i * step * (0.8 + Math.random() * 0.4),
      info.precision
    );
    const bidPrice = roundToPrecision(
      Math.max(info.minMove, midPrice - i * step * (0.8 + Math.random() * 0.4)),
      info.precision
    );
    const baseSize = info.category === 'Crypto' ? 2.5 : info.category === 'IDX' ? 8500 : 450;
    const askSize = Number(
      (baseSize * (0.2 + Math.random() * 2.4)).toFixed(info.category === 'Crypto' ? 3 : 0)
    );
    const bidSize = Number(
      (baseSize * (0.2 + Math.random() * 2.4)).toFixed(info.category === 'Crypto' ? 3 : 0)
    );

    askCum += askSize;
    bidCum += bidSize;

    asksRaw.push({ price: askPrice, size: askSize, total: Number(askCum.toFixed(2)) });
    bidsRaw.push({ price: bidPrice, size: bidSize, total: Number(bidCum.toFixed(2)) });
  }

  const maxTotal = Math.max(askCum, bidCum, 1);
  const asks = asksRaw
    .map((a) => ({ ...a, pct: Math.min(100, (a.total / maxTotal) * 100) }))
    .reverse();
  const bids = bidsRaw.map((b) => ({ ...b, pct: Math.min(100, (b.total / maxTotal) * 100) }));

  const bestAsk = asks[asks.length - 1]?.price ?? midPrice;
  const bestBid = bids[0]?.price ?? midPrice;
  const spread = Math.max(0, bestAsk - bestBid);
  const spreadPct = midPrice > 0 ? (spread / midPrice) * 100 : 0;

  return { asks, bids, spread, spreadPct };
}

function createIndicatorInstance(
  type: IndicatorType,
  overrideParams?: Record<string, number>,
  overrideColor?: string
): IndicatorConfig {
  const meta = INDICATOR_CATALOG.find((i) => i.type === type) ?? INDICATOR_CATALOG[0];
  return {
    id: generateId('ind'),
    type: meta.type,
    name: meta.shortName,
    placement: meta.placement,
    visible: true,
    color: overrideColor ?? meta.defaultColor,
    secondaryColor: meta.secondaryColor,
    tertiaryColor: meta.tertiaryColor,
    lineWidth: 2,
    lineStyle: 'solid',
    source: 'close',
    offset: 0,
    upperLevel: meta.defaultUpperLevel,
    lowerLevel: meta.defaultLowerLevel,
    showBands: true,
    showLastValue: true,
    visibility: { seconds: true, minutes: true, hours: true, days: true },
    params: { ...meta.defaultParams, ...(overrideParams ?? {}) },
  };
}

              [drawing.symbol]: [...list, drawing],

      updateDrawing: (symbol, id, patch) =>
        set((state) => {
          const list = state.drawingsBySymbol[symbol] ?? [];
          return {
            drawingsBySymbol: {
              ...state.drawingsBySymbol,
              [symbol]: list.map((d) => (d.id === id ? { ...d, ...patch } : d)),
            },
          };
        }),

      duplicateDrawing: (symbol, id) =>
        set((state) => {
          const list = state.drawingsBySymbol[symbol] ?? [];
          const src = list.find((d) => d.id === id);
          if (!src) return state;
          const clone: DrawingObject = {
            ...src,
            id: generateId('drw'),
            locked: false,
            points: src.points.map((pt) => ({
              time: pt.time,
              price: pt.price * 1.003,
            })),
            createdAt: Date.now(),
          };
          return {
            drawingsBySymbol: {
              ...state.drawingsBySymbol,
              [symbol]: [...list, clone],
            },
            selectedDrawingId: clone.id,
          };
        }),

      removeDrawing: (symbol, id) =>
        set((state) => {
          const list = state.drawingsBySymbol[symbol] ?? [];
          const removed = list.find((d) => d.id === id);
          const redo = state.redoBySymbol[symbol] ?? [];
          return {
            drawingsBySymbol: {
              ...state.drawingsBySymbol,
              [symbol]: list.filter((d) => d.id !== id),
            },
            redoBySymbol: removed
              ? { ...state.redoBySymbol, [symbol]: [removed, ...redo].slice(0, 25) }
              : state.redoBySymbol,
            selectedDrawingId: state.selectedDrawingId === id ? null : state.selectedDrawingId,
          };
        }),

      clearDrawingsForSymbol: (symbol) =>
        set((state) => ({
          drawingsBySymbol: { ...state.drawingsBySymbol, [symbol]: [] },
          redoBySymbol: { ...state.redoBySymbol, [symbol]: [] },
          selectedDrawingId: null,
        })),

      undoDrawing: (symbol) =>
        set((state) => {
          const list = state.drawingsBySymbol[symbol] ?? [];
          if (list.length === 0) return state;
          const last = list[list.length - 1];
          const redo = state.redoBySymbol[symbol] ?? [];
          return {
            drawingsBySymbol: {
              ...state.drawingsBySymbol,
              [symbol]: list.slice(0, -1),
            },
            redoBySymbol: {
              ...state.redoBySymbol,
              [symbol]: [last, ...redo].slice(0, 25),
            },
            selectedDrawingId: null,
          };
        }),

      redoDrawing: (symbol) =>
        set((state) => {
          const redo = state.redoBySymbol[symbol] ?? [];
          if (redo.length === 0) return state;
          const [restored, ...rest] = redo;
          const list = state.drawingsBySymbol[symbol] ?? [];
          return {
            drawingsBySymbol: {
              ...state.drawingsBySymbol,
              [symbol]: [...list, restored],
            },
            redoBySymbol: {
              ...state.redoBySymbol,
              [symbol]: rest,
            },
            selectedDrawingId: restored.id,
          };
        }),

      replay: DEFAULT_REPLAY,
      enterReplayMode: (paneId) =>
        set({
          replay: {
            enabled: true,
            isSelectingStart: true,
            isPlaying: false,
            speed: 1,
            currentIndex: 0,
            totalBars: 0,
            currentTimestamp: null,
            paneId,
          },
        }),

      startReplayAt: (paneId, index, totalBars, timestamp) =>
        set((state) => ({
          replay: {
            ...state.replay,
            enabled: true,
            isSelectingStart: false,
            isPlaying: false,
            paneId,
            totalBars,
            currentIndex: Math.max(5, Math.min(index, Math.max(5, totalBars - 2))),
            currentTimestamp: timestamp,
          },
        })),

      setReplaySelectMode: (isSelectingStart) =>
        set((state) => ({
          replay: { ...state.replay, isSelectingStart, isPlaying: false },
        })),

      toggleReplayPlay: () =>
        set((state) => {
          if (!state.replay.enabled) return state;
          if (state.replay.isSelectingStart) {
            const fallbackIdx = Math.max(10, Math.floor(state.replay.totalBars * 0.6));
            return {
              replay: {
                ...state.replay,
                isSelectingStart: false,
                currentIndex: state.replay.currentIndex || fallbackIdx,
                isPlaying: true,
              },
            };
          }
          if (state.replay.currentIndex >= state.replay.totalBars - 1) {
            return { replay: { ...state.replay, isPlaying: false } };
          }
          return {
            replay: { ...state.replay, isPlaying: !state.replay.isPlaying },
          };
        }),

      stepReplayForward: () =>
        set((state) => {
          if (!state.replay.enabled || state.replay.totalBars === 0) return state;
          const next = Math.min(state.replay.totalBars - 1, state.replay.currentIndex + 1);
          return {
            replay: {
              ...state.replay,
              isSelectingStart: false,
              currentIndex: next,
              isPlaying: next >= state.replay.totalBars - 1 ? false : state.replay.isPlaying,
            },
          };
        }),

      stepReplayBackward: () =>
        set((state) => {
          if (!state.replay.enabled || state.replay.totalBars === 0) return state;
          const prev = Math.max(5, state.replay.currentIndex - 1);
          return {
            replay: { ...state.replay, isSelectingStart: false, isPlaying: false, currentIndex: prev },
          };
        }),

      setReplaySpeed: (speed) =>
        set((state) => ({ replay: { ...state.replay, speed } })),

      setReplayIndex: (index, timestamp) =>
        set((state) => ({
          replay: {
            ...state.replay,
            currentIndex: Math.max(5, Math.min(index, Math.max(5, state.replay.totalBars - 1))),
            currentTimestamp: timestamp !== undefined ? timestamp : state.replay.currentTimestamp,
          },
        })),

      setReplayTotalBars: (totalBars) =>
        set((state) => ({
          replay: {
            ...state.replay,
            totalBars,
            currentIndex:
              state.replay.currentIndex > 0
                ? Math.min(state.replay.currentIndex, Math.max(5, totalBars - 1))
                : Math.max(10, Math.floor(totalBars * 0.6)),
          },
        })),

      exitReplayMode: () => set({ replay: DEFAULT_REPLAY }),

      watchlist: [
        'BTCUSDT',
        'ETHUSDT',
        'SOLUSDT',
        'XAUUSD',
        'EURUSD',
        'GBPUSD',
        'USDJPY',
        'NVDA',
        'TSLA',
        'BBCA.JK',
        'BBRI.JK',
        'TLKM.JK',
      ],

      toggleWatchlistSymbol: (symbol) =>
        set((state) => {
          const exists = state.watchlist.includes(symbol);
          return {
            watchlist: exists
              ? state.watchlist.filter((s) => s !== symbol)
              : [...state.watchlist, symbol],
          };
        }),

      quotes: {},
      updateQuote: (symbol, price, source = 'simulated', volume = 0) =>
        set((state) => {
          const prev = state.quotes[symbol];
          const info = getSymbolInfo(symbol);
          const prevClose = prev?.prevClose ?? info.basePrice;
          const change = price - prevClose;
          const changePercent = prevClose > 0 ? (change / prevClose) * 100 : 0;
          const spark = prev?.sparkline ? [...prev.sparkline, price].slice(-24) : [price];

          return {
            quotes: {
              ...state.quotes,
              [symbol]: {
                symbol,
                price,
                prevClose,
                change,
                changePercent,
                high24h: prev ? Math.max(prev.high24h, price) : price * 1.012,
                low24h: prev ? Math.min(prev.low24h, price) : price * 0.988,
                volume24h: (prev?.volume24h ?? info.basePrice * 120) + volume,
                sparkline: spark,
                source,
              },
            },
          };
        }),

      alerts: [],
      addAlert: (alert) =>
        set((state) => ({
          alerts: [
            {
              ...alert,
              id: generateId('alert'),
              active: true,
              createdAt: Date.now(),
            },
            ...state.alerts,
          ],
        })),

      toggleAlert: (id) =>
        set((state) => ({
          alerts: state.alerts.map((a) =>
            a.id === id ? { ...a, active: !a.active, triggeredAt: undefined } : a
          ),
        })),

      removeAlert: (id) =>
        set((state) => ({
          alerts: state.alerts.filter((a) => a.id !== id),
        })),

      markAlertTriggered: (id) =>
        set((state) => ({
          alerts: state.alerts.map((a) =>
            a.id === id ? { ...a, active: false, triggeredAt: Date.now() } : a
          ),
        })),

      paperBalance: 100000,
      positions: [],
      pendingOrders: [],
      closedTrades: [],

      placeMarketOrder: ({ symbol, side, quantity, price, stopLoss, takeProfit }) => {
        if (quantity <= 0 || price <= 0) return;
        const pos: PaperPosition = {
          id: generateId('pos'),
          symbol,
          side,
          quantity,
          entryPrice: price,
          stopLoss: stopLoss && stopLoss > 0 ? stopLoss : undefined,
          takeProfit: takeProfit && takeProfit > 0 ? takeProfit : undefined,
          openedAt: Date.now(),
        };
        set((state) => ({
          positions: [pos, ...state.positions],
        }));
        get().addToast({
          title: `${side === 'buy' ? 'Long' : 'Short'} Executed: ${symbol}`,
          description: `Qty ${quantity} @ ${price}`,
          variant: 'success',
        });
      },

      placeLimitOrder: ({ symbol, side, quantity, limitPrice, stopLoss, takeProfit }) => {
        if (quantity <= 0 || limitPrice <= 0) return;
        const order: PendingOrder = {
          id: generateId('ord'),
          symbol,
          side,
          type: 'limit',
          quantity,
          limitPrice,
          stopLoss: stopLoss && stopLoss > 0 ? stopLoss : undefined,
          takeProfit: takeProfit && takeProfit > 0 ? takeProfit : undefined,
          createdAt: Date.now(),
        };
        set((state) => ({
          pendingOrders: [order, ...state.pendingOrders],
        }));
        get().addToast({
          title: `Limit ${side.toUpperCase()} Placed: ${symbol}`,
          description: `Qty ${quantity} @ ${limitPrice}`,
          variant: 'info',
        });
      },

      cancelPendingOrder: (id) =>
        set((state) => ({
          pendingOrders: state.pendingOrders.filter((o) => o.id !== id),
        })),

      closePosition: (id, exitPrice, reason = 'manual') => {
        const state = get();
        const pos = state.positions.find((p) => p.id === id);
        if (!pos) return;

        const priceDiff = pos.side === 'buy' ? exitPrice - pos.entryPrice : pos.entryPrice - exitPrice;
        const pnl = priceDiff * pos.quantity;
        const notional = pos.entryPrice * pos.quantity;
        const pnlPercent = notional > 0 ? (pnl / notional) * 100 : 0;

        const closed: ClosedTrade = {
          id: generateId('trd'),
          symbol: pos.symbol,
          side: pos.side,
          quantity: pos.quantity,
          entryPrice: pos.entryPrice,
          exitPrice,
          pnl,
          pnlPercent,
          openedAt: pos.openedAt,
          closedAt: Date.now(),
          reason,
        };

        set({
          positions: state.positions.filter((p) => p.id !== id),
          closedTrades: [closed, ...state.closedTrades].slice(0, 100),
          paperBalance: state.paperBalance + pnl,
        });

        get().addToast({
          title: `${reason === 'sl' ? 'Stop Loss Hit' : reason === 'tp' ? 'Take Profit Hit' : 'Position Closed'}: ${pos.symbol}`,
          description: `PnL: ${pnl >= 0 ? '+' : ''}$${pnl.toFixed(2)} (${pnlPercent.toFixed(2)}%)`,
          variant: pnl >= 0 ? 'success' : 'danger',
        });
      },

      updatePositionTPSL: (id, stopLoss, takeProfit) =>
        set((state) => ({
          positions: state.positions.map((p) =>
            p.id === id ? { ...p, stopLoss, takeProfit } : p
          ),
        })),

      processMarketPriceForPaperTrading: (symbol, high, low, close) => {
        const state = get();
        const ordersForSymbol = state.pendingOrders.filter((o) => o.symbol === symbol);
        for (const ord of ordersForSymbol) {
          const hit = ord.side === 'buy' ? low <= ord.limitPrice : high >= ord.limitPrice;
          if (hit) {
            set((s) => ({
              pendingOrders: s.pendingOrders.filter((o) => o.id !== ord.id),
              positions: [
                {
                  id: generateId('pos'),
                  symbol: ord.symbol,
                  side: ord.side,
                  quantity: ord.quantity,
                  entryPrice: ord.limitPrice,
                  stopLoss: ord.stopLoss,
                  takeProfit: ord.takeProfit,
                  openedAt: Date.now(),
                },
                ...s.positions,
              ],
            }));
            get().addToast({
              title: `Limit Order Filled: ${ord.symbol}`,
              description: `${ord.side.toUpperCase()} ${ord.quantity} @ ${ord.limitPrice}`,
              variant: 'success',
            });
          }
        }

        const currentPositions = get().positions.filter((p) => p.symbol === symbol);
        for (const pos of currentPositions) {
          if (pos.side === 'buy') {
            if (pos.stopLoss && low <= pos.stopLoss) {
              get().closePosition(pos.id, pos.stopLoss, 'sl');
            } else if (pos.takeProfit && high >= pos.takeProfit) {
              get().closePosition(pos.id, pos.takeProfit, 'tp');
            }
          } else {
            if (pos.stopLoss && high >= pos.stopLoss) {
              get().closePosition(pos.id, pos.stopLoss, 'sl');
            } else if (pos.takeProfit && low <= pos.takeProfit) {
              get().closePosition(pos.id, pos.takeProfit, 'tp');
            }
          }
        }
      },

      resetPaperAccount: () =>
        set({
          paperBalance: 100000,
          positions: [],
          pendingOrders: [],
          closedTrades: [],
        }),

      toasts: [],
      addToast: (toast) => {
        const id = generateId('toast');
        set((state) => ({
          toasts: [{ ...toast, id, createdAt: Date.now() }, ...state.toasts].slice(0, 5),
        }));
        window.setTimeout(() => {
          get().removeToast(id);
        }, 4500);
      },

      removeToast: (id) =>
        set((state) => ({
          toasts: state.toasts.filter((t) => t.id !== id),
        })),
    }),
    {
      name: 'apexchart-pro-storage-v4',
      partialize: (state) => ({
        theme: state.theme,
        chartSettings: state.chartSettings,
        smartOverlays: state.smartOverlays,
        dataFeed: state.dataFeed,
        layout: state.layout,
        panes: state.panes,
        activePaneId: state.activePaneId,
        favoriteTimeframes: state.favoriteTimeframes,
        syncCrosshair: state.syncCrosshair,
        syncSymbol: state.syncSymbol,
        syncTimeframe: state.syncTimeframe,
        watchlist: state.watchlist,
        drawingsBySymbol: state.drawingsBySymbol,
        groupDefaultTools: state.groupDefaultTools,
        alerts: state.alerts,
        paperBalance: state.paperBalance,
        positions: state.positions,
        pendingOrders: state.pendingOrders,
        closedTrades: state.closedTrades,
      }),
    }
  )
);
