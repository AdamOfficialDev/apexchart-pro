export function isVisibleOnTimeframeGroup(
  vis: TimeframeVisibility | undefined,
  group: TimeframeGroup
): boolean {
  if (!vis) return true;
  if (group === 'Seconds') return vis.seconds;
  if (group === 'Minutes') return vis.minutes;
  if (group === 'Hours') return vis.hours;
  return vis.days;
}

// Dynamically supports ANY custom ticker typed by the user!
export function getSymbolInfo(rawSymbol: string): SymbolInfo {
  const clean = rawSymbol.trim().toUpperCase().replace(/\s+/g, '');
  const found = SYMBOLS.find(
    (s) => s.symbol.toUpperCase() === clean || s.displaySymbol.toUpperCase() === clean
  );
  if (found) return found;

  // Crypto pairs (e.g. PEPEUSDT, SUIUSDT, LINKUSDT, BTCUSD)
  if (clean.endsWith('USDT') || clean.endsWith('USDC') || clean.endsWith('FDUSD')) {
    const base = clean.replace(/(USDT|USDC|FDUSD)$/, '');
    const isMicro = base === 'PEPE' || base === 'SHIB' || base === 'BONK' || base === 'FLOKI';
    return {
      symbol: clean,
      displaySymbol: `${base}/USDT`,
      name: `${base} / TetherUS (Custom)`,
      category: 'Crypto',
      exchange: 'BINANCE',
      basePrice: isMicro ? 0.000012 : 12.5,
      precision: isMicro ? 8 : 4,
      minMove: isMicro ? 0.00000001 : 0.0001,
      volatility: 0.045,
      binanceSymbol: clean,
      twelveDataSymbol: `${base}/USD`,
      currency: 'USDT',
      description: `Live Binance spot pair ${base}/USDT.`,
    };
  }

  // Indonesian IDX Equities (e.g. ADRO.JK, ASII.JK)
  if (clean.endsWith('.JK')) {
    const code = clean.replace('.JK', '');
    return {
      symbol: clean,
      displaySymbol: code,
      name: `${code} Tbk (IDX)`,
      category: 'IDX',
      exchange: 'IDX',
      basePrice: 2500,
      precision: 0,
      minMove: 10,
      volatility: 0.02,
      twelveDataSymbol: `${code}:IDX`,
      currency: 'IDR',
      description: `Indonesian Equity ${code} on IDX.`,
    };
  }

  // 6-letter FX pairs (e.g. AUDUSD, USDCAD, EURJPY)
  if (/^[A-Z]{6}$/.test(clean)) {
    const b = clean.slice(0, 3);
    const q = clean.slice(3, 6);
    const isJpy = q === 'JPY';
    return {
      symbol: clean,
      displaySymbol: `${b}/${q}`,
      name: `${b} / ${q} Spot FX`,
      category: 'Forex',
      exchange: 'FX',
      basePrice: isJpy ? 150 : 1.1,
      precision: isJpy ? 3 : 5,
      minMove: isJpy ? 0.001 : 0.00001,
      volatility: 0.006,
      twelveDataSymbol: `${b}/${q}`,
      fxPair: [b, q],
      currency: q,
      description: `Foreign exchange spot pair ${b}/${q}.`,
    };
  }

  // Default US Stock / Custom Ticker
  return {
    symbol: clean,
    displaySymbol: clean,
    name: `${clean} (Custom Instrument)`,
    category: 'Stocks',
    exchange: 'CUSTOM',
    basePrice: 150,
    precision: 2,
    minMove: 0.01,
    volatility: 0.022,
    twelveDataSymbol: clean,
    currency: 'USD',
    description: `Custom instrument ${clean}.`,
  };
}

import { computeRSI } from '@/services/indicators';
import { Candle, RSIDivergenceConfig, RSIDivergenceSignal } from '@/types/trading';

interface PivotPoint {
  index: number;
  time: number;
  price: number;
  rsi: number;
  kind: 'high' | 'low';
}

export function computeRSIDivergences(
  candles: Candle[],
  config: RSIDivergenceConfig
): RSIDivergenceSignal[] {
  if (candles.length < config.rsiPeriod + 20) return [];

  const rsiPoints = computeRSI(candles, config.rsiPeriod, 'close');
  const rsiByTime = new Map<number, number>();
  for (const p of rsiPoints) rsiByTime.set(p.time, p.value);

  const rsiArr: (number | null)[] = candles.map((c) => rsiByTime.get(c.time) ?? null);
  const left = Math.max(2, config.pivotLeft);
  const right = Math.max(2, config.pivotRight);

  const pivotHighs: PivotPoint[] = [];
  const pivotLows: PivotPoint[] = [];

  for (let i = left; i < candles.length - right; i++) {
    const r = rsiArr[i];
    if (r === null) continue;

    let isHigh = true;
    let isLow = true;
    for (let j = i - left; j <= i + right; j++) {
      if (j === i) continue;
      if (candles[j].high >= candles[i].high) isHigh = false;
      if (candles[j].low <= candles[i].low) isLow = false;
    }

    if (isHigh) {
      pivotHighs.push({
        index: i,
        time: candles[i].time,
        price: candles[i].high,
        rsi: r,
        kind: 'high',
      });
    }
    if (isLow) {
      pivotLows.push({
        index: i,
        time: candles[i].time,
        price: candles[i].low,
        rsi: r,
        kind: 'low',
      });
    }
  }

  const signals: RSIDivergenceSignal[] = [];
  const tolFrac = Math.max(0.0005, config.doubleTopTolerancePct / 100);

  // Helper: evaluate trendline + candle body break for Regular & Exaggerated
  const buildTrendlineBreakSetup = (
    direction: 'bull' | 'bear',
    p1: PivotPoint,
    p2: PivotPoint
  ) => {
    // Find structural counter-swing between p1 and p2
    let midIdx = p1.index + 1;
    for (let k = p1.index + 1; k < p2.index; k++) {
      if (direction === 'bull') {
        if (candles[k].high > candles[midIdx].high) midIdx = k;
      } else {
        if (candles[k].low < candles[midIdx].low) midIdx = k;
      }
    }

    // Find secondary shoulder candle between midIdx and p2.index - 1 to anchor sloped counter-trendline
    let anchor2Idx = Math.max(midIdx, p2.index - 1);
    let bestShoulderDist = -Infinity;
    for (let k = midIdx + 1; k < p2.index; k++) {
      if (direction === 'bull') {
        if (candles[k].high < candles[midIdx].high && candles[k].high > bestShoulderDist) {
          bestShoulderDist = candles[k].high;
          anchor2Idx = k;
        }
      } else {
        if (
          candles[k].low > candles[midIdx].low &&
          (bestShoulderDist === -Infinity || candles[k].low < bestShoulderDist)
        ) {
          bestShoulderDist = candles[k].low;
          anchor2Idx = k;
        }
      }
    }

    const idxA = midIdx;
    const valA = direction === 'bull' ? candles[idxA].high : candles[idxA].low;
    const idxB = anchor2Idx > idxA ? anchor2Idx : p2.index - 1;
    const valB =
      idxB > idxA
        ? direction === 'bull'
          ? candles[idxB].high
          : candles[idxB].low
        : valA;

    const spanIdx = Math.max(1, idxB - idxA);
    const slope = (valB - valA) / spanIdx;

    // Scan after p2 for genuine candle BODY close breaking the counter-trendline
    let valid = false;
    let entryIdx = Math.min(candles.length - 1, p2.index + 1);
    let entryPrice = candles[entryIdx].close;
    let endPlotIdx = Math.min(candles.length - 1, p2.index + 16);

    for (let k = p2.index + 1; k <= Math.min(candles.length - 1, p2.index + 24); k++) {
      const tlAtK = valA + slope * (k - idxA);
      const c = candles[k];
      if (direction === 'bull') {
        // Candle body closes strictly above trigger trendline (or bullish candle breaks previous body high)
        if (c.close > tlAtK && c.close > c.open) {
          valid = true;
          entryIdx = k;
          entryPrice = c.close;
          endPlotIdx = Math.max(k + 2, idxB + 2);

            pos.side === 'buy'

            exitTime: c.time,
            exitPrice: c.close,
            pnlUsd,
            pnlPct: ret * 100,
            reason: 'Flip Signal',
          });
        }
        pos = { side: signal, entryTime: c.time, entryPrice: c.close };
      }
    }
  }

  const wins = trades.filter((t) => t.pnlUsd > 0);
  const losses = trades.filter((t) => t.pnlUsd <= 0);

  const grossLoss = Math.abs(losses.reduce((s, t) => s + t.pnlUsd, 0));

    strategyId,
    strategyName: STRATEGY_NAMES[strategyId],
    initialCapital,
    finalEquity: equity,
    netProfitUsd: equity - initialCapital,
    netProfitPct: ((equity - initialCapital) / initialCapital) * 100,
    winRatePct: trades.length > 0 ? (wins.length / trades.length) * 100 : 0,
    profitFactor: grossLoss > 0 ? grossWin / grossLoss : grossWin > 0 ? 99 : 0,
    maxDrawdownPct: maxDD,
    totalTrades: trades.length,
    equityCurve,
    trades,
  };
}

function createSeededRng(seedStr: string) {
  let h = 2166136261 >>> 0;
  for (let i = 0; i < seedStr.length; i++) {
    h ^= seedStr.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return () => {
    h += 0x6d2b79f5;
    let t = Math.imul(h ^ (h >>> 15), 1 | h);
    t ^= t + Math.imul(t ^ (t >>> 7), 61 | t);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function roundToPrecision(val: number, precision: number): number {
  const factor = Math.pow(10, precision);
  return Math.round(val * factor) / factor;
}

async function fetchLiveFxSpotAnchor(info: SymbolInfo): Promise<number | null> {
  if (!info.fxPair) return null;
  const now = Date.now();
  try {
    if (!FX_RATE_CACHE.rates || !FX_RATE_CACHE.fetchedAt || now - FX_RATE_CACHE.fetchedAt > 300000) {
      const controller = new AbortController();
      const timer = window.setTimeout(() => controller.abort(), 3000);
      try {
        const res = await fetch('https://open.er-api.com/v6/latest/USD', {
          signal: controller.signal,
        });
        if (res.ok) {
          const json = (await res.json()) as { rates?: Record<string, number> };
          if (json.rates) {
            FX_RATE_CACHE.rates = json.rates;
            FX_RATE_CACHE.fetchedAt = now;
          }
        }
      } finally {
        window.clearTimeout(timer);
      }
    }

    const rates = FX_RATE_CACHE.rates;
    if (!rates) return null;
    const [base, quote] = info.fxPair;
    if (base === 'USD' && rates[quote]) return rates[quote];
    if (quote === 'USD' && rates[base]) return 1 / rates[base];
    if (rates[base] && rates[quote]) return rates[quote] / rates[base];
    return null;
  } catch {
    return null;
  }
}

export function generateSyntheticCandles(
  symbolInfo: SymbolInfo,
  timeframe: TimeframeId,
  count: number,
  endAnchorTimeSec?: number,
  endAnchorPrice?: number
): Candle[] {
  const tfSec = TIMEFRAME_MAP[timeframe].seconds;
  const nowBucket = Math.floor((endAnchorTimeSec ?? Math.floor(Date.now() / 1000)) / tfSec) * tfSec;
  const rng = createSeededRng(`${symbolInfo.symbol}-${timeframe}`);

  const stepScale = Math.sqrt(tfSec / 86400);
  const stepVol = Math.max(symbolInfo.volatility * stepScale, symbolInfo.volatility * 0.045);

  const rawCloses: number[] = [];
  let price = endAnchorPrice ?? symbolInfo.basePrice;

  let trendRegime = 0;
  for (let i = 0; i < count; i++) {
    if (i % 24 === 0) {
      trendRegime = (rng() - 0.49) * stepVol * 0.45;
    }
    const u1 = Math.max(rng(), 1e-7);
    const u2 = rng();
    const gaussian = Math.sqrt(-2 * Math.log(u1)) * Math.cos(2 * Math.PI * u2);
    const shock = rng() < 0.035 ? (rng() - 0.5) * stepVol * 3.2 : 0;
    const ret = trendRegime + gaussian * stepVol + shock;
    price = Math.max(symbolInfo.minMove * 10, price * (1 + ret));
    rawCloses.push(price);
  }

  const finalGenerated = rawCloses[rawCloses.length - 1];
  const anchorFactor = endAnchorPrice ? endAnchorPrice / finalGenerated : 1;

  const candles: Candle[] = [];
  let prevClose = rawCloses[0] * anchorFactor;

  for (let i = 0; i < count; i++) {
    const t = nowBucket - (count - 1 - i) * tfSec;
    const close = roundToPrecision(rawCloses[i] * anchorFactor, symbolInfo.precision);
    const open = i === 0 ? close : prevClose;

    const bodySize = Math.abs(close - open);
    const upperWick = (rng() * 0.85 + 0.1) * Math.max(bodySize, open * stepVol * 0.65);
    const lowerWick = (rng() * 0.85 + 0.1) * Math.max(bodySize, open * stepVol * 0.65);

    const high = roundToPrecision(Math.max(open, close) + upperWick, symbolInfo.precision);
    const low = roundToPrecision(
      Math.max(symbolInfo.minMove, Math.min(open, close) - lowerWick),
      symbolInfo.precision
    );

    const baseVol =
      symbolInfo.category === 'Crypto' ? 850 : symbolInfo.category === 'IDX' ? 420000 : 18500;
    const volume = Math.round(
      baseVol * (0.4 + rng() * 1.9) * (1 + (bodySize / Math.max(open * stepVol, 1e-9)) * 0.5)
    );

    candles.push({ time: t, open, high, low, close, volume });
    prevClose = close;
  }

  return candles;
}

function aggregateCandles(source: Candle[], targetTfSec: number): Candle[] {
  if (source.length === 0) return [];
  const buckets = new Map<number, Candle>();

  for (const c of source) {
    const bucketTime = Math.floor(c.time / targetTfSec) * targetTfSec;
    const existing = buckets.get(bucketTime);
    if (!existing) {
      buckets.set(bucketTime, { ...c, time: bucketTime });
    } else {
      existing.high = Math.max(existing.high, c.high);
      existing.low = Math.min(existing.low, c.low);
      existing.close = c.close;
      existing.volume += c.volume;
    }
  }

  return Array.from(buckets.values()).sort((a, b) => a.time - b.time);
}

async function fetchBinanceKlines(
  binanceSymbol: string,
  interval: string,
  limit: number
): Promise<Candle[]> {
  const controller = new AbortController();
  const timer = window.setTimeout(() => controller.abort(), 3800);

  try {
    const url = `${BINANCE_REST_BASE}/klines?symbol=${binanceSymbol}&interval=${interval}&limit=${limit}`;
    const res = await fetch(url, { signal: controller.signal });
    if (!res.ok) throw new Error(`HTTP ${res.status}`);
    const data = (await res.json()) as unknown[][];
    if (!Array.isArray(data) || data.length === 0) throw new Error('Empty klines');

    const map = new Map<number, Candle>();
    for (const row of data) {
      const time = Math.floor(Number(row[0]) / 1000);
      map.set(time, {
        time,
        open: Number(row[1]),
        high: Number(row[2]),
        low: Number(row[3]),
        close: Number(row[4]),
        volume: Number(row[5]),
      });
    }
    return Array.from(map.values()).sort((a, b) => a.time - b.time);
  } finally {
    window.clearTimeout(timer);
  }
}

function parseTwelveDatetimeToUnixSec(dtStr: string): number {
  const iso = dtStr.includes(' ') ? dtStr.replace(' ', 'T') + 'Z' : dtStr + 'T00:00:00Z';
  const ms = Date.parse(iso);
  return Number.isFinite(ms) ? Math.floor(ms / 1000) : Math.floor(Date.now() / 1000);
}

async function fetchTwelveDataKlines(
  tdSymbol: string,
  interval: string,
  apiKey: string,
  outputsize = 380
): Promise<Candle[]> {
  const controller = new AbortController();
  const timer = window.setTimeout(() => controller.abort(), 5000);
  try {
    const url = `${TWELVE_REST_BASE}/time_series?symbol=${encodeURIComponent(
      tdSymbol
    )}&interval=${encodeURIComponent(interval)}&outputsize=${outputsize}&apikey=${encodeURIComponent(
      apiKey
    )}`;
    const res = await fetch(url, { signal: controller.signal });
    if (!res.ok) throw new Error(`Twelve Data HTTP ${res.status}`);
    const json = (await res.json()) as {
      status?: string;
      message?: string;
      values?: Array<{
        datetime: string;
        open: string;
        high: string;
        low: string;
        close: string;
        volume?: string;
      }>;
    };
    if (json.status === 'error' || !Array.isArray(json.values) || json.values.length === 0) {
      throw new Error(json.message || 'Twelve Data returned empty values');
    }

    const map = new Map<number, Candle>();
    for (const row of json.values) {
      const time = parseTwelveDatetimeToUnixSec(row.datetime);
      map.set(time, {
        time,
        open: Number(row.open),
        high: Number(row.high),
        low: Number(row.low),
        close: Number(row.close),
        volume: Number(row.volume ?? 1000) || 1000,
      });
    }
    return Array.from(map.values()).sort((a, b) => a.time - b.time);
  } finally {
    window.clearTimeout(timer);
  }
}

export async function fetchHistoricalCandles(
  symbol: string,
  timeframe: TimeframeId,
  feedConfig?: DataFeedConfig
): Promise<{ candles: Candle[]; source: FeedSourceType; warning?: string }> {
  const info = getSymbolInfo(symbol);
  const tfMeta = TIMEFRAME_MAP[timeframe];
  const tdKey = feedConfig?.twelveDataApiKey?.trim() || '';
  let tdWarning: string | undefined;

  const shouldTryTwelveData =
    Boolean(tdKey) &&
    Boolean(info.twelveDataSymbol) &&
    (feedConfig?.preferTwelveData || !info.binanceSymbol || info.symbol === 'XAUUSD');

  if (shouldTryTwelveData && info.twelveDataSymbol) {
    try {
      const interval = tfMeta.twelveDataInterval ?? '1min';
      const raw = await fetchTwelveDataKlines(
        info.twelveDataSymbol,
        interval,
        tdKey,
        Math.min(500, tfMeta.defaultBars)
      );
      if (raw.length > 0) {
        if (tfMeta.seconds < 60) {
          const lastBar = raw[raw.length - 1];
          const synth = generateSyntheticCandles(
            info,
            timeframe,
            tfMeta.defaultBars,
            Math.floor(Date.now() / 1000),
            lastBar.close
          );
          return { candles: synth, source: 'twelvedata' };
        }
        if (timeframe === '3m') {
          const agg = aggregateCandles(raw, 180);
          if (agg.length > 0) return { candles: agg, source: 'twelvedata' };
        }
        return { candles: raw, source: 'twelvedata' };
      }
    } catch (err) {
      tdWarning = (err as Error).message;
    }
  }

  const allowBinance =
    Boolean(info.binanceSymbol) &&
    (info.symbol !== 'XAUUSD' || feedConfig?.useGoldProxyWhenNoKey !== false);

  if (allowBinance && info.binanceSymbol) {
    const resolvedSource: FeedSourceType = info.symbol === 'XAUUSD' ? 'gold-proxy' : 'binance';
    try {
      if (tfMeta.binanceInterval) {
        const candles = await fetchBinanceKlines(
          info.binanceSymbol,
          tfMeta.binanceInterval,
          Math.min(1000, tfMeta.defaultBars)
        );
        return { candles, source: resolvedSource, warning: tdWarning };
      }

      if (tfMeta.seconds < 60) {
        const raw1s = await fetchBinanceKlines(info.binanceSymbol, '1s', 1000);
        const aggregated = aggregateCandles(raw1s, tfMeta.seconds);
        if (aggregated.length > 0) {
          const neededPrefix = Math.max(0, tfMeta.defaultBars - aggregated.length);
          if (neededPrefix > 0) {
            const prefix = generateSyntheticCandles(
              info,
              timeframe,
              neededPrefix,
              aggregated[0].time - tfMeta.seconds,
              aggregated[0].open
            );
            const dedup = new Map<number, Candle>();
            for (const b of [...prefix, ...aggregated]) dedup.set(b.time, b);
            return {
              candles: Array.from(dedup.values()).sort((a, b) => a.time - b.time),
              source: resolvedSource,
              warning: tdWarning,
            };
          }
          return { candles: aggregated, source: resolvedSource, warning: tdWarning };
        }
      }

      if (timeframe === '45m') {
        const raw15m = await fetchBinanceKlines(info.binanceSymbol, '15m', 1000);
        const aggregated = aggregateCandles(raw15m, tfMeta.seconds);
        if (aggregated.length > 0) {
          return { candles: aggregated, source: resolvedSource, warning: tdWarning };
        }
      }
    } catch {
      // Fall through
    }
  }

  const fxAnchor = await fetchLiveFxSpotAnchor(info);
  return {
    candles: generateSyntheticCandles(
      info,
      timeframe,
      tfMeta.defaultBars,
      undefined,
      fxAnchor ?? info.basePrice
    ),
    source: 'simulated',
    warning: tdWarning,
  };
}

export function subscribeLiveTicks(
  symbol: string,
  timeframe: TimeframeId,
  initialLastCandle: Candle,
  onTick: (candle: Candle, source: FeedSourceType) => void,
  feedConfig?: DataFeedConfig
): () => void {
  const info = getSymbolInfo(symbol);
  const tfSec = TIMEFRAME_MAP[timeframe].seconds;
  const tdKey = feedConfig?.twelveDataApiKey?.trim() || '';

  let disposed = false;
  let currentBar: Candle = { ...initialLastCandle };
  let ws: WebSocket | null = null;
  let simTimer: number | null = null;
  let wsHandshakeTimeout: number | null = null;

  const processTrade = (
    price: number,
    qty: number,
    tradeTimeSec: number,
    source: FeedSourceType
  ) => {
    if (disposed) return;
    const roundedPrice = roundToPrecision(price, info.precision);
    const bucketTime = Math.floor(tradeTimeSec / tfSec) * tfSec;

    if (bucketTime > currentBar.time) {
      currentBar = {
        time: bucketTime,
        open: roundedPrice,
        high: roundedPrice,
        low: roundedPrice,
        close: roundedPrice,
        volume: qty,
      };
    } else {
      currentBar = {
        ...currentBar,
        high: Math.max(currentBar.high, roundedPrice),
        low: Math.min(currentBar.low, roundedPrice),
        close: roundedPrice,
        volume: currentBar.volume + qty,
      };
    }
    onTick(currentBar, source);
  };

  const startSimulator = () => {
    if (disposed || simTimer !== null) return;
    const intervalMs = tfSec <= 15 ? 350 : 750;
    const tickScale = Math.sqrt(intervalMs / 1000 / 86400);
    const tickVol = Math.max(
      info.volatility * tickScale * 1.4,
      info.minMove / Math.max(currentBar.close, 1)
    );

    simTimer = window.setInterval(() => {
      if (disposed) return;
      const noise = (Math.random() - 0.496) * 2.1;
      const nextPrice = Math.max(info.minMove, currentBar.close * (1 + noise * tickVol));
      const qty = Math.round(
        (info.category === 'Crypto' ? 1.5 : info.category === 'IDX' ? 2500 : 120) *
          (0.2 + Math.random() * 1.8)
      );
      processTrade(nextPrice, qty, Math.floor(Date.now() / 1000), 'simulated');
    }, intervalMs);
  };

  const useTwelveDataWs =
    Boolean(tdKey) &&
    Boolean(info.twelveDataSymbol) &&
    (feedConfig?.preferTwelveData || !info.binanceSymbol || info.symbol === 'XAUUSD');

  if (useTwelveDataWs && info.twelveDataSymbol) {
    try {
      ws = new WebSocket(`${TWELVE_WS_BASE}?apikey=${encodeURIComponent(tdKey)}`);
      let receivedPrice = false;
      wsHandshakeTimeout = window.setTimeout(() => {
        if (!receivedPrice && !disposed) {
          ws?.close();
          startSimulator();
        }
      }, 5000);

      ws.onopen = () => {
        ws?.send(
          JSON.stringify({
            action: 'subscribe',
            params: { symbols: info.twelveDataSymbol },
          })
        );
      };

      ws.onmessage = (ev) => {
        if (disposed) return;
        try {
          const msg = JSON.parse(ev.data as string) as {
            event?: string;
            price?: number;
            timestamp?: number;
            day_volume?: number;
          };
          if (msg.event === 'price' && typeof msg.price === 'number' && msg.price > 0) {
            receivedPrice = true;
            if (wsHandshakeTimeout) {
              window.clearTimeout(wsHandshakeTimeout);
              wsHandshakeTimeout = null;
            }
            processTrade(
              msg.price,
              msg.day_volume ? 10 : 5,
              msg.timestamp ?? Math.floor(Date.now() / 1000),
              'twelvedata'
            );
          }
        } catch {
          // ignore
        }
      };

      ws.onerror = () => {
        if (!disposed && !receivedPrice) startSimulator();
      };
      ws.onclose = () => {
        if (!disposed && !receivedPrice) startSimulator();
      };
    } catch {
      startSimulator();
    }
  } else if (
    info.binanceSymbol &&
    (info.symbol !== 'XAUUSD' || feedConfig?.useGoldProxyWhenNoKey !== false)
  ) {
    const resolvedSource: FeedSourceType = info.symbol === 'XAUUSD' ? 'gold-proxy' : 'binance';
    try {
      const streamName = `${info.binanceSymbol.toLowerCase()}@trade`;
      ws = new WebSocket(`${BINANCE_WS_BASE}/${streamName}`);

      let receivedMessage = false;
      wsHandshakeTimeout = window.setTimeout(() => {
        if (!receivedMessage && !disposed) {
          ws?.close();
          startSimulator();
        }
      }, 3800);

      ws.onmessage = (event) => {
        if (disposed) return;
        receivedMessage = true;
        if (wsHandshakeTimeout) {
          window.clearTimeout(wsHandshakeTimeout);
          wsHandshakeTimeout = null;
        }
        try {
          const msg = JSON.parse(event.data as string) as {
            p?: string;
            q?: string;
            T?: number;
          };
          if (msg.p) {
            const p = Number(msg.p);
            const q = Number(msg.q ?? 0);
            const tSec = Math.floor((msg.T ?? Date.now()) / 1000);
            if (Number.isFinite(p) && p > 0) {
              processTrade(p, Number.isFinite(q) ? q : 0, tSec, resolvedSource);
            }
          }
        } catch {
          // ignore
        }
      };

      ws.onerror = () => {
        if (!disposed) startSimulator();
      };
      ws.onclose = () => {
        if (!disposed && !receivedMessage) startSimulator();
      };
    } catch {
      startSimulator();
    }
  } else {
    startSimulator();
  }

  return () => {
    disposed = true;
    if (wsHandshakeTimeout !== null) window.clearTimeout(wsHandshakeTimeout);
    if (simTimer !== null) window.clearInterval(simTimer);
    if (ws && (ws.readyState === WebSocket.OPEN || ws.readyState === WebSocket.CONNECTING)) {
      ws.close();
    }
  };
}

export function generateOrderBookSnapshot(
  symbol: string,
  midPrice: number,
  levels = 12
): { asks: OrderBookLevel[]; bids: OrderBookLevel[]; spread: number; spreadPct: number } {
  const info = getSymbolInfo(symbol);
  const step = Math.max(info.minMove, midPrice * 0.00025);

  const asksRaw: Omit<OrderBookLevel, 'pct'>[] = [];
  const bidsRaw: Omit<OrderBookLevel, 'pct'>[] = [];

  let askCum = 0;
  let bidCum = 0;

  for (let i = 1; i <= levels; i++) {
    const askPrice = roundToPrecision(
      midPrice + i * step * (0.8 + Math.random() * 0.4),
      info.precision
    );
    const bidPrice = roundToPrecision(
      Math.max(info.minMove, midPrice - i * step * (0.8 + Math.random() * 0.4)),
      info.precision
    );
    const baseSize = info.category === 'Crypto' ? 2.5 : info.category === 'IDX' ? 8500 : 450;
    const askSize = Number(
      (baseSize * (0.2 + Math.random() * 2.4)).toFixed(info.category === 'Crypto' ? 3 : 0)
    );
    const bidSize = Number(
      (baseSize * (0.2 + Math.random() * 2.4)).toFixed(info.category === 'Crypto' ? 3 : 0)
    );

    askCum += askSize;
    bidCum += bidSize;

    asksRaw.push({ price: askPrice, size: askSize, total: Number(askCum.toFixed(2)) });
    bidsRaw.push({ price: bidPrice, size: bidSize, total: Number(bidCum.toFixed(2)) });
  }

  const maxTotal = Math.max(askCum, bidCum, 1);
  const asks = asksRaw
    .map((a) => ({ ...a, pct: Math.min(100, (a.total / maxTotal) * 100) }))
    .reverse();
  const bids = bidsRaw.map((b) => ({ ...b, pct: Math.min(100, (b.total / maxTotal) * 100) }));

  const bestAsk = asks[asks.length - 1]?.price ?? midPrice;
  const bestBid = bids[0]?.price ?? midPrice;
  const spread = Math.max(0, bestAsk - bestBid);
  const spreadPct = midPrice > 0 ? (spread / midPrice) * 100 : 0;

  return { asks, bids, spread, spreadPct };
}
