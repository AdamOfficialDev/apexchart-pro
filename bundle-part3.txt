interface DrawingOverlayProps {

interface ScreenPoint {

  y: number;
}

const FIB_LEVELS = [
  { level: 0, color: '#64748B', fill: 'rgba(100,116,139,0.06)' },
  { level: 0.236, color: '#EF4444', fill: 'rgba(239,68,68,0.07)' },
  { level: 0.382, color: '#F59E0B', fill: 'rgba(245,158,11,0.07)' },

  { level: 0.618, color: '#06B6D4', fill: 'rgba(6,182,212,0.09)' },
  { level: 0.786, color: '#3B82F6', fill: 'rgba(59,130,246,0.07)' },
  { level: 1, color: '#8B5CF6', fill: 'rgba(139,92,246,0.06)' },
];

function distToSegment(p: ScreenPoint, a: ScreenPoint, b: ScreenPoint): number {
  const dx = b.x - a.x;
  const dy = b.y - a.y;
  const lenSq = dx * dx + dy * dy;
  if (lenSq === 0) return Math.hypot(p.x - a.x, p.y - a.y);
  const t = Math.max(0, Math.min(1, ((p.x - a.x) * dx + (p.y - a.y) * dy) / lenSq));
  return Math.hypot(p.x - (a.x + t * dx), p.y - (a.y + t * dy));
}

      const last = candles[candles.length - 1].time;

      } else if (time >= last) {

        // Binary search nearest candle
        let lo = 0;
        let hi = candles.length - 1;
        while (lo <= hi) {

          if (candles[mid].time === time) {
            logical = mid;
            const coord = chart.timeScale().logicalToCoordinate(logical as Logical);
            return coord !== null ? Number(coord) : null;

          if (candles[mid].time < time) lo = mid + 1;
          else hi = mid - 1;

      const coord = chart.timeScale().logicalToCoordinate(logical as Logical);

        time = candles[candles.length - 1].time + Math.round(idx - (candles.length - 1)) * tfSec;
      } else {
        const i0 = Math.floor(idx);
        const i1 = Math.min(candles.length - 1, i0 + 1);
        const frac = idx - i0;
        time = Math.round(candles[i0].time + frac * (candles[i1].time - candles[i0].time));

                bestDist = d;
                snappedPrice = candidate;
              }
            }
          }
        }
      }

      return { time, price: snappedPrice };
    },

import { Candle, ChartPaneConfig } from '@/types/trading';
import { DrawingOverlay } from './DrawingOverlay';
import { SubPaneChart } from './SubPaneChart';

interface ChartPaneProps {
  pane: ChartPaneConfig;
  isActive: boolean;
  isMultiPane: boolean;
}

type MainSeriesApi = ISeriesApi<'Candlestick' | 'Bar' | 'Line' | 'Area'>;

export const ChartPane: React.FC<ChartPaneProps> = ({ pane, isActive, isMultiPane }) => {
  const theme = useTradingStore((s) => s.theme);
  const setActivePaneId = useTradingStore((s) => s.setActivePaneId);
  const toggleIndicatorVisibility = useTradingStore((s) => s.toggleIndicatorVisibility);
  const removeIndicator = useTradingStore((s) => s.removeIndicator);
  const updateQuote = useTradingStore((s) => s.updateQuote);
  const processPaper = useTradingStore((s) => s.processMarketPriceForPaperTrading);

  const syncCrosshair = useTradingStore((s) => s.syncCrosshair);
  const syncedCrosshairTime = useTradingStore((s) => s.syncedCrosshairTime);
  const setSyncedCrosshairTime = useTradingStore((s) => s.setSyncedCrosshairTime);

  const replay = useTradingStore((s) => s.replay);
  const startReplayAt = useTradingStore((s) => s.startReplayAt);
  const setReplayTotalBars = useTradingStore((s) => s.setReplayTotalBars);
  const setReplayIndex = useTradingStore((s) => s.setReplayIndex);

  const alerts = useTradingStore((s) => s.alerts);
  const positions = useTradingStore((s) => s.positions);
  const snapshotCounter = useTradingStore((s) => s.snapshotRequestCounter);
  const addToast = useTradingStore((s) => s.addToast);

  const symbolInfo = useMemo(() => getSymbolInfo(pane.symbol), [pane.symbol]);
  const tfMeta = TIMEFRAME_MAP[pane.timeframe];

  const mainContainerRef = useRef<HTMLDivElement | null>(null);
  const chartRef = useRef<IChartApi | null>(null);
  const mainSeriesRef = useRef<MainSeriesApi | null>(null);
  const overlaySeriesRef = useRef<Map<string, ISeriesApi<'Line' | 'Histogram'>[]>>(new Map());
  const priceLinesRef = useRef<IPriceLine[]>([]);
  const drawingCanvasRef = useRef<HTMLCanvasElement | null>(null);
  const isExternalCrosshairRef = useRef(false);

  const [chartSize, setChartSize] = useState({ width: 0, height: 0 });
  const [fullCandles, setFullCandles] = useState<Candle[]>([]);
  const [dataSource, setDataSource] = useState<'binance' | 'simulated'>('simulated');
  const [loading, setLoading] = useState(true);
  const [hoveredBar, setHoveredBar] = useState<Candle | null>(null);

  const isThisPaneInReplay = replay.enabled && replay.paneId === pane.id;

  // Displayed candles (sliced if Replay mode is active on this pane)
  const displayCandles = useMemo(() => {
    if (isThisPaneInReplay && !replay.isSelectingStart && fullCandles.length > 0) {
      const end = Math.min(fullCandles.length, Math.max(5, replay.currentIndex + 1));
      return fullCandles.slice(0, end);
    }
    return fullCandles;
  }, [fullCandles, isThisPaneInReplay, replay.isSelectingStart, replay.currentIndex]);

  // 1. Initialize main lightweight-charts instance
  useEffect(() => {
    const el = mainContainerRef.current;
    if (!el) return;

    const isDark = theme === 'dark';
    const chart = createChart(el, {
      width: el.clientWidth,
      height: el.clientHeight,
      layout: {
        background: { type: ColorType.Solid, color: isDark ? '#0B0E14' : '#FFFFFF' },
        textColor: isDark ? '#94A3B8' : '#475569',
        fontFamily: '"JetBrains Mono", monospace',
        fontSize: 11,
      },
      grid: {
        vertLines: { color: isDark ? 'rgba(255,255,255,0.038)' : 'rgba(15,23,42,0.055)' },
        horzLines: { color: isDark ? 'rgba(255,255,255,0.038)' : 'rgba(15,23,42,0.055)' },
      },
      crosshair: {
        mode: CrosshairMode.Normal,
        vertLine: {
          color: isDark ? 'rgba(148,163,184,0.45)' : 'rgba(71,85,105,0.45)',
          labelBackgroundColor: '#2962FF',
        },
        horzLine: {
          color: isDark ? 'rgba(148,163,184,0.45)' : 'rgba(71,85,105,0.45)',
          labelBackgroundColor: '#2962FF',
        },
      },
      rightPriceScale: {
        borderColor: isDark ? 'rgba(255,255,255,0.075)' : 'rgba(15,23,42,0.1)',
        scaleMargins: { top: 0.1, bottom: 0.18 },
      },
      timeScale: {
        borderColor: isDark ? 'rgba(255,255,255,0.075)' : 'rgba(15,23,42,0.1)',
        timeVisible: true,
        secondsVisible: tfMeta.seconds < 60,
        rightOffset: 8,
        barSpacing: 8,
      },
      handleScroll: { vertTouchDrag: true },
      handleScale: { axisPressedMouseMove: true },
    });

    chartRef.current = chart;
    setChartSize({ width: el.clientWidth, height: el.clientHeight });

    const ro = new ResizeObserver((entries) => {
      for (const entry of entries) {
        const { width, height } = entry.contentRect;
        if (width > 0 && height > 0) {
          chart.applyOptions({ width, height });
          setChartSize({ width, height });
        }
      }
    });
    ro.observe(el);

    return () => {
      ro.disconnect();
      chart.remove();
      chartRef.current = null;
      mainSeriesRef.current = null;
      overlaySeriesRef.current.clear();
      priceLinesRef.current = [];
    };
  }, [theme]);

  // Update secondsVisible & scale mode dynamically without destroying chart
  useEffect(() => {
    const chart = chartRef.current;
    if (!chart) return;
    chart.applyOptions({
      timeScale: {
        timeVisible: true,
        secondsVisible: tfMeta.seconds < 60,
      },
      rightPriceScale: {
        mode:
          pane.scaleMode === 'logarithmic'
            ? PriceScaleMode.Logarithmic
            : pane.scaleMode === 'percentage'
            ? PriceScaleMode.Percentage
            : PriceScaleMode.Normal,
      },
    });
  }, [tfMeta.seconds, pane.scaleMode]);

  // 2. Fetch historical candles when symbol or timeframe changes
  useEffect(() => {
    let cancelled = false;
    setLoading(true);
    setHoveredBar(null);

    fetchHistoricalCandles(pane.symbol, pane.timeframe).then(({ candles, source }) => {
      if (cancelled) return;
      setFullCandles(candles);
      setDataSource(source);
      setLoading(false);

      if (candles.length > 0) {
        const last = candles[candles.length - 1];
        updateQuote(pane.symbol, last.close, source, last.volume);
      }

      if (useTradingStore.getState().replay.enabled && useTradingStore.getState().replay.paneId === pane.id) {
        setReplayTotalBars(candles.length);
      }
    });

    return () => {
      cancelled = true;
    };
  }, [pane.symbol, pane.timeframe, pane.id, updateQuote, setReplayTotalBars]);

  // Sync replay total bars when replay turns on
  useEffect(() => {
    if (isThisPaneInReplay && fullCandles.length > 0) {
      setReplayTotalBars(fullCandles.length);
    }
  }, [isThisPaneInReplay, fullCandles.length, setReplayTotalBars]);

  // 3. Subscribe to Live Ticks (paused during Replay mode on this pane)
  useEffect(() => {
    if (loading || fullCandles.length === 0 || isThisPaneInReplay) return;
    const lastCandle = fullCandles[fullCandles.length - 1];

    const unsubscribe = subscribeLiveTicks(
      pane.symbol,
      pane.timeframe,
      lastCandle,
      (tickCandle, source) => {
        setDataSource(source);
        setFullCandles((prev) => {
          if (prev.length === 0) return [tickCandle];
          const tail = prev[prev.length - 1];
          if (tickCandle.time === tail.time) {
            const copy = prev.slice();
            copy[copy.length - 1] = tickCandle;
            return copy;
          }
          if (tickCandle.time > tail.time) {
            return [...prev.slice(-799), tickCandle];
          }
          return prev;
        });

        updateQuote(pane.symbol, tickCandle.close, source, tickCandle.volume);
        processPaper(pane.symbol, tickCandle.high, tickCandle.low, tickCandle.close);
      }
    );

    return unsubscribe;
  }, [pane.symbol, pane.timeframe, loading, isThisPaneInReplay, fullCandles.length > 0]);

  // Notify paper trading & quote during Replay steps
  useEffect(() => {
    if (!isThisPaneInReplay || replay.isSelectingStart || displayCandles.length === 0) return;
    const cur = displayCandles[displayCandles.length - 1];
    updateQuote(pane.symbol, cur.close, 'simulated', cur.volume);
    processPaper(pane.symbol, cur.high, cur.low, cur.close);
    if (replay.currentTimestamp !== cur.time) {
      setReplayIndex(replay.currentIndex, cur.time);
    }
  }, [isThisPaneInReplay, replay.isSelectingStart, displayCandles, pane.symbol]);

  // 4. Create / recreate Main Price Series when chartStyle or symbol precision changes
  useEffect(() => {
    const chart = chartRef.current;
    if (!chart) return;

    if (mainSeriesRef.current) {
      try {
        chart.removeSeries(mainSeriesRef.current);
      } catch {
        // ignore
      }
      mainSeriesRef.current = null;
      priceLinesRef.current = [];
    }

    const priceFormat = {
      type: 'price' as const,
      precision: symbolInfo.precision,
      minMove: symbolInfo.minMove,
    };

    let s: MainSeriesApi;
    if (pane.chartStyle === 'hollow') {
      s = chart.addCandlestickSeries({
        upColor: 'transparent',
        downColor: '#F23645',
        borderUpColor: '#089981',
        borderDownColor: '#F23645',
        wickUpColor: '#089981',
        wickDownColor: '#F23645',
        priceFormat,
      });
    } else if (pane.chartStyle === 'bars') {
      s = chart.addBarSeries({
        upColor: '#089981',
        downColor: '#F23645',
        priceFormat,
      });
    } else if (pane.chartStyle === 'line') {
      s = chart.addLineSeries({
        color: '#2962FF',
        lineWidth: 2,
        priceFormat,
      });
    } else if (pane.chartStyle === 'area') {
      s = chart.addAreaSeries({
        lineColor: '#2962FF',
        topColor: 'rgba(41, 98, 255, 0.34)',
        bottomColor: 'rgba(41, 98, 255, 0.02)',
        lineWidth: 2,
        priceFormat,
      });
    } else {
      // 'candles' or 'heikin_ashi'
      s = chart.addCandlestickSeries({
        upColor: '#089981',
        downColor: '#F23645',
        borderVisible: false,
        wickUpColor: '#089981',
        wickDownColor: '#F23645',
        priceFormat,
      });
    }

    mainSeriesRef.current = s;
  }, [pane.chartStyle, symbolInfo.precision, symbolInfo.minMove, theme]);

  // 5. Feed price data & overlay indicators into chart
  useEffect(() => {
    const chart = chartRef.current;
    const series = mainSeriesRef.current;
    if (!chart || !series || displayCandles.length === 0) return;

    const effectiveCandles =
      pane.chartStyle === 'heikin_ashi' ? computeHeikinAshi(displayCandles) : displayCandles;

    if (pane.chartStyle === 'line' || pane.chartStyle === 'area') {
      (series as ISeriesApi<'Line' | 'Area'>).setData(
        effectiveCandles.map((c) => ({ time: c.time as UTCTimestamp, value: c.close }))
      );
    } else {
      (series as ISeriesApi<'Candlestick' | 'Bar'>).setData(
        effectiveCandles.map((c) => ({
          time: c.time as UTCTimestamp,
          open: c.open,
          high: c.high,
          low: c.low,
          close: c.close,
        }))
      );
    }

    // Clear old overlay indicator series and re-render current overlays
    for (const list of overlaySeriesRef.current.values()) {
      for (const s of list) {
        try {
          chart.removeSeries(s);
        } catch {
          // ignore
        }
      }
    }
    overlaySeriesRef.current.clear();

    const overlays = pane.indicators.filter((i) => i.placement === 'overlay' && i.visible);
    for (const ind of overlays) {
      const created: ISeriesApi<'Line' | 'Histogram'>[] = [];

      if (ind.type === 'VOLUME') {
        const volSeries = chart.addHistogramSeries({
          priceFormat: { type: 'volume' },
          priceScaleId: '',
          lastValueVisible: false,
          priceLineVisible: false,
        });
        volSeries.priceScale().applyOptions({
          scaleMargins: { top: 0.84, bottom: 0 },
        });
        volSeries.setData(
          displayCandles.map((c) => ({
            time: c.time as UTCTimestamp,
            value: c.volume,
            color:
              c.close >= c.open ? 'rgba(8, 153, 129, 0.35)' : 'rgba(242, 54, 69, 0.35)',
          }))
        );
        created.push(volSeries);
      } else if (ind.type === 'SMA') {
        const pts = computeSMA(displayCandles, ind.params.period ?? 50);
        const l = chart.addLineSeries({
          color: ind.color,
          lineWidth: 2,
          priceLineVisible: false,
          lastValueVisible: false,
        });
        l.setData(pts.map((p) => ({ time: p.time as UTCTimestamp, value: p.value })));
        created.push(l);
      } else if (ind.type === 'EMA') {
        const pts = computeEMA(displayCandles, ind.params.period ?? 21);
        const l = chart.addLineSeries({
          color: ind.color,
          lineWidth: 2,
          priceLineVisible: false,
          lastValueVisible: false,
        });
        l.setData(pts.map((p) => ({ time: p.time as UTCTimestamp, value: p.value })));
        created.push(l);
      } else if (ind.type === 'VWAP') {
        const pts = computeVWAP(displayCandles);
        const l = chart.addLineSeries({
          color: ind.color,
          lineWidth: 2,
          lineStyle: LineStyle.Solid,
          priceLineVisible: false,
          lastValueVisible: false,
        });
        l.setData(pts.map((p) => ({ time: p.time as UTCTimestamp, value: p.value })));
        created.push(l);
      } else if (ind.type === 'BB') {
        const bb = computeBollingerBands(
          displayCandles,
          ind.params.period ?? 20,
          ind.params.stdDev ?? 2
        );
        const up = chart.addLineSeries({
          color: ind.color,
          lineWidth: 1,
          priceLineVisible: false,
          lastValueVisible: false,
        });
        const mid = chart.addLineSeries({
          color: ind.secondaryColor ?? '#64748B',
          lineWidth: 1,
          lineStyle: LineStyle.Dashed,
          priceLineVisible: false,
          lastValueVisible: false,
        });
        const low = chart.addLineSeries({
          color: ind.color,
          lineWidth: 1,
          priceLineVisible: false,
          lastValueVisible: false,
        });
        up.setData(bb.upper.map((p) => ({ time: p.time as UTCTimestamp, value: p.value })));
        mid.setData(bb.middle.map((p) => ({ time: p.time as UTCTimestamp, value: p.value })));
        low.setData(bb.lower.map((p) => ({ time: p.time as UTCTimestamp, value: p.value })));
        created.push(up, mid, low);
      } else if (ind.type === 'DONCHIAN') {
        const dc = computeDonchian(displayCandles, ind.params.period ?? 20);
        const up = chart.addLineSeries({
          color: ind.color,
          lineWidth: 1,
          priceLineVisible: false,
          lastValueVisible: false,
        });
        const low = chart.addLineSeries({
          color: ind.color,
          lineWidth: 1,
          priceLineVisible: false,
          lastValueVisible: false,
        });
        up.setData(dc.upper.map((p) => ({ time: p.time as UTCTimestamp, value: p.value })));
        low.setData(dc.lower.map((p) => ({ time: p.time as UTCTimestamp, value: p.value })));
        created.push(up, low);
      } else if (ind.type === 'SUPERTREND') {
        const st = computeSupertrend(
          displayCandles,
          ind.params.period ?? 10,
          ind.params.multiplier ?? 3,
          ind.color,
          ind.secondaryColor ?? '#F43F5E'
        );
        const l = chart.addLineSeries({
          lineWidth: 2,
          priceLineVisible: false,
          lastValueVisible: false,
        });
        l.setData(
          st.map((p) => ({ time: p.time as UTCTimestamp, value: p.value, color: p.color }))
        );
        created.push(l);
      }

      overlaySeriesRef.current.set(ind.id, created);
    }
  }, [displayCandles, pane.chartStyle, pane.indicators, theme]);

  // 6. Plot active Price Alerts & Paper Trading Entry / SL / TP Price Lines
  useEffect(() => {
    const series = mainSeriesRef.current;
    if (!series) return;

    for (const pl of priceLinesRef.current) {
      try {
        series.removePriceLine(pl);
      } catch {
        // ignore
      }
    }
    priceLinesRef.current = [];

    const symAlerts = alerts.filter((a) => a.active && a.symbol === pane.symbol);
    for (const a of symAlerts) {
      priceLinesRef.current.push(
        series.createPriceLine({
          price: a.price,
          color: '#F59E0B',
          lineWidth: 1,
          lineStyle: LineStyle.Dashed,
          axisLabelVisible: true,
          title: 'Alert',
        })
      );
    }

    const symPositions = positions.filter((p) => p.symbol === pane.symbol);
    for (const pos of symPositions) {
      priceLinesRef.current.push(
        series.createPriceLine({
          price: pos.entryPrice,
          color: pos.side === 'buy' ? '#00D4AA' : '#F43F5E',
          lineWidth: 2,
          lineStyle: LineStyle.Solid,
          axisLabelVisible: true,
          title: `${pos.side === 'buy' ? 'LONG' : 'SHORT'} ${pos.quantity}`,
        })
      );
      if (pos.stopLoss) {
        priceLinesRef.current.push(
          series.createPriceLine({
            price: pos.stopLoss,
            color: '#F43F5E',
            lineWidth: 1,
            lineStyle: LineStyle.Dotted,
            axisLabelVisible: true,
            title: 'SL',
          })
        );
      }
      if (pos.takeProfit) {
        priceLinesRef.current.push(
          series.createPriceLine({
            price: pos.takeProfit,
            color: '#10B981',
            lineWidth: 1,
            lineStyle: LineStyle.Dotted,
            axisLabelVisible: true,
            title: 'TP',
          })
        );
      }
    }
  }, [alerts, positions, pane.symbol, pane.chartStyle]);

  // 7. Crosshair Readout + Crosshair Sync + Replay Click-to-Cut Handler
  useEffect(() => {
    const chart = chartRef.current;
    if (!chart) return;

    const candleByTime = new Map<number, Candle>();
    for (const c of displayCandles) candleByTime.set(c.time, c);

    const onCrosshair = (param: { time?: Time }) => {
      if (param.time !== undefined) {
        const t = Number(param.time);
        const found = candleByTime.get(t);
        if (found) setHoveredBar(found);
        if (syncCrosshair && !isExternalCrosshairRef.current) {
          setSyncedCrosshairTime(t);
        }
      } else {
        setHoveredBar(null);
      }
    };

    const onClick = (param: { time?: Time }) => {
      setActivePaneId(pane.id);
      if (isThisPaneInReplay && replay.isSelectingStart && param.time !== undefined) {
        const clickedTime = Number(param.time);
        const idx = fullCandles.findIndex((c) => c.time >= clickedTime);
        if (idx >= 0) {
          startReplayAt(pane.id, idx, fullCandles.length, fullCandles[idx].time);
        }
      }
    };

    chart.subscribeCrosshairMove(onCrosshair);
    chart.subscribeClick(onClick);
    return () => {
      chart.unsubscribeCrosshairMove(onCrosshair);
      chart.unsubscribeClick(onClick);
    };
  }, [
    displayCandles,
    fullCandles,
    syncCrosshair,
    isThisPaneInReplay,
    replay.isSelectingStart,
    pane.id,
    setActivePaneId,
    setSyncedCrosshairTime,
    startReplayAt,
  ]);

  // Receive external Crosshair Sync from another pane
  useEffect(() => {
    const chart = chartRef.current;
    const series = mainSeriesRef.current;
    if (!chart || !series || !syncCrosshair || isActive || syncedCrosshairTime === null) return;

    const bar = displayCandles.find((c) => c.time === syncedCrosshairTime);
    if (!bar) return;

    isExternalCrosshairRef.current = true;
    try {
      chart.setCrosshairPosition(bar.close, bar.time as UTCTimestamp, series);
      setHoveredBar(bar);
    } catch {
      // ignore if out of range
    }
    isExternalCrosshairRef.current = false;
  }, [syncedCrosshairTime, syncCrosshair, isActive, displayCandles]);

  // 8. Export High-Res Chart Snapshot (Chart + Drawing Overlay + Watermark)
  const prevSnapshotRef = useRef(snapshotCounter);
  useEffect(() => {
    if (snapshotCounter === prevSnapshotRef.current) return;
    prevSnapshotRef.current = snapshotCounter;
    if (!isActive || !chartRef.current) return;

    try {
      const chartCanvas = chartRef.current.takeScreenshot();
      const out = document.createElement('canvas');
      out.width = chartCanvas.width;
      out.height = chartCanvas.height;
      const ctx = out.getContext('2d');
      if (!ctx) return;

      ctx.drawImage(chartCanvas, 0, 0);
      if (drawingCanvasRef.current) {
        ctx.drawImage(drawingCanvasRef.current, 0, 0, out.width, out.height);
      }

      // Watermark
      ctx.font = '600 13px Inter, sans-serif';
      ctx.fillStyle = theme === 'dark' ? 'rgba(255,255,255,0.45)' : 'rgba(15,23,42,0.45)';
      ctx.fillText(
        `ApexChart Pro  •  ${symbolInfo.displaySymbol}  •  ${pane.timeframe}`,
        18,
        out.height - 16
      );

      const url = out.toDataURL('image/png');
      const a = document.createElement('a');
      a.href = url;
      a.download = `${pane.symbol}_${pane.timeframe}_${Date.now()}.png`;
      a.click();

      addToast({
        title: 'Chart Snapshot Saved',
        description: `${symbolInfo.displaySymbol} (${pane.timeframe}) downloaded as PNG.`,
        variant: 'success',
      });
    } catch {
      addToast({ title: 'Snapshot Failed', variant: 'danger' });
    }
  }, [snapshotCounter, isActive, pane.symbol, pane.timeframe, symbolInfo.displaySymbol, theme, addToast]);

  const activeBar = hoveredBar ?? displayCandles[displayCandles.length - 1] ?? null;
  const barChange = activeBar ? activeBar.close - activeBar.open : 0;
  const barChangePct = activeBar && activeBar.open > 0 ? (barChange / activeBar.open) * 100 : 0;
  const isBull = barChange >= 0;

  const subPaneIndicators = pane.indicators.filter((i) => i.placement === 'subpane');
  const overlayIndicators = pane.indicators.filter((i) => i.placement === 'overlay');

  return (
    <div
      onMouseDown={() => setActivePaneId(pane.id)}
      className={cn(
        'relative flex h-full w-full flex-col overflow-hidden bg-white dark:bg-[#0B0E14]',
        isMultiPane && isActive
          ? 'ring-1 ring-inset ring-blue-500/80'
          : isMultiPane
          ? 'ring-1 ring-inset ring-slate-200 dark:ring-white/[0.06]'
          : ''
      )}
    >
      {/* Main Chart + Drawing Overlay */}
      <div className="relative flex-1 min-h-0 w-full">
        {/* Top-Left Floating OHLC & Unlimited Indicators HUD */}
        <div className="pointer-events-none absolute left-3 top-2.5 z-20 flex max-w-[82%] flex-col gap-1">
          <div className="pointer-events-auto flex flex-wrap items-center gap-2 rounded-lg border border-slate-200/80 bg-white/85 px-2.5 py-1 text-xs shadow-sm backdrop-blur dark:border-white/[0.08] dark:bg-obsidian-900/80">
            <span className="font-bold tracking-tight text-slate-900 dark:text-white">
              {symbolInfo.displaySymbol}
            </span>
            <span className="rounded bg-slate-100 px-1.5 py-0.5 text-[10px] font-medium text-slate-600 dark:bg-white/[0.06] dark:text-slate-300">
              {pane.timeframe}
            </span>
            <span className="text-[10px] text-slate-400">{symbolInfo.exchange}</span>

            {/* Feed badge */}
            {isThisPaneInReplay ? (
              <span className="inline-flex items-center gap-1 rounded bg-amber-500/15 px-1.5 py-0.5 text-[10px] font-semibold text-amber-400">
                <span className="h-1.5 w-1.5 animate-pulse rounded-full bg-amber-400" />
                REPLAY
              </span>
            ) : dataSource === 'binance' ? (
              <span className="inline-flex items-center gap-1 rounded bg-emerald-500/15 px-1.5 py-0.5 text-[10px] font-semibold text-emerald-400">
                <span className="h-1.5 w-1.5 animate-pulse rounded-full bg-emerald-400" />
                LIVE WS
              </span>
            ) : (

import React, { useEffect } from 'react';
import {
  Pause,
  Play,
  Scissors,
  SkipBack,
  SkipForward,
  X,
} from 'lucide-react';
import { cn, formatTimestamp } from '@/lib/utils';
import { useTradingStore } from '@/store/useTradingStore';

const SPEEDS: Array<0.5 | 1 | 2 | 5 | 10> = [0.5, 1, 2, 5, 10];

export const ReplayControls: React.FC = () => {
  const replay = useTradingStore((s) => s.replay);
  const toggleReplayPlay = useTradingStore((s) => s.toggleReplayPlay);
  const stepForward = useTradingStore((s) => s.stepReplayForward);
  const stepBackward = useTradingStore((s) => s.stepReplayBackward);
  const setSpeed = useTradingStore((s) => s.setReplaySpeed);
  const setSelectMode = useTradingStore((s) => s.setReplaySelectMode);
  const setReplayIndex = useTradingStore((s) => s.setReplayIndex);
  const exitReplayMode = useTradingStore((s) => s.exitReplayMode);

  // Playback ticker
  useEffect(() => {
    if (!replay.enabled || !replay.isPlaying || replay.isSelectingStart) return;
    const intervalMs = Math.max(90, Math.round(1000 / replay.speed));
    const id = window.setInterval(() => {
      useTradingStore.getState().stepReplayForward();
    }, intervalMs);
    return () => window.clearInterval(id);
  }, [replay.enabled, replay.isPlaying, replay.isSelectingStart, replay.speed]);

  if (!replay.enabled) return null;

  const maxIdx = Math.max(5, replay.totalBars - 1);

  return (
    <div className="pointer-events-auto absolute bottom-5 left-1/2 z-30 -translate-x-1/2">
      <div className="glass-panel flex flex-wrap items-center gap-2 rounded-2xl border border-amber-500/30 bg-white/90 px-3.5 py-2 shadow-2xl dark:bg-obsidian-900/92">
        <div className="flex items-center gap-1.5 pr-1">
          <span className="h-2 w-2 animate-pulse rounded-full bg-amber-400" />
          <span className="text-[11px] font-bold tracking-wider text-amber-500 uppercase">
            Bar Replay
          </span>
        </div>

        <button
          onClick={() => setSelectMode(!replay.isSelectingStart)}
          title="Select starting bar on chart"
          className={cn(
            'inline-flex items-center gap-1.5 rounded-lg px-2.5 py-1 text-xs font-medium transition',
            replay.isSelectingStart
              ? 'bg-amber-500 text-slate-950'
              : 'bg-slate-100 text-slate-700 hover:bg-slate-200 dark:bg-white/[0.07] dark:text-slate-200 dark:hover:bg-white/[0.12]'
          )}
        >
          <Scissors className="h-3.5 w-3.5" />
          <span>Jump to Bar</span>
        </button>

        <div className="h-4 w-px bg-slate-200 dark:bg-white/10" />

        <div className="flex items-center gap-1">
          <button
            onClick={stepBackward}
            title="Step Backward (Left Arrow)"
            className="rounded-lg p-1.5 text-slate-600 hover:bg-slate-100 dark:text-slate-300 dark:hover:bg-white/[0.08]"
          >
            <SkipBack className="h-4 w-4" />
          </button>

          <button
            onClick={toggleReplayPlay}
            title="Play / Pause (Space)"
            className="inline-flex h-8 w-8 items-center justify-center rounded-lg bg-blue-600 text-white shadow-sm hover:bg-blue-500"
          >
            {replay.isPlaying ? (
              <Pause className="h-4 w-4" />
            ) : (
              <Play className="h-4 w-4 translate-x-[1px]" />
            )}
          </button>

          <button
            onClick={stepForward}
            title="Step Forward (Right Arrow)"
            className="rounded-lg p-1.5 text-slate-600 hover:bg-slate-100 dark:text-slate-300 dark:hover:bg-white/[0.08]"
          >
            <SkipForward className="h-4 w-4" />
          </button>
        </div>

        {/* Speed Selector */}
        <div className="flex items-center rounded-lg bg-slate-100 p-0.5 dark:bg-white/[0.06]">
          {SPEEDS.map((sp) => (
            <button
              key={sp}
              onClick={() => setSpeed(sp)}
              className={cn(
                'rounded-md px-2 py-0.5 font-mono text-[11px] font-semibold transition',
                replay.speed === sp
                  ? 'bg-blue-600 text-white shadow'
                  : 'text-slate-500 hover:text-slate-800 dark:text-slate-400 dark:hover:text-white'
              )}
            >
              {sp}x
            </button>
          ))}
        </div>

        {/* Scrubber & Bar Progress */}
        {replay.totalBars > 0 && (
          <div className="hidden md:flex items-center gap-2 pl-1">
            <input
              type="range"
              min={5}
              max={maxIdx}
              value={Math.min(maxIdx, Math.max(5, replay.currentIndex))}
              onChange={(e) => setReplayIndex(Number(e.target.value))}
              className="h-1.5 w-28 cursor-pointer accent-amber-500"
            />
            <span className="font-mono text-[11px] tabular-nums text-slate-500 dark:text-slate-400">
              {replay.currentIndex + 1}/{replay.totalBars}
            </span>
            {replay.currentTimestamp && (
              <span className="hidden lg:inline font-mono text-[11px] text-slate-400">
                {formatTimestamp(replay.currentTimestamp, true)}
              </span>
            )}
          </div>
        )}

        <div className="h-4 w-px bg-slate-200 dark:bg-white/10" />

        <button
          onClick={exitReplayMode}
          title="Exit Replay Mode"
          className="inline-flex items-center gap-1 rounded-lg px-2 py-1 text-xs font-medium text-rose-500 hover:bg-rose-500/10"
        >
          <X className="h-3.5 w-3.5" />
          <span>Exit</span>
        </button>
      </div>
    </div>
  );
};
